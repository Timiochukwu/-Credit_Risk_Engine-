## Understanding Logistic Regression

**What is Logistic Regression?**
Despite the name, it's a **classification** algorithm (not regression). It predicts probabilities between 0 and 1.

**How it works:**
```
Input features → Weighted sum → Sigmoid function → Probability (0-1)
```

**Why start with Logistic Regression?**
- ✅ Simple and interpretable
- ✅ Fast to train
- ✅ Works well as baseline
- ✅ Shows which features are important
- ✅ Industry standard for credit scoring

**Example:**
```
Features: age=35, income=500k, loan=5M, debt_ratio=2.1
↓
Model calculates: P(default) = 0.23 (23% chance of default)
↓
Decision: If > 50% → Predict default, else → Predict no default
```

```python
"""
Train machine learning models for loan default prediction.
"""

import pandas as pd
import numpy as np
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import (
    accuracy_score,
    precision_score,
    recall_score,
    f1_score,
    roc_auc_score,
    confusion_matrix,
    classification_report
)
from pathlib import Path
import pickle
import json

from src.utils.logger import logger


def load_processed_data():
    """Load preprocessed training and test data."""
    logger.info("Loading preprocessed data...")

    data_dir = Path("data/processed")

    X_train = pd.read_csv(data_dir / "X_train.csv")
    X_test = pd.read_csv(data_dir / "X_test.csv")
    y_train = pd.read_csv(data_dir / "y_train.csv")['defaulted']
    y_test = pd.read_csv(data_dir / "y_test.csv")['defaulted']

    logger.info(f"Training set: {len(X_train)} samples")
    logger.info(f"Test set: {len(X_test)} samples")

    return X_train, X_test, y_train, y_test


def train_logistic_regression(X_train, y_train):
    """
    Train Logistic Regression model.

    Args:
        X_train: Training features
        y_train: Training target

    Returns:
        Trained model
    """
    logger.info("Training Logistic Regression model...")

    # Create model
    # C=1.0 is regularization strength (lower = more regularization)
    # max_iter=1000 allows more training iterations
    model = LogisticRegression(
        C=1.0,
        max_iter=1000,
        random_state=42,
        solver='lbfgs',  # Optimization algorithm
        class_weight='balanced'  # Handle class imbalance
    )

    # Train the model
    model.fit(X_train, y_train)

    logger.info("Model training complete!")

    return model


def evaluate_model(model, X_test, y_test):
    """
    Evaluate model performance on test set.

    Args:
        model: Trained model
        X_test: Test features
        y_test: Test target

    Returns:
        Dictionary of metrics
    """
    logger.info("Evaluating model on test set...")

    # Make predictions
    y_pred = model.predict(X_test)
    y_pred_proba = model.predict_proba(X_test)[:, 1]  # Probability of default

    # Calculate metrics
    metrics = {
        'accuracy': accuracy_score(y_test, y_pred),
        'precision': precision_score(y_test, y_pred),
        'recall': recall_score(y_test, y_pred),
        'f1_score': f1_score(y_test, y_pred),
        'roc_auc': roc_auc_score(y_test, y_pred_proba)
    }

    # Confusion matrix
    cm = confusion_matrix(y_test, y_pred)

    # Log results
    logger.info("\n" + "=" * 60)
    logger.info("MODEL EVALUATION RESULTS")
    logger.info("=" * 60)
    logger.info(f"\nAccuracy:  {metrics['accuracy']:.4f} ({metrics['accuracy']*100:.2f}%)")
    logger.info(f"Precision: {metrics['precision']:.4f} ({metrics['precision']*100:.2f}%)")
    logger.info(f"Recall:    {metrics['recall']:.4f} ({metrics['recall']*100:.2f}%)")
    logger.info(f"F1 Score:  {metrics['f1_score']:.4f}")
    logger.info(f"ROC-AUC:   {metrics['roc_auc']:.4f}")

    logger.info("\nConfusion Matrix:")
    logger.info("                Predicted")
    logger.info("              No Default  Default")
    logger.info(f"Actual No     {cm[0][0]:6d}      {cm[0][1]:6d}")
    logger.info(f"Actual Yes    {cm[1][0]:6d}      {cm[1][1]:6d}")

    logger.info("\nDetailed Classification Report:")
    logger.info("\n" + classification_report(
        y_test, y_pred,
        target_names=['No Default', 'Default']
    ))

    logger.info("=" * 60)

    return metrics, cm


def get_feature_importance(model, feature_names):
    """
    Get feature importance from logistic regression coefficients.

    Args:
        model: Trained LogisticRegression model
        feature_names: List of feature names

    Returns:
        DataFrame with features and importance scores
    """
    # Get coefficients (weights)
    coefficients = model.coef_[0]

    # Create DataFrame
    importance_df = pd.DataFrame({
        'feature': feature_names,
        'coefficient': coefficients,
        'abs_coefficient': np.abs(coefficients)
    })

    # Sort by absolute value
    importance_df = importance_df.sort_values('abs_coefficient', ascending=False)

    logger.info("\n" + "=" * 60)
    logger.info("TOP 10 MOST IMPORTANT FEATURES")
    logger.info("=" * 60)
    logger.info("\nFeature                          Coefficient")
    logger.info("-" * 60)

    for idx, row in importance_df.head(10).iterrows():
        direction = "↑ increases" if row['coefficient'] > 0 else "↓ decreases"
        logger.info(f"{row['feature']:30s} {row['coefficient']:8.4f} ({direction} default risk)")

    logger.info("=" * 60)

    return importance_df


def save_model(model, metrics, output_dir="data/models"):
    """
    Save trained model and metrics.

    Args:
        model: Trained model
        metrics: Dictionary of evaluation metrics
        output_dir: Directory to save model
    """
    output_path = Path(output_dir)
    output_path.mkdir(parents=True, exist_ok=True)

    # Save model
    model_file = output_path / "logistic_regression.pkl"
    with open(model_file, 'wb') as f:
        pickle.dump(model, f)
    logger.info(f"Model saved to {model_file}")

    # Save metrics
    metrics_file = output_path / "logistic_regression_metrics.json"
    with open(metrics_file, 'w') as f:
        json.dump(metrics, f, indent=2)
    logger.info(f"Metrics saved to {metrics_file}")


def main():
    """Main training pipeline."""

    # Load data
    X_train, X_test, y_train, y_test = load_processed_data()

    # Train model
    model = train_logistic_regression(X_train, y_train)

    # Evaluate model
    metrics, cm = evaluate_model(model, X_test, y_test)

    # Feature importance
    importance_df = get_feature_importance(model, X_train.columns.tolist())

    # Save model
    save_model(model, metrics)

    logger.info("\n✓ Training complete!")
    logger.info(f"✓ Model accuracy: {metrics['accuracy']*100:.2f}%")
    logger.info(f"✓ ROC-AUC Score: {metrics['roc_auc']:.4f}")


if __name__ == "__main__":
    main()
```

**ML Concepts Explained:**

1. **Accuracy:** Percentage of correct predictions
   - (Correct predictions / Total predictions) × 100

2. **Precision:** Of predictions marked as default, how many actually defaulted?
   - High precision = few false alarms

3. **Recall:** Of actual defaults, how many did we catch?
   - High recall = we catch most defaults

4. **F1 Score:** Harmonic mean of precision and recall
   - Balances both metrics

5. **ROC-AUC:** Area Under the Receiver Operating Characteristic curve
   - Measures model's ability to distinguish classes
   - 0.5 = random guessing, 1.0 = perfect

6. **Confusion Matrix:** Shows prediction vs reality
   ```
                   Predicted
              No Default  Default
   Actual No     1200        150     ← False positives (bad for customers)
   Actual Yes     100        550     ← False negatives (bad for bank)
   ```

## Step 4: Create Prediction Module

**File:** `src/models/predict.py`

```python
"""
Make predictions using trained models.
"""

import pandas as pd
import numpy as np
import pickle
from pathlib import Path
from typing import Dict, Any

from src.data.feature_engineering import create_features, get_feature_names
from src.data.preprocessor import LoanDataPreprocessor
from src.utils.logger import logger


class LoanRiskPredictor:
    """
    Predicts loan default risk using trained model.
    """

    def __init__(self, model_name: str = "logistic_regression"):
        self.model_name = model_name
        self.model = None
        self.preprocessor = None
        self.feature_names = None

    def load_model(self):
        """Load trained model and preprocessor."""
        logger.info(f"Loading {self.model_name} model...")

        # Load model
        model_path = Path(f"data/models/{self.model_name}.pkl")
        with open(model_path, 'rb') as f:
            self.model = pickle.load(f)

        # Load preprocessor
        self.preprocessor = LoanDataPreprocessor()
        self.preprocessor.load()

        self.feature_names = self.preprocessor.feature_names

        logger.info(f"Model loaded successfully")

    def predict_single(self, application_data: Dict[str, Any]) -> Dict[str, Any]:
        """
        Predict default risk for a single loan application.

        Args:
            application_data: Dictionary with loan application details

        Returns:
            Dictionary with prediction results
        """
        if self.model is None:
            self.load_model()

        # Convert to DataFrame
        df = pd.DataFrame([application_data])

        # Feature engineering
        df = create_features(df)

        # Encode categorical features
        numerical_features, categorical_features = get_feature_names()

        for col in categorical_features:
            if col in df.columns and col in self.preprocessor.label_encoders:
                df[col] = self.preprocessor.label_encoders[col].transform(
                    df[col].astype(str)
                )

        # Select and scale features
        X = df[self.feature_names]

        # Scale numerical features
        X_scaled = X.copy()
        X_scaled[numerical_features] = self.preprocessor.scaler.transform(
            X[numerical_features]
        )

        # Make prediction
        prediction = self.model.predict(X_scaled)[0]
        probability = self.model.predict_proba(X_scaled)[0]

        # Categorize risk
        default_prob = probability[1]

        if default_prob < 0.2:
            risk_category = "VERY_LOW"
            recommendation = "APPROVE - Excellent candidate"
        elif default_prob < 0.4:
            risk_category = "LOW"
            recommendation = "APPROVE - Good candidate"
        elif default_prob < 0.6:
            risk_category = "MEDIUM"
            recommendation = "REVIEW - Manual review recommended"
        elif default_prob < 0.8:
            risk_category = "HIGH"
            recommendation = "REJECT - High risk of default"
        else:
            risk_category = "VERY_HIGH"
            recommendation = "REJECT - Very high risk"

        result = {
            'predicted_default': int(prediction),
            'default_probability': float(default_prob),
            'no_default_probability': float(probability[0]),
            'risk_category': risk_category,
            'recommendation': recommendation
        }

        logger.info(f"Prediction: {risk_category} (probability: {default_prob:.3f})")

        return result


# Global predictor instance
predictor = LoanRiskPredictor()
```

@app.post("/api/v1/predict", response_model=PredictionResponse, tags=["Predictions"])
async def predict_loan_risk(
    application: LoanApplicationRequest,
    db: Session = Depends(get_db),
    current_user = Depends(auth.get_current_active_user)
):
    """
    Predict loan default risk using ML model.

    **Requires authentication.**
    """
    logger.info(f"ML Prediction requested by {current_user.username} for {application.applicant_name}")

    # Prepare data for model
    application_data = {
        'age': application.age,
        'gender': 'Male',  # Default (not in request schema)
        'marital_status': 'SINGLE',  # Default
        'education_level': 'HND/BACHELOR',  # Default
        'employment_status': application.employment_status,
        'employer_sector': 'Banking & Finance',  # Default
        'years_employed': max(1, application.age - 25),  # Estimate
        'monthly_income': application.monthly_income,
        'loan_amount': application.loan_amount,
        'loan_purpose': application.loan_purpose or 'Business expansion',
        'requested_tenure_months': 24,  # Default
        'has_existing_loans': False,  # Default
        'num_existing_loans': 0,
        'total_existing_debt': 0,
        'has_savings_account': True,  # Default
        'savings_balance': application.monthly_income * 3,  # Estimate
        'credit_score': 650,  # Default
        'state': 'Lagos',  # Default
        'bank': 'GTBank',  # Default
    }

    try:
        # Get ML prediction
        prediction_result = predictor.predict_single(application_data)

        risk_category = RiskCategory[prediction_result['risk_category']]
        recommendation = prediction_result['recommendation']
        risk_score = prediction_result['default_probability']

    except Exception as e:
        logger.error(f"Prediction error: {e}")
        raise HTTPException(
            status_code=500,
            detail="Prediction failed. Please check model is trained."
        )

    # Save to database
    db_application = LoanApplication(
        applicant_name=application.applicant_name,
        age=application.age,
        employment_status=EmploymentStatus(application.employment_status),
        monthly_income=application.monthly_income,
        loan_amount=application.loan_amount,
        loan_purpose=application.loan_purpose,
        predicted_risk=risk_category,
        risk_score=round(risk_score, 3),
        recommendation=recommendation
    )

    db.add(db_application)
    db.commit()
    db.refresh(db_application)

    logger.info(f"ML Prediction: {risk_category.value} (score: {risk_score:.3f})")

    return PredictionResponse(
        applicant_name=application.applicant_name,
        loan_amount=application.loan_amount,
        predicted_risk=risk_category.value,
        risk_score=round(risk_score, 3),
        recommendation=recommendation,
        timestamp=datetime.utcnow()
    )

## Understanding the Metrics

**Scenario: Bank processes 1000 loans**

**With 78% accuracy, 73% precision, 69% recall:**

- **780 predictions are correct**
- Of 350 "will default" predictions, **255 actually default** (73% precision)
- Of 300 actual defaults, **207 are caught** (69% recall)
- **93 defaults are missed** (false negatives - bank loses money)
- **95 good customers rejected** (false positives - lose business)

**Trade-offs:**
- Higher precision = fewer false alarms, but miss more defaults
- Higher recall = catch more defaults, but reject more good customers
- Bank must decide which is more costly

**Issue:** "Preprocessed data not found"
**Solution:** Run `python src/data/preprocessor.py` first

**Issue:** Low accuracy (< 70%)
**Solution:** Check data quality, try more training data, or different model

**Issue:** Model won't load in API
**Solution:** Check model file exists in data/models/

**Issue:** Predictions all the same
**Solution:** Check feature scaling, verify model trained correctly
