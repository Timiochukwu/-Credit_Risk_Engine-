## Why Model Governance?

**Problem**: Without governance, ML systems face:
- **Regulatory risk**: Fines for non-compliance with GDPR, Fair Lending Act
- **Reputational damage**: Biased decisions harm customers and brand
- **Operational chaos**: Multiple model versions, no audit trail
- **Legal liability**: Cannot prove fair and transparent decisions
- **Model drift**: Performance degradation goes undetected

**Solution**: Model governance provides:
- **Model registry**: Track all model versions, metadata, and lineage
- **Audit logging**: Record every prediction with full context
- **Bias detection**: Monitor for unfair treatment across demographics
- **Compliance**: Meet regulatory requirements (GDPR, ECOA, Basel III)
- **Approval workflows**: Govern model deployment and changes

**Benefits**:
- **Regulatory compliance**: Avoid fines and legal issues
- **Fairness**: Ensure equitable treatment
- **Accountability**: Clear audit trail for decisions
- **Quality control**: Systematic model validation and approval
- **Risk management**: Track and mitigate model risks

```python
import mlflow
from mlflow.tracking import MlflowClient
from typing import Dict, Any, List, Optional
from datetime import datetime
import logging

logger = logging.getLogger(__name__)


class ModelRegistry:
    """Centralized model registry for version control and governance."""

    def __init__(self, tracking_uri: str):
        """Initialize model registry."""
        mlflow.set_tracking_uri(tracking_uri)
        self.client = MlflowClient()

    def register_model(
        self,
        model_uri: str,
        name: str,
        tags: Optional[Dict[str, str]] = None,
        description: Optional[str] = None
    ) -> str:
        """
        Register a model in the registry.

        Args:
            model_uri: URI to model artifacts (e.g., runs:/run_id/model)
            name: Name for the registered model
            tags: Additional tags (owner, purpose, etc.)
            description: Model description

        Returns:
            Model version
        """
        try:
            # Register model
            model_version = mlflow.register_model(
                model_uri=model_uri,
                name=name,
                tags=tags or {}
            )

            # Update description if provided
            if description:
                self.client.update_model_version(
                    name=name,
                    version=model_version.version,
                    description=description
                )

            logger.info(f"Registered model {name} version {model_version.version}")
            return model_version.version

        except Exception as e:
            logger.error(f"Failed to register model: {e}")
            raise

    def transition_stage(
        self,
        name: str,
        version: str,
        stage: str,
        archive_existing: bool = True
    ) -> None:
        """
        Transition model to a new stage.

        Args:
            name: Model name
            version: Model version
            stage: Target stage (Staging, Production, Archived)
            archive_existing: Archive existing models in target stage
        """
        valid_stages = ["Staging", "Production", "Archived"]
        if stage not in valid_stages:
            raise ValueError(f"Stage must be one of {valid_stages}")

        self.client.transition_model_version_stage(
            name=name,
            version=version,
            stage=stage,
            archive_existing_versions=archive_existing
        )

        logger.info(f"Transitioned {name} v{version} to {stage}")

    def get_production_model(self, name: str) -> Optional[Dict[str, Any]]:
        """Get current production model version."""
        try:
            versions = self.client.get_latest_versions(name, stages=["Production"])
            if not versions:
                return None

            version = versions[0]
            return {
                'name': version.name,
                'version': version.version,
                'run_id': version.run_id,
                'stage': version.current_stage,
                'creation_timestamp': version.creation_timestamp,
                'description': version.description,
                'tags': version.tags
            }
        except Exception as e:
            logger.error(f"Failed to get production model: {e}")
            return None

    def list_model_versions(
        self,
        name: str,
        stage: Optional[str] = None
    ) -> List[Dict[str, Any]]:
        """List all versions of a model."""
        if stage:
            versions = self.client.get_latest_versions(name, stages=[stage])
        else:
            versions = self.client.search_model_versions(f"name='{name}'")

        return [
            {
                'version': v.version,
                'stage': v.current_stage,
                'run_id': v.run_id,
                'creation_timestamp': v.creation_timestamp,
                'description': v.description,
                'tags': v.tags
            }
            for v in versions
        ]

    def add_model_metadata(
        self,
        name: str,
        version: str,
        metadata: Dict[str, str]
    ) -> None:
        """Add metadata tags to a model version."""
        for key, value in metadata.items():
            self.client.set_model_version_tag(
                name=name,
                version=version,
                key=key,
                value=str(value)
            )

        logger.info(f"Added metadata to {name} v{version}")
```

```python
from enum import Enum
from typing import Dict, Any, List
from datetime import datetime
import json


class ModelStatus(str, Enum):
    """Model lifecycle statuses."""
    DEVELOPMENT = "development"
    VALIDATION = "validation"
    STAGING = "staging"
    PRODUCTION = "production"
    DEPRECATED = "deprecated"
    ARCHIVED = "archived"


class ModelLifecycle:
    """Manage model lifecycle and governance workflows."""

    def __init__(self, registry: 'ModelRegistry'):
        self.registry = registry

    def submit_for_validation(
        self,
        model_name: str,
        version: str,
        submitter: str,
        validation_tests: Dict[str, Any]
    ) -> Dict[str, Any]:
        """
        Submit model for validation.

        Args:
            model_name: Name of the model
            version: Model version
            submitter: Person submitting
            validation_tests: Required validation tests

        Returns:
            Validation ticket
        """
        ticket = {
            'model_name': model_name,
            'version': version,
            'submitter': submitter,
            'submitted_at': datetime.now().isoformat(),
            'status': ModelStatus.VALIDATION,
            'validation_tests': validation_tests,
            'approvals': []
        }

        # Add metadata to model
        self.registry.add_model_metadata(
            name=model_name,
            version=version,
            metadata={
                'lifecycle_status': ModelStatus.VALIDATION,
                'submitted_by': submitter,
                'submitted_at': ticket['submitted_at']
            }
        )

        return ticket

    def approve_model(
        self,
        model_name: str,
        version: str,
        approver: str,
        approval_type: str,
        comments: str = ""
    ) -> None:
        """
        Approve model for progression.

        Args:
            model_name: Name of the model
            version: Model version
            approver: Person approving
            approval_type: Type of approval (technical, business, compliance)
            comments: Approval comments
        """
        approval = {
            'approver': approver,
            'type': approval_type,
            'timestamp': datetime.now().isoformat(),
            'comments': comments
        }

        # Add approval metadata
        self.registry.add_model_metadata(
            name=model_name,
            version=version,
            metadata={
                f'approval_{approval_type}': json.dumps(approval)
            }
        )

    def promote_to_production(
        self,
        model_name: str,
        version: str,
        required_approvals: List[str]
    ) -> bool:
        """
        Promote model to production after validation.

        Args:
            model_name: Name of the model
            version: Model version
            required_approvals: List of required approval types

        Returns:
            True if promoted successfully
        """
        # Get model metadata
        versions = self.registry.list_model_versions(model_name)
        target_version = next((v for v in versions if v['version'] == version), None)

        if not target_version:
            raise ValueError(f"Version {version} not found")

        # Check for required approvals
        tags = target_version.get('tags', {})
        for approval_type in required_approvals:
            approval_key = f'approval_{approval_type}'
            if approval_key not in tags:
                raise ValueError(f"Missing required approval: {approval_type}")

        # Promote to production
        self.registry.transition_stage(
            name=model_name,
            version=version,
            stage="Production",
            archive_existing=True
        )

        # Add production metadata
        self.registry.add_model_metadata(
            name=model_name,
            version=version,
            metadata={
                'lifecycle_status': ModelStatus.PRODUCTION,
                'promoted_at': datetime.now().isoformat()
            }
        )

        return True
```

### Step 2: Create Fairness Evaluator

```python
import pandas as pd
import numpy as np
from typing import Dict, List, Any
from fairlearn.metrics import (
    demographic_parity_difference,
    equalized_odds_difference,
    MetricFrame
)
from sklearn.metrics import accuracy_score, precision_score, recall_score


class FairnessEvaluator:
    """Evaluate model fairness across demographic groups."""

    @staticmethod
    def evaluate_fairness(
        y_true: np.ndarray,
        y_pred: np.ndarray,
        sensitive_features: pd.DataFrame,
        sensitive_feature_names: List[str]
    ) -> Dict[str, Any]:
        """
        Evaluate fairness metrics.

        Args:
            y_true: True labels
            y_pred: Predicted labels
            sensitive_features: DataFrame with sensitive attributes
            sensitive_feature_names: Names of sensitive features to evaluate

        Returns:
            Dictionary of fairness metrics
        """
        results = {}

        for feature_name in sensitive_feature_names:
            if feature_name not in sensitive_features.columns:
                continue

            sensitive_feature = sensitive_features[feature_name]

            # Calculate demographic parity difference
            dpd = demographic_parity_difference(
                y_true=y_true,
                y_pred=y_pred,
                sensitive_features=sensitive_feature
            )

            # Calculate equalized odds difference
            eod = equalized_odds_difference(
                y_true=y_true,
                y_pred=y_pred,
                sensitive_features=sensitive_feature
            )

            # Calculate metrics per group
            metric_frame = MetricFrame(
                metrics={
                    'accuracy': accuracy_score,
                    'precision': precision_score,
                    'recall': recall_score
                },
                y_true=y_true,
                y_pred=y_pred,
                sensitive_features=sensitive_feature
            )

            results[feature_name] = {
                'demographic_parity_difference': float(dpd),
                'equalized_odds_difference': float(eod),
                'metrics_by_group': metric_frame.by_group.to_dict(),
                'overall_metrics': {
                    k: float(v) for k, v in metric_frame.overall.items()
                },
                'max_disparity': {
                    k: float(v) for k, v in metric_frame.difference().items()
                }
            }

        return results

    @staticmethod
    def check_fairness_thresholds(
        fairness_results: Dict[str, Any],
        dpd_threshold: float = 0.1,
        eod_threshold: float = 0.1
    ) -> Dict[str, Any]:
        """
        Check if fairness metrics meet thresholds.

        Args:
            fairness_results: Results from evaluate_fairness()
            dpd_threshold: Maximum acceptable demographic parity difference
            eod_threshold: Maximum acceptable equalized odds difference

        Returns:
            Pass/fail status for each sensitive feature
        """
        results = {}

        for feature_name, metrics in fairness_results.items():
            dpd = abs(metrics['demographic_parity_difference'])
            eod = abs(metrics['equalized_odds_difference'])

            passes = dpd <= dpd_threshold and eod <= eod_threshold

            results[feature_name] = {
                'passes': passes,
                'dpd': dpd,
                'dpd_threshold': dpd_threshold,
                'dpd_passes': dpd <= dpd_threshold,
                'eod': eod,
                'eod_threshold': eod_threshold,
                'eod_passes': eod <= eod_threshold
            }

        overall_pass = all(r['passes'] for r in results.values())

        return {
            'overall_pass': overall_pass,
            'feature_results': results
        }
```

```python
from typing import Dict, Any
from datetime import datetime
import uuid


class GDPRCompliance:
    """GDPR compliance utilities."""

    @staticmethod
    def generate_explanation_report(
        prediction_id: uuid.UUID,
        input_data: Dict[str, Any],
        prediction: Dict[str, Any],
        shap_explanation: Dict[str, Any]
    ) -> Dict[str, Any]:
        """
        Generate human-readable explanation report.

        Required by GDPR Article 22 (Right to explanation).
        """
        return {
            'prediction_id': str(prediction_id),
            'timestamp': datetime.now().isoformat(),
            'decision': 'REJECT' if prediction['default_probability'] > 0.5 else 'APPROVE',
            'risk_score': prediction['default_probability'],
            'explanation': {
                'summary': FairnessEvaluator._generate_summary(shap_explanation),
                'top_factors': shap_explanation['top_factors'],
                'methodology': 'Machine learning model trained on historical credit data',
                'appeal_process': 'Contact customer service to request manual review'
            },
            'data_used': {
                k: v for k, v in input_data.items()
                if k not in ['ssn', 'account_number']  # Redact sensitive fields
            },
            'rights': {
                'right_to_access': 'You can request all data we hold about you',
                'right_to_rectification': 'You can request correction of inaccurate data',
                'right_to_erasure': 'You can request deletion of your data',
                'right_to_object': 'You can object to automated decision-making'
            }
        }

    @staticmethod
    def _generate_summary(shap_explanation: Dict[str, Any]) -> str:
        """Generate human-readable summary."""
        top_factors = shap_explanation['top_factors'][:3]

        summary = "This decision was made based on the following factors:\n"
        for i, factor in enumerate(top_factors, 1):
            direction = "increased" if factor['shap_value'] > 0 else "decreased"
            summary += f"{i}. Your {factor['feature']} ({factor['value']}) {direction} the risk.\n"
```

```python
from sqlalchemy.orm import Session
import uuid

from app.db.session import get_db

@router.get("/explanation/{prediction_id}")
async def get_explanation(
    prediction_id: uuid.UUID,
    db: Session = Depends(get_db)
):
    """
    Get detailed explanation for a prediction (GDPR Article 22).

    Args:
        prediction_id: ID of the prediction

    Returns:
        Human-readable explanation report
    """
    try:
        # Retrieve prediction from audit log
        audit_log = db.query(AuditLog).filter(
            AuditLog.id == prediction_id
        ).first()

        if not audit_log:
            raise HTTPException(status_code=404, detail="Prediction not found")

        # Generate explanation report
        report = GDPRCompliance.generate_explanation_report(
            prediction_id=prediction_id,
            input_data=audit_log.input_data,
            prediction=audit_log.prediction,
            shap_explanation=audit_log.explanation
        )

        return report

    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))


@router.delete("/data/{user_id}")
async def delete_user_data(
    user_id: str,
    db: Session = Depends(get_db),
    confirmed: bool = False
):
    """
    Delete all user data (GDPR Right to Erasure).

    Args:
        user_id: User ID
        confirmed: Confirmation flag

    Returns:
        Deletion summary
    """
    if not confirmed:
        raise HTTPException(
            status_code=400,
            detail="Set confirmed=true to proceed with deletion"
        )

    try:
        # Delete audit logs
        deleted_logs = db.query(AuditLog).filter(
            AuditLog.user_id == user_id
        ).delete()

        db.commit()

        return {
            'user_id': user_id,
            'deleted_audit_logs': deleted_logs,

        }

    except Exception as e:
```
