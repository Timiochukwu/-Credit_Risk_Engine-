```python
"""
Comprehensive model evaluation and analysis.
"""

import pandas as pd
import numpy as np
import matplotlib
matplotlib.use('Agg')  # Non-interactive backend (no display needed)
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.metrics import (
    roc_curve,
    precision_recall_curve,
    auc,
    confusion_matrix
)
import pickle
from pathlib import Path
import json

from src.utils.logger import logger

# Set style
sns.set_style('whitegrid')
plt.rcParams['figure.figsize'] = (10, 6)


def load_model_and_data(model_name):
    """
    Load trained model and test data.

    Args:
        model_name: Name of the model to evaluate

    Returns:
        model, X_test, y_test
    """
    logger.info(f"Loading {model_name} model and test data...")

    # Load model
    model_path = Path(f"data/models/{model_name}.pkl")
    with open(model_path, 'rb') as f:
        model = pickle.load(f)

    # Load test data
    data_dir = Path("data/processed")
    X_test = pd.read_csv(data_dir / "X_test.csv")
    y_test = pd.read_csv(data_dir / "y_test.csv")['defaulted']

    logger.info(f"Loaded model and {len(X_test)} test samples")

    return model, X_test, y_test


def plot_roc_curve(y_test, y_pred_proba, model_name, output_dir):
    """
    Plot ROC (Receiver Operating Characteristic) curve.

    Shows trade-off between True Positive Rate and False Positive Rate.
    """
    logger.info("Generating ROC curve...")

    fpr, tpr, thresholds = roc_curve(y_test, y_pred_proba)
    roc_auc = auc(fpr, tpr)

    plt.figure(figsize=(10, 8))
    plt.plot(fpr, tpr, color='darkorange', lw=2,
             label=f'ROC curve (AUC = {roc_auc:.4f})')
    plt.plot([0, 1], [0, 1], color='navy', lw=2, linestyle='--',
             label='Random Classifier')

    plt.xlim([0.0, 1.0])
    plt.ylim([0.0, 1.05])
    plt.xlabel('False Positive Rate', fontsize=12)
    plt.ylabel('True Positive Rate', fontsize=12)
    plt.title(f'ROC Curve - {model_name}', fontsize=14, fontweight='bold')
    plt.legend(loc="lower right", fontsize=11)
    plt.grid(True, alpha=0.3)

    output_file = output_dir / f'{model_name}_roc_curve.png'
    plt.savefig(output_file, dpi=150, bbox_inches='tight')
    plt.close()

    logger.info(f"ROC curve saved to {output_file}")

    return roc_auc


def plot_precision_recall_curve(y_test, y_pred_proba, model_name, output_dir):
    """
    Plot Precision-Recall curve.

    Useful for imbalanced datasets - shows trade-off between
    precision and recall at different thresholds.
    """
    logger.info("Generating Precision-Recall curve...")

    precision, recall, thresholds = precision_recall_curve(y_test, y_pred_proba)
    pr_auc = auc(recall, precision)

    plt.figure(figsize=(10, 8))
    plt.plot(recall, precision, color='blue', lw=2,
             label=f'PR curve (AUC = {pr_auc:.4f})')

    plt.xlim([0.0, 1.0])
    plt.ylim([0.0, 1.05])
    plt.xlabel('Recall', fontsize=12)
    plt.ylabel('Precision', fontsize=12)
    plt.title(f'Precision-Recall Curve - {model_name}', fontsize=14, fontweight='bold')
    plt.legend(loc="lower left", fontsize=11)
    plt.grid(True, alpha=0.3)

    output_file = output_dir / f'{model_name}_precision_recall_curve.png'
    plt.savefig(output_file, dpi=150, bbox_inches='tight')
    plt.close()

    logger.info(f"Precision-Recall curve saved to {output_file}")

    return pr_auc


def plot_confusion_matrix(y_test, y_pred, model_name, output_dir):
    """
    Plot confusion matrix heatmap.
    """
    logger.info("Generating confusion matrix visualization...")

    cm = confusion_matrix(y_test, y_pred)

    plt.figure(figsize=(8, 6))
    sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',
                xticklabels=['No Default', 'Default'],
                yticklabels=['No Default', 'Default'],
                cbar_kws={'label': 'Count'})

    plt.ylabel('Actual', fontsize=12)
    plt.xlabel('Predicted', fontsize=12)
    plt.title(f'Confusion Matrix - {model_name}', fontsize=14, fontweight='bold')

    output_file = output_dir / f'{model_name}_confusion_matrix.png'
    plt.savefig(output_file, dpi=150, bbox_inches='tight')
    plt.close()

    logger.info(f"Confusion matrix saved to {output_file}")


def plot_feature_importance(model, feature_names, model_name, output_dir, top_n=15):
    """
    Plot feature importance.
    """
    logger.info("Generating feature importance plot...")

    # Get feature importance based on model type
    if hasattr(model, 'feature_importances_'):
        # Tree-based models (XGBoost, Random Forest)
        importance = model.feature_importances_
    elif hasattr(model, 'coef_'):
        # Linear models (Logistic Regression)
        importance = np.abs(model.coef_[0])
    else:
        logger.warning("Model doesn't support feature importance")
        return

    # Create DataFrame
    importance_df = pd.DataFrame({
        'feature': feature_names,
        'importance': importance
    }).sort_values('importance', ascending=False).head(top_n)

    # Plot
    plt.figure(figsize=(10, 8))
    sns.barplot(data=importance_df, x='importance', y='feature', palette='viridis')

    plt.xlabel('Importance Score', fontsize=12)
    plt.ylabel('Feature', fontsize=12)
    plt.title(f'Top {top_n} Most Important Features - {model_name}',
              fontsize=14, fontweight='bold')
    plt.tight_layout()

    output_file = output_dir / f'{model_name}_feature_importance.png'
    plt.savefig(output_file, dpi=150, bbox_inches='tight')
    plt.close()

    logger.info(f"Feature importance plot saved to {output_file}")


def plot_prediction_distribution(y_test, y_pred_proba, model_name, output_dir):
    """
    Plot distribution of prediction probabilities for each class.
    """
    logger.info("Generating prediction distribution plot...")

    plt.figure(figsize=(12, 6))

    # Separate by actual class
    default_probs = y_pred_proba[y_test == 1]
    no_default_probs = y_pred_proba[y_test == 0]

    plt.hist(no_default_probs, bins=50, alpha=0.6, label='Actual: No Default',
             color='green', edgecolor='black')
    plt.hist(default_probs, bins=50, alpha=0.6, label='Actual: Default',
             color='red', edgecolor='black')

    plt.axvline(x=0.5, color='black', linestyle='--', linewidth=2,
                label='Decision Threshold')

    plt.xlabel('Predicted Default Probability', fontsize=12)
    plt.ylabel('Frequency', fontsize=12)
    plt.title(f'Prediction Distribution - {model_name}', fontsize=14, fontweight='bold')
    plt.legend(fontsize=11)
    plt.grid(True, alpha=0.3)

    output_file = output_dir / f'{model_name}_prediction_distribution.png'
    plt.savefig(output_file, dpi=150, bbox_inches='tight')
    plt.close()

    logger.info(f"Prediction distribution saved to {output_file}")


def analyze_errors(X_test, y_test, y_pred, y_pred_proba):
    """
    Analyze where the model makes errors.

    Returns analysis of false positives and false negatives.
    """
    logger.info("Analyzing prediction errors...")

    # Create results DataFrame
    results_df = X_test.copy()
    results_df['actual'] = y_test.values
    results_df['predicted'] = y_pred
    results_df['probability'] = y_pred_proba

    # Identify error types
    results_df['error_type'] = 'Correct'
    results_df.loc[
        (results_df['actual'] == 0) & (results_df['predicted'] == 1),
        'error_type'
    ] = 'False Positive'
    results_df.loc[
        (results_df['actual'] == 1) & (results_df['predicted'] == 0),
        'error_type'
    ] = 'False Negative'

    # Count errors
    error_counts = results_df['error_type'].value_counts()

    logger.info("\nError Analysis:")
    logger.info(f"  Correct predictions: {error_counts.get('Correct', 0)}")
    logger.info(f"  False Positives: {error_counts.get('False Positive', 0)}")
    logger.info(f"  False Negatives: {error_counts.get('False Negative', 0)}")

    # Analyze false positives (predicted default, but didn't)
    fp_df = results_df[results_df['error_type'] == 'False Positive']
    if len(fp_df) > 0:
        logger.info("\nFalse Positive Characteristics:")
        logger.info(f"  Average probability: {fp_df['probability'].mean():.3f}")

    # Analyze false negatives (predicted no default, but did)
    fn_df = results_df[results_df['error_type'] == 'False Negative']
    if len(fn_df) > 0:
        logger.info("\nFalse Negative Characteristics:")
        logger.info(f"  Average probability: {fn_df['probability'].mean():.3f}")

    return results_df


def generate_evaluation_report(model_name):
    """
    Generate comprehensive evaluation report with visualizations.
    """
    logger.info("=" * 60)
    logger.info(f"EVALUATING {model_name.upper()} MODEL")
    logger.info("=" * 60)

    # Create output directory
    output_dir = Path("data/models/evaluation")
    output_dir.mkdir(parents=True, exist_ok=True)

    # Load model and data
    model, X_test, y_test = load_model_and_data(model_name)

    # Make predictions
    y_pred = model.predict(X_test)
    y_pred_proba = model.predict_proba(X_test)[:, 1]

    # Generate visualizations
    logger.info("\nGenerating visualizations...")

    roc_auc = plot_roc_curve(y_test, y_pred_proba, model_name, output_dir)
    pr_auc = plot_precision_recall_curve(y_test, y_pred_proba, model_name, output_dir)
    plot_confusion_matrix(y_test, y_pred, model_name, output_dir)
    plot_feature_importance(model, X_test.columns.tolist(), model_name, output_dir)
    plot_prediction_distribution(y_test, y_pred_proba, model_name, output_dir)

    # Error analysis
    results_df = analyze_errors(X_test, y_test, y_pred, y_pred_proba)

    # Save detailed results
    results_file = output_dir / f'{model_name}_predictions.csv'
    results_df.to_csv(results_file, index=False)
    logger.info(f"\nDetailed predictions saved to {results_file}")

    logger.info("\n" + "=" * 60)
    logger.info("EVALUATION COMPLETE")
    logger.info("=" * 60)
    logger.info(f"\nVisualizations saved to: {output_dir}")

    return {
        'roc_auc': roc_auc,
        'pr_auc': pr_auc,
        'output_dir': str(output_dir)
    }


def main():
    """Evaluate all available models."""

    models_dir = Path("data/models")
    model_files = list(models_dir.glob("*.pkl"))

    if not model_files:
        logger.error("No trained models found. Run 'python src/models/train.py' first.")
        return

    model_names = [f.stem for f in model_files]

    logger.info(f"Found {len(model_names)} models to evaluate: {model_names}\n")

    for model_name in model_names:
        try:
            generate_evaluation_report(model_name)
            logger.info("\n")
        except Exception as e:
            logger.error(f"Error evaluating {model_name}: {e}")

    logger.info("✓ All evaluations complete!")


if __name__ == "__main__":
    main()
```

**Visualization Concepts Explained:**

1. **ROC Curve:** Shows classifier performance across all thresholds
   - X-axis: False Positive Rate (false alarms)
   - Y-axis: True Positive Rate (correct detections)
   - Perfect classifier: follows top-left corner
   - Random classifier: diagonal line

2. **Precision-Recall Curve:** Better for imbalanced datasets
   - X-axis: Recall (% of defaults caught)
   - Y-axis: Precision (% of predictions that are correct)
   - Trade-off: Higher recall → lower precision

3. **Confusion Matrix Heatmap:** Visual representation of errors
   - Darker blue = more samples
   - Easy to spot error patterns

4. **Feature Importance:** Which features drive predictions
   - Longer bars = more important features

5. **Prediction Distribution:** How confident is the model?
   - Good separation = confident predictions
   - Overlapping histograms = uncertain predictions

# ... existing imports ...

@router.get("/evaluation/{model_name}")
async def get_model_evaluation(model_name: str):
    """
    Get evaluation results for a specific model.

    Returns available visualization files and metrics.
    """
    eval_dir = Path("data/models/evaluation")

    if not eval_dir.exists():
        raise HTTPException(
            status_code=404,
            detail="No evaluation results found. Run 'python src/models/evaluate.py' first."
        )

    # Find visualization files for this model
    viz_files = list(eval_dir.glob(f"{model_name}_*.png"))

    if not viz_files:
        raise HTTPException(
            status_code=404,
            detail=f"No evaluation results found for model: {model_name}"
        )

    # Load metrics
    metrics_file = Path(f"data/models/{model_name}_metrics.json")
    metrics = {}

    if metrics_file.exists():
        import json
        with open(metrics_file, 'r') as f:
            metrics = json.load(f)

    visualizations = [
        {
            "name": f.stem.replace(f"{model_name}_", ""),
            "filename": f.name,
            "path": str(f)
        }
        for f in viz_files
    ]

    return {
        "model_name": model_name,
        "metrics": metrics,
        "visualizations": visualizations,
        "total_visualizations": len(visualizations)
    }


@router.get("/evaluations")
async def list_all_evaluations():
    """
    List all available model evaluations.
    """
    eval_dir = Path("data/models/evaluation")

    if not eval_dir.exists():
        raise HTTPException(
            status_code=404,
            detail="No evaluation results found."
        )

    # Find all PNG files
    viz_files = list(eval_dir.glob("*.png"))

    # Group by model
    models = {}
    for f in viz_files:
        # Extract model name from filename (before first underscore)
        parts = f.stem.split('_', 1)
        if len(parts) == 2:
            model_name = parts[0]
            if model_name not in models:
                models[model_name] = []
            models[model_name].append(f.name)

    return {
        "total_models_evaluated": len(models),
        "models": models
    }

## Interpreting the Visualizations

**ROC Curve:**
- Closer to top-left = better model
- AUC = 0.916 is excellent (>0.9 is great)
- Random classifier would be diagonal line

**Precision-Recall:**
- High precision + high recall = ideal
- Trade-off visible in curve shape
- Important for imbalanced data

**Confusion Matrix:**
- Diagonal (top-left to bottom-right) should be dark
- Off-diagonal = errors
- Compare proportions, not absolute numbers

**Feature Importance:**
- Top features drive most predictions
- Focus optimization efforts here
- Remove low-importance features to simplify

**Prediction Distribution:**
- Good separation = confident model
- Overlap near 0.5 = uncertain predictions
- Check for calibration (are probabilities accurate?)
