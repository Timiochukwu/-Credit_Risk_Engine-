## Understanding XGBoost

**What is XGBoost?**
XGBoost (eXtreme Gradient Boosting) is an ensemble learning method that combines many weak decision trees into a strong predictor.

**How it works:**
```
Tree 1 makes prediction → Calculate errors
Tree 2 tries to fix Tree 1's errors → Calculate remaining errors
Tree 3 tries to fix Tree 2's errors → ...
Final prediction = Weighted sum of all trees
```

**Why XGBoost?**
- ✅ Often wins Kaggle competitions
- ✅ Handles complex non-linear patterns
- ✅ Built-in regularization (prevents overfitting)
- ✅ Fast training with GPU support
- ✅ Handles missing values
- ✅ Industry standard for credit scoring

**XGBoost vs Logistic Regression:**

| Feature | Logistic Regression | XGBoost |
|---------|-------------------|---------|
| Complexity | Simple, linear | Complex, non-linear |
| Interpretability | High | Medium |
| Training Time | Fast (seconds) | Slower (minutes) |
| Accuracy | Good baseline | Usually better |
| Overfitting Risk | Low | Medium (needs tuning) |

"""
Train machine learning models for loan default prediction.
"""

import pandas as pd
import numpy as np
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import (
    accuracy_score,
    precision_score,
    recall_score,
    f1_score,
    roc_auc_score,
    confusion_matrix,
    classification_report
)
import xgboost as xgb
from pathlib import Path
import pickle
import json

from src.utils.logger import logger


def load_processed_data():
    """Load preprocessed training and test data."""
    logger.info("Loading preprocessed data...")

    data_dir = Path("data/processed")

    X_train = pd.read_csv(data_dir / "X_train.csv")
    X_test = pd.read_csv(data_dir / "X_test.csv")
    y_train = pd.read_csv(data_dir / "y_train.csv")['defaulted']
    y_test = pd.read_csv(data_dir / "y_test.csv")['defaulted']

    logger.info(f"Training set: {len(X_train)} samples")
    logger.info(f"Test set: {len(X_test)} samples")

    return X_train, X_test, y_train, y_test


def train_logistic_regression(X_train, y_train):
    """
    Train Logistic Regression model.

    Args:
        X_train: Training features
        y_train: Training target

    Returns:
        Trained model
    """
    logger.info("Training Logistic Regression model...")

    model = LogisticRegression(
        C=1.0,
        max_iter=1000,
        random_state=42,
        solver='lbfgs',
        class_weight='balanced'
    )

    model.fit(X_train, y_train)

    logger.info("Model training complete!")

    return model


def train_xgboost(X_train, y_train, X_test, y_test):
    """
    Train XGBoost model.

    Args:
        X_train: Training features
        y_train: Training target
        X_test: Test features (for early stopping)
        y_test: Test target (for early stopping)

    Returns:
        Trained XGBoost model
    """
    logger.info("Training XGBoost model...")

    # Calculate scale_pos_weight for class imbalance
    # This helps the model pay more attention to the minority class
    n_positive = (y_train == 1).sum()
    n_negative = (y_train == 0).sum()
    scale_pos_weight = n_negative / n_positive

    logger.info(f"Class distribution - Positive: {n_positive}, Negative: {n_negative}")
    logger.info(f"Scale pos weight: {scale_pos_weight:.2f}")

    # XGBoost parameters
    params = {
        'objective': 'binary:logistic',  # Binary classification
        'eval_metric': 'auc',            # Use ROC-AUC for evaluation
        'max_depth': 6,                  # Maximum tree depth
        'learning_rate': 0.1,            # Step size shrinkage (eta)
        'n_estimators': 200,             # Number of boosting rounds
        'subsample': 0.8,                # Fraction of samples for each tree
        'colsample_bytree': 0.8,         # Fraction of features for each tree
        'scale_pos_weight': scale_pos_weight,  # Handle class imbalance
        'random_state': 42,
        'tree_method': 'hist',           # Faster histogram-based algorithm
    }

    # Create model
    model = xgb.XGBClassifier(**params)

    # Train with early stopping
    # Stops training if validation performance doesn't improve
    model.fit(
        X_train, y_train,
        eval_set=[(X_test, y_test)],
        early_stopping_rounds=20,       # Stop if no improvement for 20 rounds
        verbose=True                     # Print progress
    )

    logger.info(f"Best iteration: {model.best_iteration}")
    logger.info(f"Best score: {model.best_score:.4f}")
    logger.info("XGBoost training complete!")

    return model


def evaluate_model(model, X_test, y_test, model_name="Model"):
    """
    Evaluate model performance on test set.

    Args:
        model: Trained model
        X_test: Test features
        y_test: Test target
        model_name: Name for logging

    Returns:
        Dictionary of metrics
    """
    logger.info(f"Evaluating {model_name} on test set...")

    # Make predictions
    y_pred = model.predict(X_test)
    y_pred_proba = model.predict_proba(X_test)[:, 1]

    # Calculate metrics
    metrics = {
        'accuracy': accuracy_score(y_test, y_pred),
        'precision': precision_score(y_test, y_pred),
        'recall': recall_score(y_test, y_pred),
        'f1_score': f1_score(y_test, y_pred),
        'roc_auc': roc_auc_score(y_test, y_pred_proba)
    }

    # Confusion matrix
    cm = confusion_matrix(y_test, y_pred)

    # Log results
    logger.info("\n" + "=" * 60)
    logger.info(f"{model_name.upper()} EVALUATION RESULTS")
    logger.info("=" * 60)
    logger.info(f"\nAccuracy:  {metrics['accuracy']:.4f} ({metrics['accuracy']*100:.2f}%)")
    logger.info(f"Precision: {metrics['precision']:.4f} ({metrics['precision']*100:.2f}%)")
    logger.info(f"Recall:    {metrics['recall']:.4f} ({metrics['recall']*100:.2f}%)")
    logger.info(f"F1 Score:  {metrics['f1_score']:.4f}")
    logger.info(f"ROC-AUC:   {metrics['roc_auc']:.4f}")

    logger.info("\nConfusion Matrix:")
    logger.info("                Predicted")
    logger.info("              No Default  Default")
    logger.info(f"Actual No     {cm[0][0]:6d}      {cm[0][1]:6d}")
    logger.info(f"Actual Yes    {cm[1][0]:6d}      {cm[1][1]:6d}")

    logger.info("\nDetailed Classification Report:")
    logger.info("\n" + classification_report(
        y_test, y_pred,
        target_names=['No Default', 'Default']
    ))

    logger.info("=" * 60)

    return metrics, cm


def get_xgboost_feature_importance(model, feature_names):
    """
    Get feature importance from XGBoost model.

    Args:
        model: Trained XGBoost model
        feature_names: List of feature names

    Returns:
        DataFrame with features and importance scores
    """
    # Get importance scores
    importance_scores = model.feature_importances_

    # Create DataFrame
    importance_df = pd.DataFrame({
        'feature': feature_names,
        'importance': importance_scores
    })

    # Sort by importance
    importance_df = importance_df.sort_values('importance', ascending=False)

    logger.info("\n" + "=" * 60)
    logger.info("TOP 10 MOST IMPORTANT FEATURES (XGBoost)")
    logger.info("=" * 60)
    logger.info("\nFeature                          Importance")
    logger.info("-" * 60)

    for idx, row in importance_df.head(10).iterrows():
        logger.info(f"{row['feature']:30s} {row['importance']:8.4f}")

    logger.info("=" * 60)

    return importance_df


def compare_models(lr_metrics, xgb_metrics):
    """
    Compare performance of different models.

    Args:
        lr_metrics: Logistic Regression metrics
        xgb_metrics: XGBoost metrics
    """
    logger.info("\n" + "=" * 60)
    logger.info("MODEL COMPARISON")
    logger.info("=" * 60)

    logger.info("\n{:20s} {:>15s} {:>15s} {:>10s}".format(
        "Metric", "Logistic Reg", "XGBoost", "Winner"
    ))
    logger.info("-" * 60)

    for metric in ['accuracy', 'precision', 'recall', 'f1_score', 'roc_auc']:
        lr_val = lr_metrics[metric]
        xgb_val = xgb_metrics[metric]
        winner = "XGBoost" if xgb_val > lr_val else ("Tie" if xgb_val == lr_val else "LogReg")

        logger.info("{:20s} {:15.4f} {:15.4f} {:>10s}".format(
            metric.replace('_', ' ').title(),
            lr_val,
            xgb_val,
            winner
        ))

    logger.info("=" * 60)


def save_model(model, metrics, model_name, output_dir="data/models"):
    """
    Save trained model and metrics.

    Args:
        model: Trained model
        metrics: Dictionary of evaluation metrics
        model_name: Name of the model
        output_dir: Directory to save model
    """
    output_path = Path(output_dir)
    output_path.mkdir(parents=True, exist_ok=True)

    # Save model
    model_file = output_path / f"{model_name}.pkl"
    with open(model_file, 'wb') as f:
        pickle.dump(model, f)
    logger.info(f"Model saved to {model_file}")

    # Save metrics
    metrics_file = output_path / f"{model_name}_metrics.json"
    with open(metrics_file, 'w') as f:
        json.dump(metrics, f, indent=2)
    logger.info(f"Metrics saved to {metrics_file}")


def main():
    """Main training pipeline."""

    # Load data
    X_train, X_test, y_train, y_test = load_processed_data()

    # Train Logistic Regression
    logger.info("\n" + "🔹" * 30)
    logger.info("TRAINING LOGISTIC REGRESSION")
    logger.info("🔹" * 30)
    lr_model = train_logistic_regression(X_train, y_train)
    lr_metrics, lr_cm = evaluate_model(lr_model, X_test, y_test, "Logistic Regression")
    save_model(lr_model, lr_metrics, "logistic_regression")

    # Train XGBoost
    logger.info("\n" + "🔹" * 30)
    logger.info("TRAINING XGBOOST")
    logger.info("🔹" * 30)
    xgb_model = train_xgboost(X_train, y_train, X_test, y_test)
    xgb_metrics, xgb_cm = evaluate_model(xgb_model, X_test, y_test, "XGBoost")

    # Feature importance
    xgb_importance = get_xgboost_feature_importance(xgb_model, X_train.columns.tolist())

    # Save XGBoost
    save_model(xgb_model, xgb_metrics, "xgboost")

    # Compare models
    compare_models(lr_metrics, xgb_metrics)

    # Summary
    logger.info("\n" + "✓" * 60)
    logger.info("TRAINING COMPLETE!")
    logger.info("✓" * 60)
    logger.info(f"\nLogistic Regression - Accuracy: {lr_metrics['accuracy']*100:.2f}%, ROC-AUC: {lr_metrics['roc_auc']:.4f}")
    logger.info(f"XGBoost             - Accuracy: {xgb_metrics['accuracy']*100:.2f}%, ROC-AUC: {xgb_metrics['roc_auc']:.4f}")

    if xgb_metrics['roc_auc'] > lr_metrics['roc_auc']:
        logger.info("\n🏆 Winner: XGBoost")
    else:
        logger.info("\n🏆 Winner: Logistic Regression")


if __name__ == "__main__":
    main()
```

**XGBoost Concepts Explained:**

1. **max_depth:** How deep each tree can grow
   - Deeper = more complex patterns, but more overfitting risk

2. **learning_rate:** How much each tree contributes
   - Lower = slower learning, but often better results
   - Typical range: 0.01 to 0.3

3. **n_estimators:** Number of trees to build
   - More trees = better performance, but slower training
   - We use early stopping to find optimal number

4. **subsample:** Fraction of data for each tree
   - 0.8 = use 80% of data (randomly sampled)
   - Prevents overfitting

5. **colsample_bytree:** Fraction of features for each tree
   - 0.8 = use 80% of features
   - Adds randomness, improves generalization

6. **scale_pos_weight:** Handles class imbalance
   - If 70% no-default, 30% default → weight = 70/30 = 2.33

7. **Early Stopping:** Stops when validation score stops improving
   - Prevents wasting time and overfitting

"""
Make predictions using trained models.
"""

import pandas as pd
import numpy as np
import pickle
from pathlib import Path
from typing import Dict, Any, Optional

from src.data.feature_engineering import create_features, get_feature_names
from src.data.preprocessor import LoanDataPreprocessor
from src.utils.logger import logger


class LoanRiskPredictor:
    """
    Predicts loan default risk using trained models.
    Supports multiple models: logistic_regression, xgboost
    """

    def __init__(self, model_name: str = "xgboost"):
        self.model_name = model_name
        self.model = None
        self.preprocessor = None
        self.feature_names = None

    def load_model(self, model_name: Optional[str] = None):
        """
        Load trained model and preprocessor.

        Args:
            model_name: Override default model name
        """
        if model_name:
            self.model_name = model_name

        logger.info(f"Loading {self.model_name} model...")

        # Load model
        model_path = Path(f"data/models/{self.model_name}.pkl")

        if not model_path.exists():
            raise FileNotFoundError(
                f"Model file not found: {model_path}. "
                f"Run 'python src/models/train.py' first."
            )

        with open(model_path, 'rb') as f:
            self.model = pickle.load(f)

        # Load preprocessor
        self.preprocessor = LoanDataPreprocessor()
        self.preprocessor.load()

        self.feature_names = self.preprocessor.feature_names

        logger.info(f"✓ {self.model_name} model loaded successfully")

    def predict_single(self, application_data: Dict[str, Any]) -> Dict[str, Any]:
        """
        Predict default risk for a single loan application.

        Args:
            application_data: Dictionary with loan application details

        Returns:
            Dictionary with prediction results
        """
        if self.model is None:
            self.load_model()

        # Convert to DataFrame
        df = pd.DataFrame([application_data])

        # Feature engineering
        df = create_features(df)

        # Encode categorical features
        numerical_features, categorical_features = get_feature_names()

        for col in categorical_features:
            if col in df.columns and col in self.preprocessor.label_encoders:
                df[col] = self.preprocessor.label_encoders[col].transform(
                    df[col].astype(str)
                )

        # Select and scale features
        X = df[self.feature_names]

        # Scale numerical features
        X_scaled = X.copy()
        X_scaled[numerical_features] = self.preprocessor.scaler.transform(
            X[numerical_features]
        )

        # Make prediction
        prediction = self.model.predict(X_scaled)[0]
        probability = self.model.predict_proba(X_scaled)[0]

        # Categorize risk
        default_prob = probability[1]

        if default_prob < 0.2:
            risk_category = "VERY_LOW"
            recommendation = "APPROVE - Excellent candidate"
        elif default_prob < 0.4:
            risk_category = "LOW"
            recommendation = "APPROVE - Good candidate"
        elif default_prob < 0.6:
            risk_category = "MEDIUM"
            recommendation = "REVIEW - Manual review recommended"
        elif default_prob < 0.8:
            risk_category = "HIGH"
            recommendation = "REJECT - High risk of default"
        else:
            risk_category = "VERY_HIGH"
            recommendation = "REJECT - Very high risk"

        result = {
            'model_used': self.model_name,
            'predicted_default': int(prediction),
            'default_probability': float(default_prob),
            'no_default_probability': float(probability[0]),
            'risk_category': risk_category,
            'recommendation': recommendation
        }

        logger.info(f"[{self.model_name}] Prediction: {risk_category} (p={default_prob:.3f})")

        return result


# Global predictor instance (defaults to XGBoost)
predictor = LoanRiskPredictor(model_name="xgboost")

"""
Model management API endpoints.
"""

from fastapi import APIRouter, HTTPException
from pathlib import Path
import json

from src.models.predict import LoanRiskPredictor
from src.utils.logger import logger

router = APIRouter(prefix="/api/v1/models", tags=["Models"])


@router.get("/list")
async def list_available_models():
    """
    List all available trained models.

    Returns model names and their performance metrics.
    """
    models_dir = Path("data/models")

    if not models_dir.exists():
        raise HTTPException(
            status_code=404,
            detail="Models directory not found. Train models first."
        )

    # Find all model files
    model_files = list(models_dir.glob("*.pkl"))
    model_names = [f.stem for f in model_files]

    # Load metrics for each model
    models_info = []

    for model_name in model_names:
        metrics_file = models_dir / f"{model_name}_metrics.json"

        if metrics_file.exists():
            with open(metrics_file, 'r') as f:
                metrics = json.load(f)

            models_info.append({
                "model_name": model_name,
                "accuracy": round(metrics.get('accuracy', 0) * 100, 2),
                "precision": round(metrics.get('precision', 0) * 100, 2),
                "recall": round(metrics.get('recall', 0) * 100, 2),
                "f1_score": round(metrics.get('f1_score', 0), 4),
                "roc_auc": round(metrics.get('roc_auc', 0), 4)
            })
        else:
            models_info.append({
                "model_name": model_name,
                "metrics": "Not available"
            })

    return {
        "total_models": len(models_info),
        "models": models_info
    }


@router.get("/compare")
async def compare_models():
    """
    Compare all available models side-by-side.
    """
    models_dir = Path("data/models")
    metrics_files = list(models_dir.glob("*_metrics.json"))

    if not metrics_files:
        raise HTTPException(
            status_code=404,
            detail="No model metrics found. Train models first."
        )

    comparison = {}

    for metrics_file in metrics_files:
        model_name = metrics_file.stem.replace('_metrics', '')

        with open(metrics_file, 'r') as f:
            metrics = json.load(f)

        comparison[model_name] = {
            "accuracy": round(metrics.get('accuracy', 0) * 100, 2),
            "precision": round(metrics.get('precision', 0) * 100, 2),
            "recall": round(metrics.get('recall', 0) * 100, 2),
            "f1_score": round(metrics.get('f1_score', 0), 4),
            "roc_auc": round(metrics.get('roc_auc', 0), 4)
        }

    # Determine best model for each metric
    best_models = {}
    metrics_list = ['accuracy', 'precision', 'recall', 'f1_score', 'roc_auc']

    for metric in metrics_list:
        scores = {name: data[metric] for name, data in comparison.items()}
        best_model = max(scores, key=scores.get)
        best_models[metric] = {
            "model": best_model,
            "score": scores[best_model]
        }

    return {
        "models": comparison,
        "best_by_metric": best_models
    }
```
