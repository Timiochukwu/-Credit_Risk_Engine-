## Understanding MLflow

**What is MLflow?**
An open-source platform for managing the ML lifecycle, including experimentation, reproducibility, and deployment.

**Key components:**

1. **MLflow Tracking:**
   - Log parameters (learning_rate, max_depth, etc.)
   - Log metrics (accuracy, ROC-AUC, etc.)
   - Save artifacts (models, plots, etc.)
   - Compare runs

2. **MLflow Models:**
   - Standard format for packaging models
   - Deploy anywhere (REST API, cloud, edge)

3. **MLflow Model Registry:**
   - Centralized model store
   - Version management
   - Staging → Production workflow

**Why MLflow?**
- ✅ Track every experiment automatically
- ✅ Reproduce results exactly
- ✅ Compare 100s of runs easily
- ✅ Team collaboration
- ✅ Industry standard (Databricks, Netflix, Walmart)

```python
"""
Train machine learning models with MLflow tracking.
"""

import pandas as pd
import numpy as np
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import (
    accuracy_score,
    precision_score,
    recall_score,
    f1_score,
    roc_auc_score,
    confusion_matrix,
    classification_report
)
import xgboost as xgb
from pathlib import Path
import pickle
import json
import mlflow
import mlflow.sklearn
import mlflow.xgboost

from src.utils.logger import logger


# Set MLflow tracking URI
mlflow.set_tracking_uri("file:./mlruns")


def load_processed_data():
    """Load preprocessed training and test data."""
    logger.info("Loading preprocessed data...")

    data_dir = Path("data/processed")

    X_train = pd.read_csv(data_dir / "X_train.csv")
    X_test = pd.read_csv(data_dir / "X_test.csv")
    y_train = pd.read_csv(data_dir / "y_train.csv")['defaulted']
    y_test = pd.read_csv(data_dir / "y_test.csv")['defaulted']

    logger.info(f"Training set: {len(X_train)} samples")
    logger.info(f"Test set: {len(X_test)} samples")

    return X_train, X_test, y_train, y_test


def train_logistic_regression_with_mlflow(X_train, X_test, y_train, y_test):
    """
    Train Logistic Regression with MLflow tracking.
    """
    logger.info("Training Logistic Regression with MLflow tracking...")

    # Start MLflow run
    with mlflow.start_run(run_name="logistic_regression") as run:

        # Log parameters
        params = {
            'model_type': 'LogisticRegression',
            'C': 1.0,
            'max_iter': 1000,
            'solver': 'lbfgs',
            'class_weight': 'balanced'
        }
        mlflow.log_params(params)

        # Train model
        model = LogisticRegression(**{k: v for k, v in params.items() if k != 'model_type'})
        model.fit(X_train, y_train)

        # Make predictions
        y_pred = model.predict(X_test)
        y_pred_proba = model.predict_proba(X_test)[:, 1]

        # Calculate metrics
        metrics = {
            'accuracy': accuracy_score(y_test, y_pred),
            'precision': precision_score(y_test, y_pred),
            'recall': recall_score(y_test, y_pred),
            'f1_score': f1_score(y_test, y_pred),
            'roc_auc': roc_auc_score(y_test, y_pred_proba),
            'train_samples': len(X_train),
            'test_samples': len(X_test)
        }

        # Log metrics to MLflow
        mlflow.log_metrics(metrics)

        # Log confusion matrix as dict
        cm = confusion_matrix(y_test, y_pred)
        mlflow.log_metric("true_negatives", int(cm[0][0]))
        mlflow.log_metric("false_positives", int(cm[0][1]))
        mlflow.log_metric("false_negatives", int(cm[1][0]))
        mlflow.log_metric("true_positives", int(cm[1][1]))

        # Log model
        mlflow.sklearn.log_model(model, "model")

        # Log additional artifacts
        # Save feature importance
        importance_df = pd.DataFrame({
            'feature': X_train.columns,
            'coefficient': np.abs(model.coef_[0])
        }).sort_values('coefficient', ascending=False)

        importance_file = "feature_importance.csv"
        importance_df.to_csv(importance_file, index=False)
        mlflow.log_artifact(importance_file)
        Path(importance_file).unlink()  # Clean up

        # Log tags
        mlflow.set_tags({
            "algorithm": "LogisticRegression",
            "dataset": "Nigerian Loan Applications",
            "framework": "scikit-learn"
        })

        logger.info(f"MLflow Run ID: {run.info.run_id}")
        logger.info(f"Accuracy: {metrics['accuracy']:.4f}")
        logger.info(f"ROC-AUC: {metrics['roc_auc']:.4f}")

        return model, metrics, run.info.run_id


def train_xgboost_with_mlflow(X_train, X_test, y_train, y_test):
    """
    Train XGBoost with MLflow tracking.
    """
    logger.info("Training XGBoost with MLflow tracking...")

    # Start MLflow run
    with mlflow.start_run(run_name="xgboost") as run:

        # Calculate scale_pos_weight
        n_positive = (y_train == 1).sum()
        n_negative = (y_train == 0).sum()
        scale_pos_weight = n_negative / n_positive

        # Parameters
        params = {
            'model_type': 'XGBoost',
            'objective': 'binary:logistic',
            'eval_metric': 'auc',
            'max_depth': 6,
            'learning_rate': 0.1,
            'n_estimators': 200,
            'subsample': 0.8,
            'colsample_bytree': 0.8,
            'scale_pos_weight': scale_pos_weight,
            'random_state': 42,
            'tree_method': 'hist'
        }

        # Log parameters
        mlflow.log_params(params)

        # Train model
        model = xgb.XGBClassifier(**{k: v for k, v in params.items() if k != 'model_type'})

        model.fit(
            X_train, y_train,
            eval_set=[(X_test, y_test)],
            early_stopping_rounds=20,
            verbose=False
        )

        # Make predictions
        y_pred = model.predict(X_test)
        y_pred_proba = model.predict_proba(X_test)[:, 1]

        # Calculate metrics
        metrics = {
            'accuracy': accuracy_score(y_test, y_pred),
            'precision': precision_score(y_test, y_pred),
            'recall': recall_score(y_test, y_pred),
            'f1_score': f1_score(y_test, y_pred),
            'roc_auc': roc_auc_score(y_test, y_pred_proba),
            'best_iteration': int(model.best_iteration),
            'train_samples': len(X_train),
            'test_samples': len(X_test)
        }

        # Log metrics
        mlflow.log_metrics(metrics)

        # Log confusion matrix
        cm = confusion_matrix(y_test, y_pred)
        mlflow.log_metric("true_negatives", int(cm[0][0]))
        mlflow.log_metric("false_positives", int(cm[0][1]))
        mlflow.log_metric("false_negatives", int(cm[1][0]))
        mlflow.log_metric("true_positives", int(cm[1][1]))

        # Log model
        mlflow.xgboost.log_model(model, "model")

        # Log feature importance
        importance_df = pd.DataFrame({
            'feature': X_train.columns,
            'importance': model.feature_importances_
        }).sort_values('importance', ascending=False)

        importance_file = "feature_importance.csv"
        importance_df.to_csv(importance_file, index=False)
        mlflow.log_artifact(importance_file)
        Path(importance_file).unlink()

        # Log tags
        mlflow.set_tags({
            "algorithm": "XGBoost",
            "dataset": "Nigerian Loan Applications",
            "framework": "xgboost"
        })

        logger.info(f"MLflow Run ID: {run.info.run_id}")
        logger.info(f"Accuracy: {metrics['accuracy']:.4f}")
        logger.info(f"ROC-AUC: {metrics['roc_auc']:.4f}")

        return model, metrics, run.info.run_id


def save_model(model, metrics, model_name, output_dir="data/models"):
    """Save model and metrics (keeping backward compatibility)."""
    output_path = Path(output_dir)
    output_path.mkdir(parents=True, exist_ok=True)

    # Save model
    model_file = output_path / f"{model_name}.pkl"
    with open(model_file, 'wb') as f:
        pickle.dump(model, f)
    logger.info(f"Model saved to {model_file}")

    # Save metrics
    metrics_file = output_path / f"{model_name}_metrics.json"
    with open(metrics_file, 'w') as f:
        json.dump(metrics, f, indent=2)
    logger.info(f"Metrics saved to {metrics_file}")


def main():
    """Main training pipeline with MLflow."""

    # Set experiment name
    mlflow.set_experiment("loan-default-prediction")

    logger.info("\n" + "=" * 60)
    logger.info("TRAINING WITH MLFLOW TRACKING")
    logger.info("=" * 60)

    # Load data
    X_train, X_test, y_train, y_test = load_processed_data()

    # Train Logistic Regression
    logger.info("\n" + "🔹" * 30)
    logger.info("TRAINING LOGISTIC REGRESSION")
    logger.info("🔹" * 30)
    lr_model, lr_metrics, lr_run_id = train_logistic_regression_with_mlflow(
        X_train, X_test, y_train, y_test
    )
    save_model(lr_model, lr_metrics, "logistic_regression")

    # Train XGBoost
    logger.info("\n" + "🔹" * 30)
    logger.info("TRAINING XGBOOST")
    logger.info("🔹" * 30)
    xgb_model, xgb_metrics, xgb_run_id = train_xgboost_with_mlflow(
        X_train, X_test, y_train, y_test
    )
    save_model(xgb_model, xgb_metrics, "xgboost")

    # Summary
    logger.info("\n" + "=" * 60)
    logger.info("TRAINING COMPLETE")
    logger.info("=" * 60)
    logger.info(f"\nLogistic Regression:")
    logger.info(f"  Run ID: {lr_run_id}")
    logger.info(f"  Accuracy: {lr_metrics['accuracy']*100:.2f}%")
    logger.info(f"  ROC-AUC: {lr_metrics['roc_auc']:.4f}")

    logger.info(f"\nXGBoost:")
    logger.info(f"  Run ID: {xgb_run_id}")
    logger.info(f"  Accuracy: {xgb_metrics['accuracy']*100:.2f}%")
    logger.info(f"  ROC-AUC: {xgb_metrics['roc_auc']:.4f}")

    logger.info("\n" + "=" * 60)
    logger.info("View results in MLflow UI:")
    logger.info("  Run: mlflow ui")
    logger.info("  Then open: http://localhost:5000")
    logger.info("=" * 60)


if __name__ == "__main__":
    main()
```

**MLflow Concepts Explained:**

1. **Tracking URI:** Where to store experiment data
   - `file:./mlruns` = local directory
   - Can use remote servers in production

2. **Experiment:** Group of related runs
   - Example: "loan-default-prediction"
   - Organize by project

3. **Run:** Single execution of training code
   - Each model training = one run
   - Automatically tracked

4. **Parameters:** Configuration values
   - learning_rate, max_depth, etc.
   - Logged once per run

5. **Metrics:** Performance measures
   - accuracy, ROC-AUC, etc.
   - Can log multiple times (for tracking over epochs)

6. **Artifacts:** Files (models, plots, data)
   - Saved for reproducibility

7. **Tags:** Metadata labels
   - Algorithm, dataset, framework

## Step 5: Create MLflow Query Module

**File:** `src/models/mlflow_utils.py`

```python
"""
Utilities for querying MLflow experiments.
"""

import mlflow
from mlflow.tracking import MlflowClient
import pandas as pd
from typing import List, Dict, Optional

from src.utils.logger import logger


mlflow.set_tracking_uri("file:./mlruns")
client = MlflowClient()


def list_experiments() -> List[Dict]:
    """
    List all MLflow experiments.

    Returns:
        List of experiment dictionaries
    """
    experiments = client.search_experiments()

    results = []
    for exp in experiments:
        results.append({
            'experiment_id': exp.experiment_id,
            'name': exp.name,
            'artifact_location': exp.artifact_location,
            'lifecycle_stage': exp.lifecycle_stage
        })

    return results


def get_best_run(experiment_name: str, metric: str = "roc_auc") -> Optional[Dict]:
    """
    Get the best run from an experiment based on a metric.

    Args:
        experiment_name: Name of the experiment
        metric: Metric to optimize (default: roc_auc)

    Returns:
        Dictionary with best run details
    """
    experiment = client.get_experiment_by_name(experiment_name)

    if not experiment:
        logger.warning(f"Experiment '{experiment_name}' not found")
        return None

    # Search runs
    runs = client.search_runs(
        experiment_ids=[experiment.experiment_id],
        order_by=[f"metrics.{metric} DESC"],
        max_results=1
    )

    if not runs:
        logger.warning(f"No runs found in experiment '{experiment_name}'")
        return None

    best_run = runs[0]

    return {
        'run_id': best_run.info.run_id,
        'run_name': best_run.data.tags.get('mlflow.runName', 'Unknown'),
        'metrics': best_run.data.metrics,
        'params': best_run.data.params,
        'tags': best_run.data.tags,
        'start_time': best_run.info.start_time,
        'end_time': best_run.info.end_time
    }


def compare_runs(experiment_name: str, metric_names: List[str] = None) -> pd.DataFrame:
    """
    Compare all runs in an experiment.

    Args:
        experiment_name: Name of the experiment
        metric_names: Metrics to include (None = all)

    Returns:
        DataFrame with run comparison
    """
    experiment = client.get_experiment_by_name(experiment_name)

    if not experiment:
        logger.warning(f"Experiment '{experiment_name}' not found")
        return pd.DataFrame()

    runs = client.search_runs(experiment_ids=[experiment.experiment_id])

    if not runs:
        return pd.DataFrame()

    # Extract run data
    data = []
    for run in runs:
        row = {
            'run_id': run.info.run_id,
            'run_name': run.data.tags.get('mlflow.runName', 'Unknown'),
            'status': run.info.status,
            'start_time': run.info.start_time
        }

        # Add parameters
        row.update({f"param_{k}": v for k, v in run.data.params.items()})

        # Add metrics
        if metric_names:
            row.update({f"metric_{k}": run.data.metrics[k]
                       for k in metric_names if k in run.data.metrics})
        else:
            row.update({f"metric_{k}": v for k, v in run.data.metrics.items()})

        data.append(row)

    df = pd.DataFrame(data)

    return df


def load_model_from_run(run_id: str):
    """
    Load a model from a specific MLflow run.

    Args:
        run_id: MLflow run ID

    Returns:
        Loaded model
    """
    model_uri = f"runs:/{run_id}/model"
    model = mlflow.pyfunc.load_model(model_uri)

    logger.info(f"Model loaded from run {run_id}")

    return model


def get_run_metrics(run_id: str) -> Dict:
    """
    Get metrics for a specific run.

    Args:
        run_id: MLflow run ID

    Returns:
        Dictionary of metrics
    """
    run = client.get_run(run_id)
    return run.data.metrics


def main():
    """Example usage of MLflow utilities."""

    logger.info("=" * 60)
    logger.info("MLFLOW EXPERIMENTS QUERY")
    logger.info("=" * 60)

    # List experiments
    logger.info("\nExperiments:")
    experiments = list_experiments()
    for exp in experiments:
        logger.info(f"  - {exp['name']} (ID: {exp['experiment_id']})")

    # Get best run
    logger.info("\nBest run (by ROC-AUC):")
    best_run = get_best_run("loan-default-prediction", metric="roc_auc")

    if best_run:
        logger.info(f"  Run Name: {best_run['run_name']}")
        logger.info(f"  Run ID: {best_run['run_id']}")
        logger.info(f"  ROC-AUC: {best_run['metrics'].get('roc_auc', 'N/A'):.4f}")
        logger.info(f"  Accuracy: {best_run['metrics'].get('accuracy', 'N/A'):.4f}")

    # Compare runs
    logger.info("\nRun Comparison:")
    comparison_df = compare_runs(
        "loan-default-prediction",
        metric_names=['accuracy', 'roc_auc', 'precision', 'recall']
    )

    if not comparison_df.empty:
        logger.info(f"\n{comparison_df.to_string()}")

    logger.info("\n" + "=" * 60)


if __name__ == "__main__":
    main()
```

"""
MLflow experiment tracking API endpoints.
"""

from fastapi import APIRouter, HTTPException
from typing import List, Optional

from src.models.mlflow_utils import (
    list_experiments,
    get_best_run,
    compare_runs,
    get_run_metrics
)
from src.utils.logger import logger

router = APIRouter(prefix="/api/v1/mlflow", tags=["MLflow"])


@router.get("/experiments")
async def get_experiments():
    """
    List all MLflow experiments.
    """
    experiments = list_experiments()
    return {
        "total_experiments": len(experiments),
        "experiments": experiments
    }


@router.get("/experiments/{experiment_name}/best")
async def get_best_model(
    experiment_name: str,
    metric: str = "roc_auc"
):
    """
    Get the best run from an experiment.

    Args:
        experiment_name: Name of the experiment
        metric: Metric to optimize (roc_auc, accuracy, etc.)
    """
    best_run = get_best_run(experiment_name, metric)

    if not best_run:
        raise HTTPException(
            status_code=404,
            detail=f"No runs found in experiment: {experiment_name}"
        )

    return best_run


@router.get("/experiments/{experiment_name}/compare")
async def compare_experiment_runs(
    experiment_name: str,
    metrics: Optional[str] = None
):
    """
    Compare all runs in an experiment.

    Args:
        experiment_name: Name of the experiment
        metrics: Comma-separated list of metrics to include
    """
    metric_list = metrics.split(',') if metrics else None

    df = compare_runs(experiment_name, metric_list)

    if df.empty:
        raise HTTPException(
            status_code=404,
            detail=f"No runs found in experiment: {experiment_name}"
        )

    return {
        "experiment_name": experiment_name,
        "total_runs": len(df),
        "runs": df.to_dict('records')
    }


@router.get("/runs/{run_id}/metrics")
async def get_run_metrics_endpoint(run_id: str):
    """
    Get metrics for a specific run.
    """
    try:
        metrics = get_run_metrics(run_id)
        return {
            "run_id": run_id,
            "metrics": metrics
        }
    except Exception as e:
        raise HTTPException(
            status_code=404,
            detail=f"Run not found: {run_id}"
        )

## Best Practices

**What to log:**
- ✅ All hyperparameters
- ✅ Final metrics (accuracy, ROC-AUC, etc.)
- ✅ Training time
- ✅ Data version/size
- ✅ Model artifacts
- ✅ Feature importance

**Naming conventions:**
- Experiment: `project-name-task`
- Run: `algorithm-name` or `algorithm-version`
- Tags: algorithm, dataset, framework

**Organization:**
- One experiment per project
- Consistent parameter names
- Descriptive run names

## Troubleshooting

**Issue:** "mlruns" folder not created
**Solution:** Run training script first, MLflow creates it automatically

**Issue:** UI shows no experiments
**Solution:** Check tracking URI matches: `mlflow.set_tracking_uri("file:./mlruns")`

**Issue:** Can't load model from run
**Solution:** Verify model was logged: `mlflow.<framework>.log_model()`

**Issue:** Metrics not showing in UI
**Solution:** Ensure metrics logged before run ends
