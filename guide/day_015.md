## Understanding Model Explainability

**Why explainability matters:**

1. **Regulatory Compliance:**
   - GDPR "right to explanation"
   - Fair lending laws
   - Banks must explain loan decisions

2. **Trust:**
   - Users want to know "why was I rejected?"
   - Builds confidence in AI systems

3. **Debugging:**
   - Find model biases
   - Identify data issues
   - Improve features

4. **Business Insights:**
   - Understand what drives risk
   - Inform policy decisions

**Black Box vs Glass Box:**
- **Black Box:** Neural nets, ensembles (accurate but opaque)
- **Glass Box:** Linear models, decision trees (interpretable but less accurate)
- **Solution:** Make black boxes explainable!

**SHAP vs LIME:**

| Feature | SHAP | LIME |
|---------|------|------|
| Method | Game theory | Local linear approximation |
| Speed | Slower | Faster |
| Consistency | Yes | No |
| Global/Local | Both | Local only |
| Best for | Tree models | Any model |

## Step 2: Create SHAP Explainer Module

**File:** `src/models/explainability.py`

```python
"""
Model explainability using SHAP and LIME.
"""

import pandas as pd
import numpy as np
import shap
import lime
import lime.lime_tabular
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from pathlib import Path
import pickle

from src.utils.logger import logger


class ModelExplainer:
    """
    Provides explanations for model predictions using SHAP and LIME.
    """

    def __init__(self, model, model_type="tree"):
        """
        Initialize explainer.

        Args:
            model: Trained model
            model_type: "tree" for tree-based models, "linear" for linear models
        """
        self.model = model
        self.model_type = model_type
        self.shap_explainer = None
        self.lime_explainer = None

    def initialize_shap(self, X_background):
        """
        Initialize SHAP explainer.

        Args:
            X_background: Background dataset for SHAP (typically training data sample)
        """
        logger.info("Initializing SHAP explainer...")

        if self.model_type == "tree":
            # TreeExplainer for tree-based models (XGBoost, LightGBM, RF)
            self.shap_explainer = shap.TreeExplainer(self.model)
        else:
            # KernelExplainer for any model (slower but works for all)
            self.shap_explainer = shap.KernelExplainer(
                self.model.predict_proba,
                X_background
            )

        logger.info("SHAP explainer initialized")

    def initialize_lime(self, X_train, feature_names, class_names=None):
        """
        Initialize LIME explainer.

        Args:
            X_train: Training data
            feature_names: List of feature names
            class_names: List of class names (e.g., ['No Default', 'Default'])
        """
        logger.info("Initializing LIME explainer...")

        self.lime_explainer = lime.lime_tabular.LimeTabularExplainer(
            training_data=X_train.values,
            feature_names=feature_names,
            class_names=class_names or ['Class 0', 'Class 1'],
            mode='classification',
            random_state=42
        )

        logger.info("LIME explainer initialized")

    def explain_with_shap(self, X_instance):
        """
        Get SHAP values for a single instance.

        Args:
            X_instance: Single instance to explain (DataFrame or numpy array)

        Returns:
            SHAP values
        """
        if self.shap_explainer is None:
            raise ValueError("SHAP explainer not initialized. Call initialize_shap() first.")

        shap_values = self.shap_explainer.shap_values(X_instance)

        # For binary classification, shap_values might be a list
        if isinstance(shap_values, list):
            shap_values = shap_values[1]  # Get values for positive class

        return shap_values

    def explain_with_lime(self, X_instance, num_features=10):
        """
        Get LIME explanation for a single instance.

        Args:
            X_instance: Single instance to explain (numpy array)
            num_features: Number of top features to show

        Returns:
            LIME explanation object
        """
        if self.lime_explainer is None:
            raise ValueError("LIME explainer not initialized. Call initialize_lime() first.")

        explanation = self.lime_explainer.explain_instance(
            X_instance,
            self.model.predict_proba,
            num_features=num_features
        )

        return explanation

    def plot_shap_waterfall(self, X_instance, feature_names, output_file):
        """
        Create SHAP waterfall plot for a single prediction.

        Shows how each feature contributes to moving the prediction
        from the base value to the final prediction.

        Args:
            X_instance: Single instance
            feature_names: List of feature names
            output_file: Path to save plot
        """
        logger.info("Creating SHAP waterfall plot...")

        shap_values = self.explain_with_shap(X_instance)

        # Create explanation object for visualization
        if len(X_instance.shape) == 1:
            X_instance = X_instance.reshape(1, -1)

        explanation = shap.Explanation(
            values=shap_values[0] if len(shap_values.shape) > 1 else shap_values,
            base_values=self.shap_explainer.expected_value,
            data=X_instance[0],
            feature_names=feature_names
        )

        # Create waterfall plot
        plt.figure(figsize=(10, 8))
        shap.waterfall_plot(explanation, show=False)
        plt.tight_layout()
        plt.savefig(output_file, dpi=150, bbox_inches='tight')
        plt.close()

        logger.info(f"Waterfall plot saved to {output_file}")

    def plot_shap_force(self, X_instance, feature_names, output_file):
        """
        Create SHAP force plot.

        Shows features pushing prediction higher (red) or lower (blue).

        Args:
            X_instance: Single instance
            feature_names: List of feature names
            output_file: Path to save plot
        """
        logger.info("Creating SHAP force plot...")

        shap_values = self.explain_with_shap(X_instance)

        if len(X_instance.shape) == 1:
            X_instance = X_instance.reshape(1, -1)

        # Create force plot
        shap.force_plot(
            self.shap_explainer.expected_value,
            shap_values[0] if len(shap_values.shape) > 1 else shap_values,
            X_instance[0],
            feature_names=feature_names,
            matplotlib=True,
            show=False
        )

        plt.savefig(output_file, dpi=150, bbox_inches='tight')
        plt.close()

        logger.info(f"Force plot saved to {output_file}")

    def get_top_features_explanation(self, X_instance, feature_names, n=10):
        """
        Get top N features contributing to prediction.

        Args:
            X_instance: Single instance
            feature_names: List of feature names
            n: Number of top features

        Returns:
            Dictionary with feature contributions
        """
        shap_values = self.explain_with_shap(X_instance)

        if len(shap_values.shape) > 1:
            shap_values = shap_values[0]

        # Get feature contributions
        feature_contributions = pd.DataFrame({
            'feature': feature_names,
            'shap_value': shap_values,
            'feature_value': X_instance[0] if len(X_instance.shape) > 1 else X_instance
        })

        # Sort by absolute SHAP value
        feature_contributions['abs_shap'] = feature_contributions['shap_value'].abs()
        feature_contributions = feature_contributions.sort_values('abs_shap', ascending=False)

        # Get top N
        top_features = feature_contributions.head(n)

        result = {
            'base_value': float(self.shap_explainer.expected_value),
            'prediction_value': float(self.shap_explainer.expected_value + shap_values.sum()),
            'top_features': []
        }

        for _, row in top_features.iterrows():
            result['top_features'].append({
                'feature': row['feature'],
                'feature_value': float(row['feature_value']),
                'shap_value': float(row['shap_value']),
                'effect': 'increases risk' if row['shap_value'] > 0 else 'decreases risk'
            })

        return result


def load_model_and_data():
    """Load trained model and data for explanation."""

    # Load XGBoost model
    model_path = Path("data/models/xgboost.pkl")
    with open(model_path, 'rb') as f:
        model = pickle.load(f)

    # Load data
    X_train = pd.read_csv("data/processed/X_train.csv")
    X_test = pd.read_csv("data/processed/X_test.csv")

    return model, X_train, X_test


def main():
    """
    Example: Generate explanations for a test instance.
    """
    logger.info("=" * 60)
    logger.info("MODEL EXPLAINABILITY DEMO")
    logger.info("=" * 60)

    # Load model and data
    model, X_train, X_test = load_model_and_data()

    # Create explainer
    explainer = ModelExplainer(model, model_type="tree")

    # Initialize SHAP (use sample of training data for background)
    background_data = X_train.sample(n=100, random_state=42)
    explainer.initialize_shap(background_data)

    # Initialize LIME
    explainer.initialize_lime(
        X_train,
        feature_names=list(X_train.columns),
        class_names=['No Default', 'Default']
    )

    # Explain a test instance
    test_instance = X_test.iloc[0:1].values
    feature_names = list(X_test.columns)

    logger.info(f"\nExplaining prediction for test instance 0...")

    # Get prediction
    prediction = model.predict(test_instance)[0]
    probability = model.predict_proba(test_instance)[0][1]

    logger.info(f"Prediction: {prediction} (Default)" if prediction == 1 else f"Prediction: {prediction} (No Default)")
    logger.info(f"Default Probability: {probability:.3f}")

    # SHAP explanation
    logger.info("\nGenerating SHAP explanations...")

    output_dir = Path("data/models/explanations")
    output_dir.mkdir(parents=True, exist_ok=True)

    explainer.plot_shap_waterfall(
        test_instance,
        feature_names,
        output_dir / "shap_waterfall.png"
    )

    explainer.plot_shap_force(
        test_instance,
        feature_names,
        output_dir / "shap_force.png"
    )

    # Get top features
    explanation = explainer.get_top_features_explanation(
        test_instance,
        feature_names,
        n=10
    )

    logger.info("\nTop 10 Feature Contributions:")
    logger.info(f"Base value (average): {explanation['base_value']:.3f}")
    logger.info(f"Prediction value: {explanation['prediction_value']:.3f}")

    for feat in explanation['top_features']:
        sign = '+' if feat['shap_value'] > 0 else ''
        logger.info(
            f"  {feat['feature']:30s} = {feat['feature_value']:8.2f} → "
            f"{sign}{feat['shap_value']:7.3f} ({feat['effect']})"
        )

    # LIME explanation
    logger.info("\nGenerating LIME explanation...")
    lime_exp = explainer.explain_with_lime(test_instance[0], num_features=10)

    logger.info("\nLIME Top Features:")
    for feature, weight in lime_exp.as_list():
        logger.info(f"  {feature}: {weight:+.3f}")

    # Save LIME plot
    fig = lime_exp.as_pyplot_figure()
    fig.savefig(output_dir / "lime_explanation.png", dpi=150, bbox_inches='tight')
    plt.close()

    logger.info(f"\nExplanation plots saved to: {output_dir}")
    logger.info("=" * 60)


if __name__ == "__main__":
    main()
```

**SHAP Concepts Explained:**

1. **Shapley Values:**
   - From game theory (Nobel Prize 1994)
   - Fair allocation of credit among players
   - Applied to features: how much does each feature contribute?

2. **Base Value:**
   - Average prediction across training data
   - Starting point for explanation

3. **SHAP Value:**
   - How much a feature moves prediction from base value
   - Positive = increases risk
   - Negative = decreases risk

4. **Waterfall Plot:**
   ```
   Base value (0.32)
   + debt_to_income_ratio (0.15)
   + num_existing_loans (0.08)
   - credit_score (-0.12)
   - monthly_income (-0.05)
   = Final prediction (0.38)
   ```

"""
Model explainability API endpoints.
"""

from fastapi import APIRouter, HTTPException
from typing import Dict, Any
import pickle
from pathlib import Path

from src.models.explainability import ModelExplainer
from src.api.schemas import LoanApplicationRequest
from src.utils.logger import logger

router = APIRouter(prefix="/api/v1/explain", tags=["Explainability"])


# Load model at startup
model_path = Path("data/models/xgboost.pkl")
if model_path.exists():
    with open(model_path, 'rb') as f:
        global_model = pickle.load(f)
    global_explainer = ModelExplainer(global_model, model_type="tree")

    # Load background data
    import pandas as pd
    X_train = pd.read_csv("data/processed/X_train.csv")
    background = X_train.sample(n=100, random_state=42)
    global_explainer.initialize_shap(background)
    global_explainer.initialize_lime(
        X_train,
        feature_names=list(X_train.columns),
        class_names=['No Default', 'Default']
    )
else:
    global_model = None
    global_explainer = None


@router.post("/prediction")
async def explain_prediction(application: LoanApplicationRequest) -> Dict[str, Any]:
    """
    Explain a loan default prediction.

    Returns top features contributing to the prediction with SHAP values.
    """
    if global_explainer is None:
        raise HTTPException(
            status_code=503,
            detail="Explainer not initialized. Train model first."
        )

    try:
        # Prepare features (simplified - in production use full preprocessing)
        import numpy as np
        features = np.array([[
            application.age,
            application.monthly_income,
            application.loan_amount,
            # ... other features (would need full preprocessing)
        ] + [0] * 31])  # Pad to match feature count

        feature_names = list(pd.read_csv("data/processed/X_train.csv").columns)

        # Get explanation
        explanation = global_explainer.get_top_features_explanation(
            features,
            feature_names,
            n=10
        )

        # Get prediction
        prediction = global_model.predict(features)[0]
        probability = global_model.predict_proba(features)[0][1]

        return {
            "applicant_name": application.applicant_name,
            "prediction": int(prediction),
            "default_probability": round(float(probability), 3),
            "explanation": explanation
        }

    except Exception as e:
        logger.error(f"Explanation error: {e}")
        raise HTTPException(
            status_code=500,
            detail=f"Failed to generate explanation: {str(e)}"
        )

## Interpreting SHAP Values

**Example explanation:**

```
Applicant: Young Borrower
Prediction: 72.3% default risk (HIGH)

Base risk: 32.5% (average)

Feature contributions:
+ debt_to_income_ratio (4.2): +21.5% ← Major red flag!
+ low credit_score (520):     +14.5% ← Poor credit history
+ many existing loans (5):    + 8.7% ← Over-leveraged
- stable employment:          - 3.2% ← One positive factor
+ young age (25):             + 1.9% ← Limited experience

Final: 72.3% default risk → REJECT
```
