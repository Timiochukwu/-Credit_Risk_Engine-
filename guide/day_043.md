## Why Blockchain for Audits?

**Problem**: Traditional audit logs can be:
- **Tampered with**: Database entries can be modified
- **Deleted**: No proof if logs are removed
- **Not verifiable**: Can't prove logs weren't altered
- **Trust issues**: Requires trusting database administrators
- **Compliance gaps**: Regulations require tamper-proof records

**Solution**: Blockchain provides:
- **Immutability**: Once written, cannot be changed
- **Cryptographic verification**: Prove data integrity
- **Distributed ledger**: No single point of failure
- **Audit trail**: Complete provenance of all changes
- **Smart contracts**: Automated compliance enforcement

**Benefits**:
- **Regulatory compliance**: Meet strictest audit requirements
- **Tamper-proof**: Mathematical guarantee of integrity
- **Transparency**: Auditors can verify all records
- **Trust**: No need to trust administrators
- **Legal evidence**: Cryptographically verifiable proof

```python
import hashlib
import json
from datetime import datetime
from typing import Dict, Any, List, Optional
from dataclasses import dataclass, asdict
import logging

logger = logging.getLogger(__name__)


@dataclass
class Block:
    """Represents a block in the blockchain."""
    index: int
    timestamp: str
    data: Dict[str, Any]
    previous_hash: str
    hash: str
    nonce: int = 0

    def to_dict(self) -> dict:
        """Convert block to dictionary."""
        return asdict(self)


class BlockchainAudit:
    """
    Blockchain-based audit trail for predictions.

    Uses proof-of-work and cryptographic hashing for immutability.
    """

    def __init__(self, difficulty: int = 2):
        """
        Initialize blockchain.

        Args:
            difficulty: Mining difficulty (number of leading zeros)
        """
        self.chain: List[Block] = []
        self.difficulty = difficulty
        self.pending_data: List[Dict[str, Any]] = []

        # Create genesis block
        self.create_genesis_block()

    def create_genesis_block(self):
        """Create the first block in the chain."""
        genesis_block = Block(
            index=0,
            timestamp=datetime.now().isoformat(),
            data={"type": "genesis", "message": "Credit Risk Audit Chain"},
            previous_hash="0",
            hash="",
            nonce=0
        )
        genesis_block.hash = self.calculate_hash(genesis_block)
        self.chain.append(genesis_block)
        logger.info("Genesis block created")

    def calculate_hash(self, block: Block) -> str:
        """
        Calculate SHA-256 hash of block.

        Args:
            block: Block to hash

        Returns:
            Hexadecimal hash string
        """
        block_dict = block.to_dict()
        block_dict.pop('hash', None)  # Remove hash itself
        block_string = json.dumps(block_dict, sort_keys=True)
        return hashlib.sha256(block_string.encode()).hexdigest()

    def mine_block(self, block: Block) -> Block:
        """
        Mine block using proof-of-work.

        Args:
            block: Block to mine

        Returns:
            Mined block with valid hash
        """
        target = "0" * self.difficulty

        while not block.hash.startswith(target):
            block.nonce += 1
            block.hash = self.calculate_hash(block)

        logger.info(f"Block mined: {block.hash} (nonce: {block.nonce})")
        return block

    def add_block(self, data: Dict[str, Any]) -> Block:
        """
        Add new block to chain.

        Args:
            data: Block data (prediction, audit event, etc.)

        Returns:
            Added block
        """
        previous_block = self.chain[-1]

        new_block = Block(
            index=len(self.chain),
            timestamp=datetime.now().isoformat(),
            data=data,
            previous_hash=previous_block.hash,
            hash="",
            nonce=0
        )

        # Mine the block
        mined_block = self.mine_block(new_block)

        # Add to chain
        self.chain.append(mined_block)

        logger.info(f"Block {mined_block.index} added to chain")
        return mined_block

    def is_chain_valid(self) -> bool:
        """
        Verify blockchain integrity.

        Returns:
            True if chain is valid, False otherwise
        """
        for i in range(1, len(self.chain)):
            current_block = self.chain[i]
            previous_block = self.chain[i - 1]

            # Verify hash
            if current_block.hash != self.calculate_hash(current_block):
                logger.error(f"Invalid hash at block {i}")
                return False

            # Verify link to previous block
            if current_block.previous_hash != previous_block.hash:
                logger.error(f"Invalid previous hash at block {i}")
                return False

            # Verify proof-of-work
            if not current_block.hash.startswith("0" * self.difficulty):
                logger.error(f"Invalid proof-of-work at block {i}")
                return False

        return True

    def get_block(self, index: int) -> Optional[Block]:
        """Get block by index."""
        if 0 <= index < len(self.chain):
            return self.chain[index]
        return None

    def get_latest_block(self) -> Block:
        """Get the latest block."""
        return self.chain[-1]

    def search_blocks(
        self,
        criteria: Dict[str, Any]
    ) -> List[Block]:
        """
        Search blocks by criteria.

        Args:
            criteria: Search criteria (e.g., {"type": "prediction"})

        Returns:
            List of matching blocks
        """
        results = []

        for block in self.chain:
            match = True
            for key, value in criteria.items():
                if block.data.get(key) != value:
                    match = False
                    break

            if match:
                results.append(block)

        return results

    def to_json(self) -> str:
        """Export blockchain to JSON."""
        return json.dumps([block.to_dict() for block in self.chain], indent=2)

    def get_chain_info(self) -> Dict[str, Any]:
        """Get blockchain statistics."""
        return {
            'length': len(self.chain),
            'difficulty': self.difficulty,
            'is_valid': self.is_chain_valid(),
            'latest_block': self.get_latest_block().to_dict(),
            'total_predictions': len([
                b for b in self.chain
                if b.data.get('type') == 'prediction'
            ])
        }
```

```python
from typing import Dict, Any
import uuid
from datetime import datetime

from app.blockchain.hash_chain import blockchain, Block
from app.blockchain.storage import BlockchainStorage


class BlockchainAuditService:
    """Service for blockchain audit operations."""

    def __init__(self):
        self.blockchain = blockchain
        self.storage = BlockchainStorage()

    def record_prediction(
        self,
        prediction_id: str,
        input_data: Dict[str, Any],
        prediction: Dict[str, Any],
        user_id: str = None
    ) -> Block:
        """
        Record prediction on blockchain.

        Args:
            prediction_id: Unique prediction ID
            input_data: Input features
            prediction: Prediction result
            user_id: User who made prediction

        Returns:
            Block added to chain
        """
        block_data = {
            'type': 'prediction',
            'prediction_id': prediction_id,
            'timestamp': datetime.now().isoformat(),
            'user_id': user_id,
            'input_hash': self._hash_data(input_data),
            'prediction_hash': self._hash_data(prediction),
            'default_probability': prediction.get('default_probability'),
            'risk_category': prediction.get('risk_category')
        }

        block = self.blockchain.add_block(block_data)

        # Persist to database
        self.storage.save_block(block)

        return block

    def record_model_deployment(
        self,
        model_name: str,
        model_version: str,
        deployed_by: str
    ) -> Block:
        """Record model deployment event."""
        block_data = {
            'type': 'model_deployment',
            'model_name': model_name,
            'model_version': model_version,
            'deployed_by': deployed_by,
            'timestamp': datetime.now().isoformat()
        }

        block = self.blockchain.add_block(block_data)
        self.storage.save_block(block)

        return block

    def verify_prediction(self, prediction_id: str) -> Dict[str, Any]:
        """
        Verify prediction has not been tampered with.

        Args:
            prediction_id: Prediction ID to verify

        Returns:
            Verification result
        """
        # Find block with prediction
        blocks = self.blockchain.search_blocks({
            'type': 'prediction',
            'prediction_id': prediction_id
        })

        if not blocks:
            return {
                'verified': False,
                'reason': 'Prediction not found on blockchain'
            }

        block = blocks[0]

        # Verify chain integrity
        if not self.blockchain.is_chain_valid():
            return {
                'verified': False,
                'reason': 'Blockchain integrity compromised'
            }

        # Verify block hash
        calculated_hash = self.blockchain.calculate_hash(block)
        if calculated_hash != block.hash:
            return {
                'verified': False,
                'reason': 'Block hash mismatch'
            }

        return {
            'verified': True,
            'block_index': block.index,
            'block_hash': block.hash,
            'timestamp': block.timestamp,
            'data': block.data
        }

    def get_audit_trail(
        self,
        prediction_id: str = None,
        user_id: str = None
    ) -> List[Dict[str, Any]]:
        """
        Get audit trail for predictions.

        Args:
            prediction_id: Filter by prediction ID
            user_id: Filter by user ID

        Returns:
            List of audit records
        """
        criteria = {'type': 'prediction'}

        if prediction_id:
            criteria['prediction_id'] = prediction_id

        if user_id:
            criteria['user_id'] = user_id

        blocks = self.blockchain.search_blocks(criteria)

        return [
            {
                'block_index': block.index,
                'timestamp': block.timestamp,
                'hash': block.hash,
                'data': block.data
            }
            for block in blocks
        ]

    @staticmethod
    def _hash_data(data: Dict[str, Any]) -> str:
        """Hash data for privacy."""
        import hashlib
        import json
        data_string = json.dumps(data, sort_keys=True)
        return hashlib.sha256(data_string.encode()).hexdigest()


# Global service instance
blockchain_service = BlockchainAuditService()
```

```python
from fastapi import APIRouter, HTTPException
from typing import Optional

from app.blockchain.service import blockchain_service

router = APIRouter()


@router.get("/info")
async def get_blockchain_info():
    """Get blockchain statistics."""
    return blockchain_service.blockchain.get_chain_info()

async def verify_prediction(prediction_id: str):

        raise HTTPException(status_code=400, detail=result['reason'])

    return result


@router.get("/audit-trail")
async def get_audit_trail(
    prediction_id: Optional[str] = None,
    user_id: Optional[str] = None
):
    """
    Get blockchain audit trail.

    Returns:
        List of blockchain audit records

    trail = blockchain_service.get_audit_trail(
        prediction_id=prediction_id,
        user_id=user_id
    )

    return {
        'count': len(trail),
        'records': trail
    }


@router.get("/block/{index}")
async def get_block(index: int):
    """Get blockchain block by index."""
    block = blockchain_service.blockchain.get_block(index)

    if not block:
        raise HTTPException(status_code=404, detail="Block not found")

    return block.to_dict()

async def validate_blockchain():
```

### Step 2: Blockchain Storage

**`app/blockchain/storage.py`**:

```python
from sqlalchemy.orm import Session
from sqlalchemy import Column, Integer, String, DateTime, JSON
from sqlalchemy.ext.declarative import declarative_base
from datetime import datetime

from app.blockchain.hash_chain import Block
from app.db.session import SessionLocal

Base = declarative_base()


class BlockchainBlockDB(Base):
    """Database model for blockchain blocks."""
    __tablename__ = 'blockchain_blocks'

    id = Column(Integer, primary_key=True)
    block_index = Column(Integer, unique=True, nullable=False, index=True)
    timestamp = Column(DateTime(timezone=True), nullable=False)
    data = Column(JSON, nullable=False)
    previous_hash = Column(String(64), nullable=False)
    hash = Column(String(64), nullable=False, index=True)
    nonce = Column(Integer, nullable=False)
    created_at = Column(DateTime(timezone=True), default=datetime.now)


class BlockchainStorage:
    """Persist blockchain to database."""

    def save_block(self, block: Block):
        """Save block to database."""
        db = SessionLocal()

        try:
            db_block = BlockchainBlockDB(
                block_index=block.index,
                timestamp=datetime.fromisoformat(block.timestamp),
                data=block.data,
                previous_hash=block.previous_hash,
                hash=block.hash,
                nonce=block.nonce
            )

            db.add(db_block)
            db.commit()

        except Exception as e:
            db.rollback()
            raise
        finally:
            db.close()

    def load_chain(self) -> List[Block]:
        """Load blockchain from database."""
        db = SessionLocal()

        try:
            db_blocks = db.query(BlockchainBlockDB).order_by(
                BlockchainBlockDB.block_index
            ).all()

            blocks = [
                Block(
                    index=db_block.block_index,
                    timestamp=db_block.timestamp.isoformat(),
                    data=db_block.data,
                    previous_hash=db_block.previous_hash,
                    hash=db_block.hash,
                    nonce=db_block.nonce
                )
                for db_block in db_blocks
            ]

            return blocks

        finally:
            db.close()
```
