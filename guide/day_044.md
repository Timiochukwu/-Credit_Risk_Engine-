## Why Integration Patterns?

**Problem**: With multiple systems (REST API, WebSockets, Kafka, Blockchain):
- **Complexity**: Hard to coordinate between systems
- **Consistency**: Keeping data synchronized
- **Failure handling**: What if one system fails?
- **Scalability**: Each system scales differently
- **Debugging**: Hard to trace requests across systems

**Solution**: Integration patterns provide:
- **CQRS**: Separate read and write models
- **Event Sourcing**: Store all changes as events
- **Saga Pattern**: Distributed transactions
- **Orchestration**: Coordinate multiple systems
- **Compensation**: Rollback on failures

**`app/cqrs/commands.py`**:

```python
from dataclasses import dataclass
from typing import Dict, Any
import uuid


@dataclass
class PredictionCommand:
    """Command to create prediction."""
    command_id: str
    input_data: Dict[str, Any]
    user_id: str = None

    @classmethod
    def create(cls, input_data: Dict[str, Any], user_id: str = None):
        return cls(
            command_id=str(uuid.uuid4()),
            input_data=input_data,
            user_id=user_id
        )


class CommandHandler:
    """Handle commands (writes)."""

    async def handle_prediction_command(self, command: PredictionCommand):
        """
        Handle prediction command:
        1. Send to Kafka for processing
        2. Return command ID
        3. Consumer processes and updates read model
        """
        from app.kafka.async_producer import async_kafka_producer

        # Send to Kafka
        await async_kafka_producer.send_prediction_request(
            request_id=command.command_id,
            input_data=command.input_data,
            user_id=command.user_id
        )

        return {
            'command_id': command.command_id,
            'status': 'accepted'
        }
```

```python
from typing import List, Dict, Any, Optional
from sqlalchemy.orm import Session


class QueryHandler:
    """Handle queries (reads)."""

    def get_prediction_result(
        self,
        db: Session,
        command_id: str
    ) -> Optional[Dict[str, Any]]:
        """
        Get prediction result from read model.

        Read from optimized materialized view.
        """
        from app.db.models import PredictionView

        result = db.query(PredictionView).filter(
            PredictionView.command_id == command_id
        ).first()

        if result:
            return result.to_dict()
        return None

    def get_user_predictions(
        self,
        db: Session,
        user_id: str,
        limit: int = 100
    ) -> List[Dict[str, Any]]:
        """Get all predictions for a user (read model)."""
        from app.db.models import PredictionView

        results = db.query(PredictionView).filter(
            PredictionView.user_id == user_id
        ).order_by(
            PredictionView.timestamp.desc()
        ).limit(limit).all()

        return [r.to_dict() for r in results]
```

```python
from fastapi import APIRouter, Depends, BackgroundTasks
from sqlalchemy.orm import Session

from app.cqrs.commands import PredictionCommand, CommandHandler
from app.cqrs.queries import QueryHandler
from app.db.session import get_db

router = APIRouter()
command_handler = CommandHandler()
query_handler = QueryHandler()


@router.post("/commands/predict")
async def create_prediction_command(
    data: PredictionInput,
    background_tasks: BackgroundTasks
):
    """
    Create prediction command (WRITE).

    Returns immediately with command ID.
    Actual processing happens asynchronously.
    """
    command = PredictionCommand.create(
        input_data=data.dict(),
        user_id=None  # From auth
    )

    result = await command_handler.handle_prediction_command(command)

    return result


@router.get("/queries/prediction/{command_id}")
async def get_prediction_query(
    command_id: str,
    db: Session = Depends(get_db)
):
    """
    Query prediction result (READ).

    Reads from optimized read model.
    """
    result = query_handler.get_prediction_result(db, command_id)

    if not result:
        return {
            'command_id': command_id,
            'status': 'processing'
        }

    return result


@router.get("/queries/user/{user_id}/predictions")
async def get_user_predictions_query(
    user_id: str,
    limit: int = 100,
    db: Session = Depends(get_db)
):
    """Query all predictions for user (READ)."""
    predictions = query_handler.get_user_predictions(db, user_id, limit)

    return {
        'user_id': user_id,
        'count': len(predictions),
        'predictions': predictions
    }
```

### Event Store

**`app/events/event_store.py`**:

```python
from dataclasses import dataclass
from typing import List, Dict, Any
from datetime import datetime
import json


@dataclass
class Event:
    """Domain event."""
    event_id: str
    event_type: str
    aggregate_id: str
    data: Dict[str, Any]
    timestamp: datetime
    version: int


class EventStore:
    """Store all events (source of truth)."""

    def append_event(
        self,
        event_type: str,
        aggregate_id: str,
        data: Dict[str, Any]
    ) -> Event:
        """
        Append event to store.

        Args:
            event_type: Type of event (PredictionRequested, etc.)
            aggregate_id: ID of aggregate (prediction_id)
            data: Event data

        Returns:
            Stored event
        """
        from app.db.session import SessionLocal
        from app.db.models import EventLog

        db = SessionLocal()

        try:
            # Get next version
            latest = db.query(EventLog).filter(
                EventLog.aggregate_id == aggregate_id
            ).order_by(EventLog.version.desc()).first()

            version = (latest.version + 1) if latest else 1

            # Create event
            event = EventLog(
                event_id=str(uuid.uuid4()),
                event_type=event_type,
                aggregate_id=aggregate_id,
                data=data,
                timestamp=datetime.now(),
                version=version
            )

            db.add(event)
            db.commit()
            db.refresh(event)

            return Event(
                event_id=event.event_id,
                event_type=event.event_type,
                aggregate_id=event.aggregate_id,
                data=event.data,
                timestamp=event.timestamp,
                version=event.version
            )

        finally:
            db.close()

    def get_events(
        self,
        aggregate_id: str
    ) -> List[Event]:
        """Get all events for an aggregate."""
        from app.db.session import SessionLocal
        from app.db.models import EventLog

        db = SessionLocal()

        try:
            events = db.query(EventLog).filter(
                EventLog.aggregate_id == aggregate_id
            ).order_by(EventLog.version).all()

            return [
                Event(
                    event_id=e.event_id,
                    event_type=e.event_type,
                    aggregate_id=e.aggregate_id,
                    data=e.data,
                    timestamp=e.timestamp,
                    version=e.version
                )
                for e in events
            ]

        finally:
            db.close()


event_store = EventStore()
```

### Event Projection

**`app/events/projections.py`**:

```python
class PredictionProjection:
    """
    Project events to read model.

    Listens to events and updates materialized views.
    """

    def project_prediction_requested(self, event: Event):
        """Handle PredictionRequested event."""
        from app.db.session import SessionLocal
        from app.db.models import PredictionView

        db = SessionLocal()

        try:
            # Create read model entry
            view = PredictionView(
                command_id=event.aggregate_id,
                status='processing',
                created_at=event.timestamp
            )

            db.add(view)
            db.commit()

        finally:
            db.close()

    def project_prediction_completed(self, event: Event):
        """Handle PredictionCompleted event."""
        from app.db.session import SessionLocal
        from app.db.models import PredictionView

        db = SessionLocal()

        try:
            # Update read model
            view = db.query(PredictionView).filter(
                PredictionView.command_id == event.aggregate_id
            ).first()

            if view:
                view.status = 'completed'
                view.result = event.data['prediction']
                view.completed_at = event.timestamp
                db.commit()

        finally:
            db.close()
```

## Part 3: Saga Pattern

### Distributed Transaction Saga

**`app/sagas/prediction_saga.py`**:

```python
from enum import Enum
from typing import Dict, Any
import logging

logger = logging.getLogger(__name__)


class SagaStep(Enum):
    """Saga execution steps."""
    VALIDATE_INPUT = "validate_input"
    MAKE_PREDICTION = "make_prediction"
    RECORD_BLOCKCHAIN = "record_blockchain"
    SEND_NOTIFICATION = "send_notification"
    UPDATE_ANALYTICS = "update_analytics"


class PredictionSaga:
    """
    Saga for prediction workflow.

    Coordinates: Prediction -> Blockchain -> WebSocket -> Analytics

    If any step fails, compensate (rollback) previous steps.
    """

    def __init__(self, prediction_id: str, input_data: Dict[str, Any]):
        self.prediction_id = prediction_id
        self.input_data = input_data
        self.completed_steps = []
        self.state = {}

    async def execute(self):
        """Execute saga steps."""
        try:
            # Step 1: Validate input
            await self._validate_input()
            self.completed_steps.append(SagaStep.VALIDATE_INPUT)

            # Step 2: Make prediction
            prediction = await self._make_prediction()
            self.state['prediction'] = prediction
            self.completed_steps.append(SagaStep.MAKE_PREDICTION)

            # Step 3: Record on blockchain
            block = await self._record_blockchain(prediction)
            self.state['block'] = block
            self.completed_steps.append(SagaStep.RECORD_BLOCKCHAIN)

            # Step 4: Send WebSocket notification
            await self._send_notification(prediction)
            self.completed_steps.append(SagaStep.SEND_NOTIFICATION)

            # Step 5: Update analytics
            await self._update_analytics(prediction)
            self.completed_steps.append(SagaStep.UPDATE_ANALYTICS)

            logger.info(f"Saga completed successfully: {self.prediction_id}")
            return self.state

        except Exception as e:
            logger.error(f"Saga failed: {e}")
            # Compensate (rollback)
            await self._compensate()
            raise

    async def _validate_input(self):
        """Step 1: Validate input data."""
        # Validation logic
        if not self.input_data.get('loan_amount'):
            raise ValueError("Missing loan_amount")

    async def _make_prediction(self):
        """Step 2: Make ML prediction."""
        from app.ml.predictor import predictor
        return predictor.predict(self.input_data)

    async def _record_blockchain(self, prediction):
        """Step 3: Record on blockchain."""
        from app.blockchain.service import blockchain_service
        return blockchain_service.record_prediction(
            prediction_id=self.prediction_id,
            input_data=self.input_data,
            prediction=prediction
        )

    async def _send_notification(self, prediction):
        """Step 4: Send WebSocket notification."""
        from app.websockets.events import broadcast_prediction
        await broadcast_prediction(
            prediction=prediction,
            input_data=self.input_data
        )

    async def _update_analytics(self, prediction):
        """Step 5: Update analytics/metrics."""
        from app.kafka.producer import kafka_producer
        kafka_producer.produce_prediction_result(
            request_id=self.prediction_id,
            prediction=prediction,
            input_data=self.input_data
        )

    async def _compensate(self):
        """
        Compensate (rollback) completed steps.

        Executes compensation in reverse order.
        """
        logger.warning(f"Compensating saga: {self.prediction_id}")

        for step in reversed(self.completed_steps):
            try:
                if step == SagaStep.UPDATE_ANALYTICS:
                    await self._compensate_analytics()
                elif step == SagaStep.SEND_NOTIFICATION:
                    await self._compensate_notification()
                elif step == SagaStep.RECORD_BLOCKCHAIN:
                    await self._compensate_blockchain()
                elif step == SagaStep.MAKE_PREDICTION:
                    await self._compensate_prediction()

            except Exception as e:
                logger.error(f"Compensation failed for {step}: {e}")

    async def _compensate_analytics(self):
        """Compensate analytics update."""
        # Send compensation event to Kafka
        from app.kafka.producer import kafka_producer
        kafka_producer.produce_audit_event(
            'prediction_rollback',
            {'prediction_id': self.prediction_id}
        )

    async def _compensate_notification(self):
        """Compensate notification."""
        # Send rollback notification
        from app.websockets.events import broadcast_system_event
        await broadcast_system_event(
            'prediction_rollback',
            f'Prediction {self.prediction_id} rolled back',
            severity='warning'
        )

    async def _compensate_blockchain(self):
        """Compensate blockchain record."""
        # Note: Can't delete from blockchain!
        # Record compensation event instead
        from app.blockchain.service import blockchain_service
        blockchain_service.blockchain.add_block({
            'type': 'compensation',
            'original_prediction_id': self.prediction_id,
            'reason': 'saga_rollback'
        })

    async def _compensate_prediction(self):
        """Compensate prediction."""
        # Mark prediction as invalid
        logger.info(f"Prediction {self.prediction_id} marked as invalid")
```

```python
@router.post("/predict/saga")
async def predict_with_saga(data: PredictionInput):

    Make prediction using Saga pattern.

    prediction_id = str(uuid.uuid4())

    saga = PredictionSaga(

        input_data=data.dict()

        return {
            'prediction_id': prediction_id,
            'status': 'completed',
            'prediction': result['prediction'],
            'block_hash': result['block'].hash
        }

    except Exception as e:
        return {
            'prediction_id': prediction_id,
            'status': 'failed',
            'error': str(e),
            'compensated': True
        }
```

### Orchestration Service

```python
class SystemCoordinator:
    """
    Coordinate WebSockets, Kafka, and Blockchain.

    Central orchestration point for all advanced features.
    """

    async def process_prediction_full_stack(
        self,
        prediction_id: str,
        input_data: Dict[str, Any],
        user_id: str = None
    ):
        """
        Full-stack prediction processing:

        1. Send to Kafka queue
        2. Consumer processes prediction
        3. Record on blockchain
        4. Broadcast via WebSocket
        5. Update event store
        """
        from app.kafka.async_producer import async_kafka_producer
        from app.events.event_store import event_store

        # 1. Store event
        event_store.append_event(
            event_type='PredictionRequested',
            aggregate_id=prediction_id,
            data={'input_data': input_data, 'user_id': user_id}
        )

        # 2. Send to Kafka
        await async_kafka_producer.send_prediction_request(
            request_id=prediction_id,
            input_data=input_data,
            user_id=user_id
        )

        # Consumer will:
        # - Process prediction
        # - Record on blockchain
        # - Broadcast via WebSocket
        # - Publish result to Kafka

        return {
            'prediction_id': prediction_id,
            'status': 'processing',
            'message': 'Prediction queued for processing across all systems'
        }


coordinator = SystemCoordinator()
```
