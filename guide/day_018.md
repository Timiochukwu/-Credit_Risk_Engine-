## Goal
Build an automated model retraining system that monitors model performance and data drift, then automatically triggers retraining when performance degrades or significant drift is detected. Implement model versioning, rollback capabilities, and scheduled retraining jobs.

---

## Why Automated Retraining?

In production, ML models degrade over time due to:
- **Concept drift**: The relationship between features and target changes
- **Data drift**: Input data distribution shifts
- **Performance degradation**: Model accuracy decreases over time
- **New patterns**: Business environment evolves

Manual retraining is:
- Time-consuming and error-prone
- Difficult to scale
- May miss critical degradation periods

**Solution**: Automated retraining pipeline that:
- Monitors performance metrics continuously
- Detects drift automatically
- Triggers retraining based on rules
- Versions models for easy rollback
- Logs all retraining events

```python
"""
Model Version Manager
Handles model versioning, storage, and retrieval
"""
import os
import joblib
import json
import shutil
from datetime import datetime
from typing import Optional, List, Dict, Any
from pathlib import Path


class ModelVersionManager:
    """Manages model versions and metadata"""

    def __init__(self, registry_path: str = "models/registry"):
        self.registry_path = Path(registry_path)
        self.registry_path.mkdir(parents=True, exist_ok=True)
        self.metadata_file = self.registry_path / "versions.json"
        self._ensure_metadata_file()

    def _ensure_metadata_file(self):
        """Create metadata file if it doesn't exist"""
        if not self.metadata_file.exists():
            self._save_metadata({"versions": [], "current_version": None})

    def _load_metadata(self) -> Dict[str, Any]:
        """Load version metadata"""
        with open(self.metadata_file, 'r') as f:
            return json.load(f)

    def _save_metadata(self, metadata: Dict[str, Any]):
        """Save version metadata"""
        with open(self.metadata_file, 'w') as f:
            json.dump(metadata, f, indent=2, default=str)

    def save_model(
        self,
        model: Any,
        model_name: str,
        metrics: Dict[str, float],
        metadata: Optional[Dict[str, Any]] = None
    ) -> str:
        """
        Save a new model version

        Args:
            model: Trained model object
            model_name: Name of the model (e.g., 'xgboost', 'ensemble')
            metrics: Performance metrics
            metadata: Additional metadata

        Returns:
            Version ID
        """
        # Generate version ID
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        version_id = f"{model_name}_v{timestamp}"

        # Create version directory
        version_dir = self.registry_path / version_id
        version_dir.mkdir(exist_ok=True)

        # Save model file
        model_path = version_dir / "model.pkl"
        joblib.dump(model, model_path)

        # Prepare version metadata
        version_metadata = {
            "version_id": version_id,
            "model_name": model_name,
            "created_at": datetime.now().isoformat(),
            "metrics": metrics,
            "model_path": str(model_path),
            "is_active": False,
            "metadata": metadata or {}
        }

        # Update registry metadata
        registry_meta = self._load_metadata()
        registry_meta["versions"].append(version_metadata)

        # Cleanup old versions
        self._cleanup_old_versions(model_name, registry_meta)

        self._save_metadata(registry_meta)

        return version_id

    def _cleanup_old_versions(
        self,
        model_name: str,
        registry_meta: Dict[str, Any],
        max_versions: int = 10
    ):
        """Remove old versions beyond max_versions"""
        # Get versions for this model
        model_versions = [
            v for v in registry_meta["versions"]
            if v["model_name"] == model_name
        ]

        # Sort by creation date
        model_versions.sort(key=lambda x: x["created_at"], reverse=True)

        # Remove old versions
        if len(model_versions) > max_versions:
            versions_to_remove = model_versions[max_versions:]

            for version in versions_to_remove:
                # Delete directory
                version_dir = Path(version["model_path"]).parent
                if version_dir.exists():
                    shutil.rmtree(version_dir)

                # Remove from metadata
                registry_meta["versions"] = [
                    v for v in registry_meta["versions"]
                    if v["version_id"] != version["version_id"]
                ]

    def load_model(self, version_id: str) -> Any:
        """Load a specific model version"""
        metadata = self._load_metadata()

        version_info = next(
            (v for v in metadata["versions"] if v["version_id"] == version_id),
            None
        )

        if not version_info:
            raise ValueError(f"Version {version_id} not found")

        model_path = version_info["model_path"]
        return joblib.load(model_path)

    def set_active_version(self, version_id: str):
        """Set a version as the active production model"""
        metadata = self._load_metadata()

        # Deactivate all versions
        for version in metadata["versions"]:
            version["is_active"] = False

        # Activate specified version
        version_info = next(
            (v for v in metadata["versions"] if v["version_id"] == version_id),
            None
        )

        if not version_info:
            raise ValueError(f"Version {version_id} not found")

        version_info["is_active"] = True
        metadata["current_version"] = version_id

        self._save_metadata(metadata)

    def get_active_version(self) -> Optional[Dict[str, Any]]:
        """Get the currently active version"""
        metadata = self._load_metadata()
        current_version_id = metadata.get("current_version")

        if not current_version_id:
            return None

        return next(
            (v for v in metadata["versions"] if v["version_id"] == current_version_id),
            None
        )

    def list_versions(self, model_name: Optional[str] = None) -> List[Dict[str, Any]]:
        """List all versions, optionally filtered by model name"""
        metadata = self._load_metadata()
        versions = metadata["versions"]

        if model_name:
            versions = [v for v in versions if v["model_name"] == model_name]

        # Sort by creation date (newest first)
        versions.sort(key=lambda x: x["created_at"], reverse=True)

        return versions

    def rollback_to_version(self, version_id: str):
        """Rollback to a previous version"""
        self.set_active_version(version_id)
        return f"Rolled back to version {version_id}"
```

**Key Concepts:**

1. **Version ID Format**: `{model_name}_v{timestamp}` (e.g., `xgboost_v20240115_143052`)

2. **Metadata Storage**: JSON file tracking all versions with metrics and paths

3. **Active Version**: One version marked as production model at a time

4. **Automatic Cleanup**: Keeps only last N versions to save space

5. **Rollback**: Can switch back to any previous version instantly

```python
"""
Retraining Trigger Manager
Evaluates conditions and decides when to trigger retraining
"""
from typing import Dict, Any, List, Tuple
from datetime import datetime, timedelta
import pandas as pd
from sqlalchemy.orm import Session

from src.retraining.config import retraining_config
from src.monitoring.drift_detector import DriftDetector
from src.models.database_models import PredictionLog


class RetrainingTriggerManager:
    """Manages retraining trigger logic"""

    def __init__(self, db: Session):
        self.db = db
        self.config = retraining_config
        self.drift_detector = DriftDetector()

    def should_retrain(self) -> Tuple[bool, Dict[str, Any]]:
        """
        Check if retraining should be triggered

        Returns:
            Tuple of (should_retrain: bool, reasons: dict)
        """
        reasons = {
            "triggered": False,
            "performance_degradation": False,
            "data_drift": False,
            "sufficient_new_data": False,
            "scheduled": False,
            "details": {}
        }

        # Check 1: Performance degradation
        performance_check = self._check_performance_degradation()
        if performance_check["triggered"]:
            reasons["performance_degradation"] = True
            reasons["details"]["performance"] = performance_check
            reasons["triggered"] = True

        # Check 2: Data drift
        drift_check = self._check_data_drift()
        if drift_check["triggered"]:
            reasons["data_drift"] = True
            reasons["details"]["drift"] = drift_check
            reasons["triggered"] = True

        # Check 3: Sufficient new data
        data_check = self._check_sufficient_new_data()
        if data_check["triggered"]:
            reasons["sufficient_new_data"] = True
            reasons["details"]["new_data"] = data_check

        # Check 4: Scheduled retraining
        schedule_check = self._check_scheduled_retraining()
        if schedule_check["triggered"]:
            reasons["scheduled"] = True
            reasons["details"]["schedule"] = schedule_check
            reasons["triggered"] = True

        return reasons["triggered"], reasons

    def _check_performance_degradation(self) -> Dict[str, Any]:
        """Check if model performance has degraded"""
        result = {"triggered": False, "metrics": {}}

        try:
            # Get recent predictions with ground truth
            lookback_date = datetime.now() - timedelta(
                days=self.config.RETRAINING_WINDOW_DAYS
            )

            recent_predictions = self.db.query(PredictionLog).filter(
                PredictionLog.created_at >= lookback_date,
                PredictionLog.actual_outcome.isnot(None)  # Only labeled data
            ).all()

            if len(recent_predictions) < 100:  # Need minimum samples
                return result

            # Calculate current performance
            y_true = [p.actual_outcome for p in recent_predictions]
            y_pred = [1 if p.prediction >= 0.5 else 0 for p in recent_predictions]

            from sklearn.metrics import accuracy_score, roc_auc_score, f1_score

            accuracy = accuracy_score(y_true, y_pred)

            # Get prediction probabilities for ROC-AUC
            y_scores = [p.prediction for p in recent_predictions]
            roc_auc = roc_auc_score(y_true, y_scores)

            f1 = f1_score(y_true, y_pred)

            result["metrics"] = {
                "accuracy": accuracy,
                "roc_auc": roc_auc,
                "f1_score": f1,
                "sample_count": len(recent_predictions)
            }

            # Check thresholds
            if (accuracy < self.config.MIN_ACCURACY or
                roc_auc < self.config.MIN_ROC_AUC or
                f1 < self.config.MIN_F1_SCORE):
                result["triggered"] = True
                result["reason"] = "Performance below thresholds"

        except Exception as e:
            result["error"] = str(e)

        return result

    def _check_data_drift(self) -> Dict[str, Any]:
        """Check if significant data drift detected"""
        result = {"triggered": False, "drift_features": []}

        try:
            # Load reference data (training data)
            reference_df = pd.read_csv("data/processed/X_train.csv")

            # Get recent production data
            lookback_date = datetime.now() - timedelta(
                days=self.config.RETRAINING_WINDOW_DAYS
            )

            recent_predictions = self.db.query(PredictionLog).filter(
                PredictionLog.created_at >= lookback_date
            ).limit(5000).all()  # Sample recent predictions

            if len(recent_predictions) < 500:
                return result

            # Convert to DataFrame
            production_data = []
            for pred in recent_predictions:
                production_data.append(pred.features)

            production_df = pd.DataFrame(production_data)

            # Check drift for numerical features
            numerical_features = reference_df.select_dtypes(
                include=['int64', 'float64']
            ).columns

            drift_count = 0
            for feature in numerical_features:
                if feature not in production_df.columns:
                    continue

                ks_stat, p_value = self.drift_detector.kolmogorov_smirnov_test(
                    reference_df[feature].values,
                    production_df[feature].values
                )

                if ks_stat > self.config.MAX_KS_STATISTIC:
                    drift_count += 1
                    result["drift_features"].append({
                        "feature": feature,
                        "ks_statistic": ks_stat,
                        "p_value": p_value
                    })

            if drift_count >= self.config.MAX_DRIFT_FEATURES:
                result["triggered"] = True
                result["reason"] = f"{drift_count} features with significant drift"

        except Exception as e:
            result["error"] = str(e)

        return result

    def _check_sufficient_new_data(self) -> Dict[str, Any]:
        """Check if enough new labeled data available"""
        result = {"triggered": False, "new_sample_count": 0}

        try:
            lookback_date = datetime.now() - timedelta(
                days=self.config.RETRAINING_WINDOW_DAYS
            )

            new_samples = self.db.query(PredictionLog).filter(
                PredictionLog.created_at >= lookback_date,
                PredictionLog.actual_outcome.isnot(None)
            ).count()

            result["new_sample_count"] = new_samples

            if new_samples >= self.config.MIN_NEW_SAMPLES:
                result["triggered"] = True
                result["reason"] = f"{new_samples} new labeled samples available"

        except Exception as e:
            result["error"] = str(e)

        return result

    def _check_scheduled_retraining(self) -> Dict[str, Any]:
        """Check if scheduled retraining is due"""
        result = {"triggered": False}

        if not self.config.SCHEDULE_ENABLED:
            return result

        # This would integrate with APScheduler
        # For now, simple logic: check last retraining date
        # In production, APScheduler handles this automatically

        result["triggered"] = False  # APScheduler handles timing
        result["reason"] = "Managed by scheduler"

        return result
```

**Key Concepts:**

1. **Multi-Condition Triggers**:
   - Performance degradation (accuracy/ROC-AUC drop)
   - Data drift (KS test on features)
   - Sufficient new data (labeled samples)
   - Scheduled time (cron-based)

2. **Lookback Window**: Analyzes last N days of production data

3. **Ground Truth**: Needs `actual_outcome` field in `PredictionLog` (feedback loop)

4. **Configurable Thresholds**: All limits defined in config

## File 4: Retraining Pipeline

**File:** `src/retraining/retrain_pipeline.py`

```python
"""
Automated Retraining Pipeline
Executes full model retraining workflow
"""
import os
import pandas as pd
from datetime import datetime
from typing import Dict, Any, Optional
from sqlalchemy.orm import Session

from src.retraining.model_versioner import ModelVersionManager
from src.retraining.trigger_manager import RetrainingTriggerManager
from src.retraining.config import retraining_config
from src.data.preprocessor import DataPreprocessor
from src.models.train_ensemble import train_all_models
from src.utils.logger import get_logger


logger = get_logger(__name__)


class RetrainingPipeline:
    """Automated model retraining pipeline"""

    def __init__(self, db: Session):
        self.db = db
        self.version_manager = ModelVersionManager()
        self.trigger_manager = RetrainingTriggerManager(db)
        self.config = retraining_config

    def check_and_retrain(self, force: bool = False) -> Dict[str, Any]:
        """
        Check if retraining needed and execute if so

        Args:
            force: Force retraining regardless of triggers

        Returns:
            Dict with retraining results
        """
        result = {
            "retrained": False,
            "timestamp": datetime.now().isoformat(),
            "triggers": {},
            "new_version": None,
            "metrics": {}
        }

        # Check triggers
        if not force:
            should_retrain, reasons = self.trigger_manager.should_retrain()
            result["triggers"] = reasons

            if not should_retrain:
                logger.info("No retraining triggers activated")
                return result

            logger.info(f"Retraining triggered: {reasons}")
        else:
            logger.info("Forced retraining initiated")
            result["triggers"]["forced"] = True

        # Execute retraining
        try:
            retraining_result = self._execute_retraining()
            result.update(retraining_result)
            result["retrained"] = True

            logger.info(f"Retraining completed successfully: {result['new_version']}")

        except Exception as e:
            logger.error(f"Retraining failed: {str(e)}")
            result["error"] = str(e)

        return result

    def _execute_retraining(self) -> Dict[str, Any]:
        """Execute the retraining workflow"""
        logger.info("Starting retraining pipeline...")

        # Step 1: Prepare fresh training data
        logger.info("Step 1: Preparing training data...")
        train_data = self._prepare_training_data()

        # Step 2: Preprocess data
        logger.info("Step 2: Preprocessing data...")
        X_train, X_test, y_train, y_test = self._preprocess_data(train_data)

        # Step 3: Train models
        logger.info("Step 3: Training models...")
        models_results = train_all_models(X_train, X_test, y_train, y_test)

        # Step 4: Select best model
        logger.info("Step 4: Selecting best model...")
        best_model_info = self._select_best_model(models_results)

        # Step 5: Version and save model
        logger.info("Step 5: Versioning model...")
        version_id = self.version_manager.save_model(
            model=best_model_info["model"],
            model_name=best_model_info["name"],
            metrics=best_model_info["metrics"],
            metadata={
                "training_samples": len(X_train),
                "test_samples": len(X_test),
                "retraining_timestamp": datetime.now().isoformat(),
                "all_models": {
                    name: {"metrics": info["metrics"]}
                    for name, info in models_results.items()
                }
            }
        )

        # Step 6: Validate new model
        logger.info("Step 6: Validating new model...")
        validation_passed = self._validate_new_model(best_model_info)

        if validation_passed:
            # Set as active version
            self.version_manager.set_active_version(version_id)
            logger.info(f"New model activated: {version_id}")
        else:
            logger.warning(f"New model failed validation: {version_id}")

        return {
            "new_version": version_id,
            "model_name": best_model_info["name"],
            "metrics": best_model_info["metrics"],
            "validation_passed": validation_passed,
            "training_samples": len(X_train),
            "test_samples": len(X_test)
        }

    def _prepare_training_data(self) -> pd.DataFrame:
        """Prepare training data from database and CSV"""
        # Load original training data
        original_df = pd.read_csv("data/raw/loan_applications.csv")

        # Get new labeled predictions from production
        from src.models.database_models import PredictionLog

        new_predictions = self.db.query(PredictionLog).filter(
            PredictionLog.actual_outcome.isnot(None)
        ).all()

        if new_predictions:
            # Convert to DataFrame
            new_data = []
            for pred in new_predictions:
                row = pred.features.copy()
                row["default"] = pred.actual_outcome
                new_data.append(row)

            new_df = pd.DataFrame(new_data)

            # Combine with original data
            combined_df = pd.concat([original_df, new_df], ignore_index=True)
            logger.info(f"Added {len(new_df)} new samples to training data")
        else:
            combined_df = original_df
            logger.info("No new production data, using original dataset")

        return combined_df

    def _preprocess_data(self, df: pd.DataFrame) -> tuple:
        """Preprocess data for training"""
        preprocessor = DataPreprocessor()

        # Save for preprocessing
        temp_file = "data/temp/retrain_data.csv"
        os.makedirs("data/temp", exist_ok=True)
        df.to_csv(temp_file, index=False)

        # Preprocess
        X_train, X_test, y_train, y_test = preprocessor.preprocess_data(temp_file)

        return X_train, X_test, y_train, y_test

    def _select_best_model(self, models_results: Dict[str, Any]) -> Dict[str, Any]:
        """Select best model based on ROC-AUC"""
        best_model = None
        best_roc_auc = 0
        best_name = None

        for name, info in models_results.items():
            roc_auc = info["metrics"]["roc_auc"]
            if roc_auc > best_roc_auc:
                best_roc_auc = roc_auc
                best_model = info["model"]
                best_name = name

        return {
            "name": best_name,
            "model": best_model,
            "metrics": models_results[best_name]["metrics"]
        }

    def _validate_new_model(self, model_info: Dict[str, Any]) -> bool:
        """Validate new model meets minimum standards"""
        metrics = model_info["metrics"]

        # Check minimum thresholds
        if metrics["roc_auc"] < 0.75:  # Minimum acceptable ROC-AUC
            logger.warning(f"ROC-AUC too low: {metrics['roc_auc']}")
            return False

        if metrics["accuracy"] < 0.70:  # Minimum acceptable accuracy
            logger.warning(f"Accuracy too low: {metrics['accuracy']}")
            return False

        # Compare with current production model
        active_version = self.version_manager.get_active_version()
        if active_version:
            current_roc_auc = active_version["metrics"].get("roc_auc", 0)

            # New model should be at least as good (with small margin)
            if metrics["roc_auc"] < current_roc_auc - 0.02:
                logger.warning(
                    f"New model worse than current: "
                    f"{metrics['roc_auc']} < {current_roc_auc}"
                )
                return False

        return True
```

**Key Concepts:**

1. **Data Combination**: Merges original training data with new production feedback

2. **Full Pipeline**: Data prep → preprocess → train → validate → version

3. **Best Model Selection**: Chooses highest ROC-AUC from all trained models

4. **Validation Gate**: New model must meet minimum standards and not regress

5. **Automatic Activation**: Only activates if validation passes

```python
"""
Retraining API Routes
Endpoints for triggering and managing retraining
"""
from fastapi import APIRouter, Depends, HTTPException, BackgroundTasks
from sqlalchemy.orm import Session
from typing import Dict, Any, List

from src.utils.database import get_db
from src.utils.security import get_current_user
from src.retraining.retrain_pipeline import RetrainingPipeline
from src.retraining.model_versioner import ModelVersionManager
from src.retraining.trigger_manager import RetrainingTriggerManager
from src.models.database_models import User


router = APIRouter(prefix="/api/v1/retraining", tags=["retraining"])


@router.post("/trigger", response_model=Dict[str, Any])
async def trigger_retraining(
    force: bool = False,
    background_tasks: BackgroundTasks = None,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user)
):
    """
    Trigger model retraining

    Args:
        force: Force retraining regardless of triggers
        background_tasks: FastAPI background tasks
    """
    pipeline = RetrainingPipeline(db)

    # Run in background if background_tasks provided
    if background_tasks:
        background_tasks.add_task(pipeline.check_and_retrain, force=force)
        return {
            "status": "queued",
            "message": "Retraining job queued in background"
        }

    # Run synchronously
    result = pipeline.check_and_retrain(force=force)
    return result


@router.get("/check-triggers", response_model=Dict[str, Any])
async def check_retraining_triggers(
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user)
):
    """Check if retraining triggers are activated"""
    trigger_manager = RetrainingTriggerManager(db)
    should_retrain, reasons = trigger_manager.should_retrain()

    return {
        "should_retrain": should_retrain,
        "reasons": reasons
    }


@router.get("/versions", response_model=List[Dict[str, Any]])
async def list_model_versions(
    model_name: str = None,
    current_user: User = Depends(get_current_user)
):
    """List all model versions"""
    version_manager = ModelVersionManager()
```

## File 8: Create Scheduled Retraining Job

**File:** `src/retraining/scheduler.py`

```python
"""
Scheduled Retraining Job
Uses APScheduler for periodic retraining checks
"""
from apscheduler.schedulers.background import BackgroundScheduler
from apscheduler.triggers.cron import CronTrigger
from datetime import datetime

from src.retraining.retrain_pipeline import RetrainingPipeline
from src.retraining.config import retraining_config
from src.utils.database import SessionLocal
from src.utils.logger import get_logger


logger = get_logger(__name__)


class RetrainingScheduler:
    """Background scheduler for automated retraining"""

    def __init__(self):
        self.scheduler = BackgroundScheduler()
        self.config = retraining_config

    def start(self):
        """Start the scheduler"""
        if not self.config.SCHEDULE_ENABLED:
            logger.info("Scheduled retraining disabled")
            return

        # Add retraining job
        self.scheduler.add_job(
            func=self._run_retraining_check,
            trigger=CronTrigger.from_crontab(self.config.SCHEDULE_CRON),
            id="retraining_check",
            name="Periodic Retraining Check",
            replace_existing=True
        )

        self.scheduler.start()
        logger.info(f"Retraining scheduler started: {self.config.SCHEDULE_CRON}")

    def stop(self):
        """Stop the scheduler"""
        self.scheduler.shutdown()
        logger.info("Retraining scheduler stopped")

    def _run_retraining_check(self):
        """Execute retraining check"""
        logger.info("Running scheduled retraining check...")

        db = SessionLocal()
        try:
            pipeline = RetrainingPipeline(db)
            result = pipeline.check_and_retrain(force=False)

            if result["retrained"]:
                logger.info(f"Scheduled retraining completed: {result}")
            else:
                logger.info("No retraining needed")

        except Exception as e:
            logger.error(f"Scheduled retraining failed: {str(e)}")

        finally:
            db.close()


# Global scheduler instance
retraining_scheduler = RetrainingScheduler()
```
