Today you'll add event streaming capabilities using Apache Kafka. You'll implement asynchronous prediction processing, event-driven architecture, stream processing, and integrate Kafka with your existing REST API without breaking functionality.

## Why Kafka?

**Problem**: Synchronous processing limitations:
- **Blocking requests**: API waits for model inference
- **No event history**: Can't replay past events
- **Tight coupling**: Services directly depend on each other
- **Limited scalability**: Can't independently scale producers/consumers
- **No fault tolerance**: Lost requests if service crashes

**Solution**: Kafka provides:
- **Asynchronous processing**: Decouple request/response
- **Event sourcing**: Complete history of all events
- **Scalability**: Independently scale producers and consumers
- **Fault tolerance**: Guaranteed message delivery
- **Stream processing**: Real-time analytics and transformations

**Benefits**:
- **Higher throughput**: 100,000+ events/second
- **Decoupled architecture**: Microservices independence
- **Event replay**: Reprocess historical data
- **Multiple consumers**: Same event, different use cases
- **Exactly-once semantics**: No duplicate processing

### Step 1: Create Kafka Producer

**`app/kafka/producer.py`**:

```python
from confluent_kafka import Producer
from typing import Dict, Any, Optional
import json
import logging
from datetime import datetime

from app.core.config import settings

logger = logging.getLogger(__name__)


class KafkaProducer:
    """Kafka producer for sending prediction events."""

    def __init__(self):
        self.config = {
            'bootstrap.servers': settings.KAFKA_BOOTSTRAP_SERVERS,
            'client.id': 'credit-risk-producer',
            'acks': 'all',  # Wait for all replicas
            'retries': 3,
            'max.in.flight.requests.per.connection': 5,
            'enable.idempotence': True,  # Exactly-once semantics
            'compression.type': 'snappy'
        }
        self.producer = Producer(self.config)

    def delivery_report(self, err, msg):
        """Callback for message delivery confirmation."""
        if err is not None:
            logger.error(f'Message delivery failed: {err}')
        else:
            logger.debug(
                f'Message delivered to {msg.topic()} '
                f'[{msg.partition()}] at offset {msg.offset()}'
            )

    def produce_prediction_request(
        self,
        request_id: str,
        input_data: Dict[str, Any],
        user_id: Optional[str] = None
    ):
        """
        Send prediction request to Kafka.

        Args:
            request_id: Unique request ID
            input_data: Prediction input features
            user_id: User making the request
        """
        message = {
            'request_id': request_id,
            'timestamp': datetime.now().isoformat(),
            'user_id': user_id,
            'input_data': input_data,
            'event_type': 'prediction_request'
        }

        try:
            self.producer.produce(
                topic='prediction-requests',
                key=request_id.encode('utf-8'),
                value=json.dumps(message).encode('utf-8'),
                callback=self.delivery_report
            )
            self.producer.poll(0)  # Trigger callbacks

        except Exception as e:
            logger.error(f"Failed to produce message: {e}")
            raise

    def produce_prediction_result(
        self,
        request_id: str,
        prediction: Dict[str, Any],
        input_data: Dict[str, Any],
        explanation: Optional[Dict[str, Any]] = None
    ):
        """
        Send prediction result to Kafka.

        Args:
            request_id: Unique request ID
            prediction: Prediction result
            input_data: Input features used
            explanation: SHAP explanation
        """
        message = {
            'request_id': request_id,
            'timestamp': datetime.now().isoformat(),
            'prediction': prediction,
            'input_data': input_data,
            'explanation': explanation,
            'event_type': 'prediction_result'
        }

        try:
            # Send to main results topic
            self.producer.produce(
                topic='prediction-results',
                key=request_id.encode('utf-8'),
                value=json.dumps(message).encode('utf-8'),
                callback=self.delivery_report
            )

            # Also send to high-risk topic if applicable
            if prediction.get('default_probability', 0) > 0.7:
                self.producer.produce(
                    topic='high-risk-predictions',
                    key=request_id.encode('utf-8'),
                    value=json.dumps(message).encode('utf-8'),
                    callback=self.delivery_report
                )

            self.producer.poll(0)

        except Exception as e:
            logger.error(f"Failed to produce result: {e}")
            raise

    def produce_audit_event(
        self,
        event_type: str,
        event_data: Dict[str, Any]
    ):
        """
        Send audit event to Kafka.

        Args:
            event_type: Type of audit event
            event_data: Event data
        """
        message = {
            'event_type': event_type,
            'timestamp': datetime.now().isoformat(),
            'data': event_data
        }

        try:
            self.producer.produce(
                topic='audit-events',
                value=json.dumps(message).encode('utf-8'),
                callback=self.delivery_report
            )
            self.producer.poll(0)

        except Exception as e:
            logger.error(f"Failed to produce audit event: {e}")

    def flush(self):
        """Wait for all messages to be delivered."""
        self.producer.flush()

    def close(self):
        """Close producer and flush remaining messages."""
        self.producer.flush(timeout=10)


# Global producer instance
kafka_producer = KafkaProducer()
```

### Step 2: Async Kafka Producer

**`app/kafka/async_producer.py`**:

from aiokafka import AIOKafkaProducer
from typing import Dict, Any, Optional
import json
import logging

from app.core.config import settings

logger = logging.getLogger(__name__)


class AsyncKafkaProducer:
    """Async Kafka producer for non-blocking operations."""

    def __init__(self):
        self.producer: Optional[AIOKafkaProducer] = None

    async def start(self):
        """Start the producer."""
        self.producer = AIOKafkaProducer(
            bootstrap_servers=settings.KAFKA_BOOTSTRAP_SERVERS,
            value_serializer=lambda v: json.dumps(v).encode('utf-8'),
            key_serializer=lambda k: k.encode('utf-8') if k else None,
            acks='all',
            enable_idempotence=True,
            compression_type='snappy'
        )
        await self.producer.start()

    async def stop(self):
        """Stop the producer."""
        if self.producer:
            await self.producer.stop()

    async def send_prediction_request(
        self,
        request_id: str,
        input_data: Dict[str, Any],
        user_id: Optional[str] = None
    ):
        """Send prediction request asynchronously."""
        if not self.producer:
            await self.start()

        message = {
            'request_id': request_id,
            'timestamp': datetime.now().isoformat(),
            'user_id': user_id,
            'input_data': input_data
        }

        await self.producer.send_and_wait(
            'prediction-requests',
            value=message,
            key=request_id
        )


# Global async producer
async_kafka_producer = AsyncKafkaProducer()

from confluent_kafka import Consumer, KafkaError
from typing import Callable, Dict, Any
import json
import logging
from datetime import datetime

from app.core.config import settings

logger = logging.getLogger(__name__)


class KafkaConsumer:
    """Kafka consumer for processing prediction requests."""

    def __init__(self, group_id: str):
        self.config = {
            'bootstrap.servers': settings.KAFKA_BOOTSTRAP_SERVERS,
            'group.id': group_id,
            'auto.offset.reset': 'earliest',
            'enable.auto.commit': False,  # Manual commit for exactly-once
            'max.poll.interval.ms': 300000,  # 5 minutes
            'session.timeout.ms': 45000
        }
        self.consumer = Consumer(self.config)

    def subscribe(self, topics: list):
        """Subscribe to topics."""
        self.consumer.subscribe(topics)
        logger.info(f"Subscribed to topics: {topics}")

    def consume(
        self,
        message_handler: Callable[[Dict[str, Any]], None],
        max_messages: int = None
    ):
        """
        Consume messages and process them.

        Args:
            message_handler: Function to handle each message
            max_messages: Maximum messages to process (None = infinite)
        """
        messages_processed = 0

        try:
            while max_messages is None or messages_processed < max_messages:
                msg = self.consumer.poll(timeout=1.0)

                if msg is None:
                    continue

                if msg.error():
                    if msg.error().code() == KafkaError._PARTITION_EOF:
                        logger.info(f'End of partition {msg.partition()}')
                    else:
                        logger.error(f'Consumer error: {msg.error()}')
                    continue

                # Parse message
                try:
                    message_data = json.loads(msg.value().decode('utf-8'))

                    # Process message
                    message_handler(message_data)

                    # Commit offset (exactly-once)
                    self.consumer.commit(msg)

                    messages_processed += 1

                except Exception as e:
                    logger.error(f"Error processing message: {e}")
                    # Don't commit - will retry

        except KeyboardInterrupt:
            logger.info("Consumer interrupted")
        finally:
            self.consumer.close()


# Prediction request processor
def process_prediction_request(message: Dict[str, Any]):
    """
    Process prediction request from Kafka.

    Args:
        message: Prediction request message
    """
    from app.ml.predictor import predictor
    from app.explainability.shap_explainer import explainer
    from app.kafka.producer import kafka_producer
    from app.cache.redis_client import cache

    request_id = message['request_id']
    input_data = message['input_data']

    logger.info(f"Processing prediction request: {request_id}")

    try:
        # Make prediction
        result = predictor.predict(input_data)

        # Get explanation
        explanation = explainer.explain_prediction(
            pd.DataFrame([input_data])
        )

        # Send result to Kafka
        kafka_producer.produce_prediction_result(
            request_id=request_id,
            prediction=result,
            input_data=input_data,
            explanation=explanation
        )

        # Cache result
        cache.set(
            f"prediction_result:{request_id}",
            result,
            ttl=3600  # 1 hour
        )

        logger.info(f"Completed prediction: {request_id}")

    except Exception as e:
        logger.error(f"Prediction failed for {request_id}: {e}")
        # Send error event
        kafka_producer.produce_audit_event(
            'prediction_error',
            {
                'request_id': request_id,
                'error': str(e)
            }
        )
```

from confluent_kafka import Consumer, Producer
import json
from collections import defaultdict
from datetime import datetime, timedelta


class PredictionStatsProcessor:
    """
    Stream processor for real-time prediction statistics.

    Computes running statistics from prediction stream.
    """

    def __init__(self):
        self.stats = {
            'total_predictions': 0,
            'high_risk_count': 0,
            'avg_probability': 0.0,
            'predictions_per_minute': defaultdict(int)
        }

        self.consumer = Consumer({
            'bootstrap.servers': 'kafka:9092',
            'group.id': 'stats-processor',
            'auto.offset.reset': 'latest'
        })

        self.producer = Producer({
            'bootstrap.servers': 'kafka:9092'
        })

    def process(self):
        """Process prediction results stream."""
        self.consumer.subscribe(['prediction-results'])

        while True:
            msg = self.consumer.poll(1.0)

            if msg is None:
                continue

            if msg.error():
                continue

            # Parse message
            data = json.loads(msg.value().decode('utf-8'))
            prediction = data['prediction']

            # Update stats
            self.stats['total_predictions'] += 1

            prob = prediction['default_probability']
            if prob > 0.7:
                self.stats['high_risk_count'] += 1

            # Update running average
            total = self.stats['total_predictions']
            current_avg = self.stats['avg_probability']
            self.stats['avg_probability'] = (
                (current_avg * (total - 1) + prob) / total
            )

            # Track per-minute rate
            minute_key = datetime.now().strftime('%Y-%m-%d %H:%M')
            self.stats['predictions_per_minute'][minute_key] += 1

            # Publish updated stats
            self.producer.produce(
                'prediction-stats',
                value=json.dumps(self.stats).encode('utf-8')
            )
            self.producer.poll(0)

            # Commit
            self.consumer.commit(msg)
```

### 1. Use Exactly-Once Semantics

### 2. Handle Poison Pills

```python
MAX_RETRIES = 3
RETRY_DELAY = 5  # seconds

def process_with_retry(message):
    retries = 0
    while retries < MAX_RETRIES:
        try:
            process_prediction_request(message)
            return
        except Exception as e:
            retries += 1
            if retries >= MAX_RETRIES:
                # Send to dead letter queue
                kafka_producer.produce(
                    'prediction-errors-dlq',
                    value=json.dumps(message)
                )
            time.sleep(RETRY_DELAY)
```
