## Understanding Ensemble Learning

**What is ensemble learning?**
Combining multiple models to make better predictions than any single model alone.

**The wisdom of crowds:**
- Model A: 85% accurate
- Model B: 83% accurate
- Model C: 84% accurate
- Ensemble (A+B+C): 87% accurate!

**Types of ensembles:**

1. **Bagging (Bootstrap Aggregating):**
   - Train same algorithm on different data subsets
   - Example: Random Forest (many decision trees)
   - Reduces variance

2. **Boosting:**
   - Sequential training, each model fixes previous errors
   - Example: XGBoost, LightGBM
   - Reduces bias

3. **Stacking:**
   - Train meta-model on predictions of base models
   - Learn how to combine models optimally

4. **Voting:**
   - Average or vote on predictions
   - Simple but effective

```python
"""
Train ensemble models: LightGBM, Random Forest, and combinations.
"""

import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier, VotingClassifier, StackingClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import (
    accuracy_score,
    precision_score,
    recall_score,
    f1_score,
    roc_auc_score
)
import xgboost as xgb
import lightgbm as lgb
from pathlib import Path
import pickle
import mlflow

from src.utils.logger import logger


mlflow.set_tracking_uri("file:./mlruns")


def load_data():
    """Load training and test data."""
    logger.info("Loading processed data...")

    X_train = pd.read_csv("data/processed/X_train.csv")
    X_test = pd.read_csv("data/processed/X_test.csv")
    y_train = pd.read_csv("data/processed/y_train.csv")['defaulted']
    y_test = pd.read_csv("data/processed/y_test.csv")['defaulted']

    return X_train, X_test, y_train, y_test


def train_lightgbm(X_train, X_test, y_train, y_test):
    """
    Train LightGBM model.

    LightGBM is Microsoft's gradient boosting framework.
    Advantages:
    - Faster training than XGBoost
    - Lower memory usage
    - Better accuracy on large datasets
    """
    logger.info("Training LightGBM...")

    with mlflow.start_run(run_name="lightgbm"):

        params = {
            'objective': 'binary',
            'metric': 'auc',
            'boosting_type': 'gbdt',
            'num_leaves': 31,
            'learning_rate': 0.1,
            'n_estimators': 200,
            'max_depth': 6,
            'subsample': 0.8,
            'colsample_bytree': 0.8,
            'random_state': 42,
            'verbose': -1
        }

        mlflow.log_params(params)

        model = lgb.LGBMClassifier(**params)

        model.fit(
            X_train, y_train,
            eval_set=[(X_test, y_test)],
            callbacks=[lgb.early_stopping(20), lgb.log_evaluation(0)]
        )

        # Evaluate
        y_pred = model.predict(X_test)
        y_pred_proba = model.predict_proba(X_test)[:, 1]

        metrics = {
            'accuracy': accuracy_score(y_test, y_pred),
            'precision': precision_score(y_test, y_pred),
            'recall': recall_score(y_test, y_pred),
            'f1_score': f1_score(y_test, y_pred),
            'roc_auc': roc_auc_score(y_test, y_pred_proba)
        }

        mlflow.log_metrics(metrics)
        mlflow.lightgbm.log_model(model, "model")

        logger.info(f"LightGBM - Accuracy: {metrics['accuracy']:.4f}, ROC-AUC: {metrics['roc_auc']:.4f}")

        return model, metrics


def train_random_forest(X_train, X_test, y_train, y_test):
    """
    Train Random Forest model.

    Random Forest creates many decision trees and averages their predictions.
    Advantages:
    - Robust to overfitting
    - Handles non-linear relationships
    - Works well out-of-the-box
    """
    logger.info("Training Random Forest...")

    with mlflow.start_run(run_name="random_forest"):

        params = {
            'n_estimators': 200,
            'max_depth': 12,
            'min_samples_split': 10,
            'min_samples_leaf': 4,
            'max_features': 'sqrt',
            'random_state': 42,
            'n_jobs': -1,
            'class_weight': 'balanced'
        }

        mlflow.log_params(params)

        model = RandomForestClassifier(**params)
        model.fit(X_train, y_train)

        # Evaluate
        y_pred = model.predict(X_test)
        y_pred_proba = model.predict_proba(X_test)[:, 1]

        metrics = {
            'accuracy': accuracy_score(y_test, y_pred),
            'precision': precision_score(y_test, y_pred),
            'recall': recall_score(y_test, y_pred),
            'f1_score': f1_score(y_test, y_pred),
            'roc_auc': roc_auc_score(y_test, y_pred_proba)
        }

        mlflow.log_metrics(metrics)
        mlflow.sklearn.log_model(model, "model")

        logger.info(f"Random Forest - Accuracy: {metrics['accuracy']:.4f}, ROC-AUC: {metrics['roc_auc']:.4f}")

        return model, metrics


def train_voting_ensemble(X_train, X_test, y_train, y_test):
    """
    Train Voting Classifier ensemble.

    Combines predictions from multiple models by averaging probabilities.
    """
    logger.info("Training Voting Ensemble...")

    with mlflow.start_run(run_name="voting_ensemble"):

        # Define base models
        xgb_model = xgb.XGBClassifier(
            max_depth=6,
            learning_rate=0.1,
            n_estimators=200,
            random_state=42,
            eval_metric='logloss'
        )

        lgb_model = lgb.LGBMClassifier(
            num_leaves=31,
            learning_rate=0.1,
            n_estimators=200,
            random_state=42,
            verbose=-1
        )

        rf_model = RandomForestClassifier(
            n_estimators=200,
            max_depth=12,
            random_state=42,
            n_jobs=-1
        )

        # Create voting ensemble
        voting_clf = VotingClassifier(
            estimators=[
                ('xgboost', xgb_model),
                ('lightgbm', lgb_model),
                ('random_forest', rf_model)
            ],
            voting='soft',  # Use probability averages
            n_jobs=-1
        )

        mlflow.log_param("ensemble_type", "voting")
        mlflow.log_param("voting_method", "soft")
        mlflow.log_param("n_estimators", 3)

        # Train
        voting_clf.fit(X_train, y_train)

        # Evaluate
        y_pred = voting_clf.predict(X_test)
        y_pred_proba = voting_clf.predict_proba(X_test)[:, 1]

        metrics = {
            'accuracy': accuracy_score(y_test, y_pred),
            'precision': precision_score(y_test, y_pred),
            'recall': recall_score(y_test, y_pred),
            'f1_score': f1_score(y_test, y_pred),
            'roc_auc': roc_auc_score(y_test, y_pred_proba)
        }

        mlflow.log_metrics(metrics)
        mlflow.sklearn.log_model(voting_clf, "model")

        logger.info(f"Voting Ensemble - Accuracy: {metrics['accuracy']:.4f}, ROC-AUC: {metrics['roc_auc']:.4f}")

        return voting_clf, metrics


def train_stacking_ensemble(X_train, X_test, y_train, y_test):
    """
    Train Stacking Classifier ensemble.

    Uses a meta-model to learn how to best combine base model predictions.
    Often achieves best performance but more complex.
    """
    logger.info("Training Stacking Ensemble...")

    with mlflow.start_run(run_name="stacking_ensemble"):

        # Define base models
        base_models = [
            ('xgboost', xgb.XGBClassifier(
                max_depth=6,
                learning_rate=0.1,
                n_estimators=200,
                random_state=42,
                eval_metric='logloss'
            )),
            ('lightgbm', lgb.LGBMClassifier(
                num_leaves=31,
                learning_rate=0.1,
                n_estimators=200,
                random_state=42,
                verbose=-1
            )),
            ('random_forest', RandomForestClassifier(
                n_estimators=200,
                max_depth=12,
                random_state=42,
                n_jobs=-1
            ))
        ]

        # Meta-model (learns to combine base models)
        meta_model = LogisticRegression(max_iter=1000, random_state=42)

        # Create stacking ensemble
        stacking_clf = StackingClassifier(
            estimators=base_models,
            final_estimator=meta_model,
            cv=5,  # 5-fold cross-validation
            n_jobs=-1
        )

        mlflow.log_param("ensemble_type", "stacking")
        mlflow.log_param("meta_model", "LogisticRegression")
        mlflow.log_param("cv_folds", 5)
        mlflow.log_param("n_base_models", len(base_models))

        # Train
        logger.info("Training stacking ensemble (this may take a few minutes)...")
        stacking_clf.fit(X_train, y_train)

        # Evaluate
        y_pred = stacking_clf.predict(X_test)
        y_pred_proba = stacking_clf.predict_proba(X_test)[:, 1]

        metrics = {
            'accuracy': accuracy_score(y_test, y_pred),
            'precision': precision_score(y_test, y_pred),
            'recall': recall_score(y_test, y_pred),
            'f1_score': f1_score(y_test, y_pred),
            'roc_auc': roc_auc_score(y_test, y_pred_proba)
        }

        mlflow.log_metrics(metrics)
        mlflow.sklearn.log_model(stacking_clf, "model")

        logger.info(f"Stacking Ensemble - Accuracy: {metrics['accuracy']:.4f}, ROC-AUC: {metrics['roc_auc']:.4f}")

        return stacking_clf, metrics


def save_model(model, name, metrics):
    """Save model to disk."""
    output_dir = Path("data/models")
    output_dir.mkdir(parents=True, exist_ok=True)

    # Save model
    model_file = output_dir / f"{name}.pkl"
    with open(model_file, 'wb') as f:
        pickle.dump(model, f)

    # Save metrics
    import json
    metrics_file = output_dir / f"{name}_metrics.json"
    with open(metrics_file, 'w') as f:
        json.dump(metrics, f, indent=2)

    logger.info(f"Model saved: {model_file}")


def main():
    """Train all ensemble models."""

    mlflow.set_experiment("loan-default-ensemble")

    logger.info("=" * 60)
    logger.info("ENSEMBLE MODEL TRAINING")
    logger.info("=" * 60)

    # Load data
    X_train, X_test, y_train, y_test = load_data()

    # Train models
    all_results = {}

    # 1. LightGBM
    logger.info("\n" + "▶" * 30)
    lgb_model, lgb_metrics = train_lightgbm(X_train, X_test, y_train, y_test)
    save_model(lgb_model, "lightgbm", lgb_metrics)
    all_results['LightGBM'] = lgb_metrics

    # 2. Random Forest
    logger.info("\n" + "▶" * 30)
    rf_model, rf_metrics = train_random_forest(X_train, X_test, y_train, y_test)
    save_model(rf_model, "random_forest", rf_metrics)
    all_results['Random Forest'] = rf_metrics

    # 3. Voting Ensemble
    logger.info("\n" + "▶" * 30)
    voting_model, voting_metrics = train_voting_ensemble(X_train, X_test, y_train, y_test)
    save_model(voting_model, "voting_ensemble", voting_metrics)
    all_results['Voting Ensemble'] = voting_metrics

    # 4. Stacking Ensemble
    logger.info("\n" + "▶" * 30)
    stacking_model, stacking_metrics = train_stacking_ensemble(X_train, X_test, y_train, y_test)
    save_model(stacking_model, "stacking_ensemble", stacking_metrics)
    all_results['Stacking Ensemble'] = stacking_metrics

    # Summary comparison
    logger.info("\n" + "=" * 60)
    logger.info("MODEL COMPARISON")
    logger.info("=" * 60)

    # Create comparison table
    comparison_df = pd.DataFrame(all_results).T
    comparison_df = comparison_df[['accuracy', 'precision', 'recall', 'f1_score', 'roc_auc']]

    logger.info(f"\n{comparison_df.to_string()}")

    # Find best model
    best_model = comparison_df['roc_auc'].idxmax()
    best_score = comparison_df['roc_auc'].max()

    logger.info("\n" + "=" * 60)
    logger.info(f"🏆 BEST MODEL: {best_model}")
    logger.info(f"   ROC-AUC: {best_score:.4f}")
    logger.info("=" * 60)


if __name__ == "__main__":
    main()
```

**Ensemble Concepts Explained:**

1. **Voting (Soft):**
   ```
   Model A: 0.3 (30% default prob)
   Model B: 0.4 (40% default prob)
   Model C: 0.5 (50% default prob)
   Average: 0.4 (40% default prob)
   ```

2. **Stacking:**
   ```
   Base Models → Predictions → Meta-Model → Final Prediction
   [XGB, LGB, RF] → [0.3, 0.4, 0.5] → LogReg → 0.42
   ```

3. **Why Ensembles Work:**
   - Different models make different errors
   - Averaging reduces variance
   - Combines strengths of each model

## Step 4: Create Ensemble Prediction Module

**File:** `src/models/ensemble_predict.py`

```python
"""
Make predictions using ensemble models.
"""

import pickle
from pathlib import Path
import numpy as np

from src.utils.logger import logger


def load_ensemble_model(model_name="stacking_ensemble"):
    """Load saved ensemble model."""
    model_path = Path(f"data/models/{model_name}.pkl")

    with open(model_path, 'rb') as f:
        model = pickle.load(f)

    logger.info(f"Loaded {model_name} model")

    return model


def predict_with_ensemble(model, features):
    """
    Make prediction with ensemble model.

    Args:
        model: Loaded ensemble model
        features: Feature vector (numpy array or DataFrame row)

    Returns:
        (prediction, probability)
    """
    prediction = model.predict([features])[0]
    probability = model.predict_proba([features])[0][1]

    return int(prediction), float(probability)


def main():
    """Example ensemble prediction."""

    # Load model
    model = load_ensemble_model("stacking_ensemble")

    # Example features (would come from preprocessing in production)
    example_features = np.random.randn(34)  # 34 features

    prediction, probability = predict_with_ensemble(model, example_features)

    logger.info(f"Prediction: {prediction}")
    logger.info(f"Probability: {probability:.3f}")


if __name__ == "__main__":
    main()
```

## When to Use Each Ensemble Method

**Voting:**
- ✅ Quick to implement
- ✅ Easy to understand
- ✅ Good baseline ensemble
- ❌ Simple averaging may not be optimal

**Stacking:**
- ✅ Usually best performance
- ✅ Learns optimal combination
- ❌ More complex
- ❌ Slower training
- ❌ Harder to interpret

**Choose stacking when:**
- Maximum accuracy needed
- Have enough training data
- Can afford longer training time

**Choose voting when:**
- Need quick ensemble
- Want interpretability
- Limited computational resources

## Troubleshooting

**Issue:** Stacking takes too long
**Solution:** Reduce cv_folds from 5 to 3, or use fewer base models

**Issue:** Ensemble worse than best single model
**Solution:** Check model diversity - models should be different

**Issue:** Out of memory error
**Solution:** Train models separately, combine later

**Issue:** Voting gives same result as best model
**Solution:** Normal if models are very similar
