"""
End-to-end integration tests for Credit Risk Engine.

Tests the complete flow from API request to prediction, including:
- Database connections
- Redis caching
- Model inference
- SHAP explanations
- Audit logging
"""
import pytest
import requests
import time
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
import redis


class TestE2EIntegration:
    """End-to-end integration tests."""

    @classmethod
    def setup_class(cls):
        """Setup test fixtures."""
        cls.base_url = "http://localhost:8000"
        cls.db_url = "postgresql://postgres:password@localhost:5432/credit_risk_db"
        cls.redis_client = redis.Redis(host='localhost', port=6379, db=0)

        # Database session
        engine = create_engine(cls.db_url)
        Session = sessionmaker(bind=engine)
        cls.db = Session()

    def test_01_health_check(self):
        """Test health endpoint."""
        response = requests.get(f"{self.base_url}/health")

        assert response.status_code == 200
        data = response.json()
        assert data['status'] == 'healthy'
        assert 'database' in data
        assert 'redis' in data
        assert 'model' in data

    def test_02_prediction_flow(self):
        """Test complete prediction flow."""
        payload = {
            "loan_amount": 50000,
            "age": 35,
            "employment_length": 5,
            "income": 75000,
            "debt_to_income": 0.35,
            "credit_score": 720,
            "num_credit_lines": 8,
            "loan_purpose": "debt_consolidation",
            "state_code": "LA"
        }

        # Make prediction
        start_time = time.time()
        response = requests.post(
            f"{self.base_url}/api/v1/predict",
            json=payload
        )
        response_time = (time.time() - start_time) * 1000  # ms

        # Verify response
        assert response.status_code == 200
        data = response.json()
        assert 'default_probability' in data
        assert 'prediction' in data
        assert 0 <= data['default_probability'] <= 1

        # Verify response time SLA (< 500ms)
        assert response_time < 500, f"Response time {response_time}ms exceeds 500ms SLA"

        # Verify audit log was created
        from app.db.models import AuditLog
        latest_log = self.db.query(AuditLog).order_by(
            AuditLog.timestamp.desc()
        ).first()

        assert latest_log is not None
        assert latest_log.event_type == 'prediction'
        assert latest_log.prediction is not None

    def test_03_explanation_flow(self):
        """Test explanation generation."""
        payload = {
            "loan_amount": 50000,
            "age": 35,
            "employment_length": 5,
            "income": 75000,
            "debt_to_income": 0.35,
            "credit_score": 720,
            "num_credit_lines": 8,
            "loan_purpose": "debt_consolidation",
            "state_code": "LA"
        }

        response = requests.post(
            f"{self.base_url}/api/v1/explain/explain",
            json=payload
        )

        assert response.status_code == 200
        data = response.json()
        assert 'base_value' in data
        assert 'prediction' in data
        assert 'contributions' in data
        assert len(data['contributions']) > 0
        assert 'top_factors' in data

    def test_04_caching_works(self):
        """Test Redis caching."""
        payload = {
            "loan_amount": 50000,
            "age": 35,
            "employment_length": 5,
            "income": 75000,
            "debt_to_income": 0.35,
            "credit_score": 720,
            "num_credit_lines": 8,
            "loan_purpose": "debt_consolidation",
            "state_code": "LA"
        }

        # First request (cache miss)
        start = time.time()
        response1 = requests.post(f"{self.base_url}/api/v1/predict", json=payload)
        time1 = (time.time() - start) * 1000

        # Second request (cache hit)
        start = time.time()
        response2 = requests.post(f"{self.base_url}/api/v1/predict", json=payload)
        time2 = (time.time() - start) * 1000

        # Cache hit should be faster
        assert time2 < time1, "Cached request should be faster"

        # Results should match
        assert response1.json()['default_probability'] == \
               response2.json()['default_probability']

    def test_05_what_if_analysis(self):
        """Test what-if analysis."""
        payload = {
            "base_features": {
                "loan_amount": 50000,
                "age": 35,
                "employment_length": 5,
                "income": 75000,
                "debt_to_income": 0.35,
                "credit_score": 720,
                "num_credit_lines": 8,
                "loan_purpose": "debt_consolidation",
                "state_code": "LA"
            },
            "scenarios": [
                {
                    "name": "Higher income",
                    "changes": {"income": 85000}
                }
            ]
        }

        response = requests.post(
            f"{self.base_url}/api/v1/whatif/scenarios",
            json=payload
        )

        assert response.status_code == 200
        data = response.json()
        assert 'base_risk' in data
        assert 'scenarios' in data
        assert len(data['scenarios']) == 1

    def test_06_model_card(self):
        """Test model card endpoint."""
        response = requests.get(f"{self.base_url}/api/v1/model-info/model-card")

        assert response.status_code == 200
        data = response.json()
        assert 'model_details' in data
        assert 'intended_use' in data
        assert 'metrics' in data
        assert 'ethical_considerations' in data

    def test_07_concurrent_requests(self):
        """Test system under concurrent load."""
        import concurrent.futures

        payload = {
            "loan_amount": 50000,
            "age": 35,
            "employment_length": 5,
            "income": 75000,
            "debt_to_income": 0.35,
            "credit_score": 720,
            "num_credit_lines": 8,
            "loan_purpose": "debt_consolidation",
            "state_code": "LA"
        }

        def make_request():
            response = requests.post(f"{self.base_url}/api/v1/predict", json=payload)
            return response.status_code == 200

        # Send 50 concurrent requests
        with concurrent.futures.ThreadPoolExecutor(max_workers=10) as executor:
            futures = [executor.submit(make_request) for _ in range(50)]
            results = [f.result() for f in concurrent.futures.as_completed(futures)]

        # All requests should succeed
        assert all(results), "Some concurrent requests failed"

    @classmethod
    def teardown_class(cls):
        """Cleanup."""
        cls.db.close()
