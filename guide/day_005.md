## Understanding Synthetic Data

**What is synthetic data?**
Artificially generated data that mimics real-world patterns. We use it because:
- Real loan data is confidential and regulated
- We can control distributions and edge cases
- Perfect for development and testing
- No privacy concerns

**Our approach:**
- Generate realistic Nigerian applicant profiles
- Create correlations (e.g., higher income → lower default risk)
- Add controlled randomness for variety
- Export to CSV for ML training

```python
Generate synthetic Nigerian loan application data.
"""

import pandas as pd
import numpy as np
from faker import Faker
from datetime import datetime, timedelta
import random
from pathlib import Path

from src.utils.logger import logger

# Set random seeds for reproducibility
np.random.seed(42)
random.seed(42)
fake = Faker()

# Nigerian-specific data
NIGERIAN_FIRST_NAMES = [
    "Adebayo", "Chinwe", "Emeka", "Folake", "Ibrahim", "Ngozi", "Oluwaseun",
    "Chioma", "Tunde", "Amara", "Yusuf", "Blessing", "Chukwudi", "Fatima",

    "Adedayo", "Amarachi", "Bashir", "Temitope", "Chidi", "Zainab"
]

NIGERIAN_LAST_NAMES = [
    "Adeyemi", "Okafor", "Musa", "Bello", "Eze", "Williams", "Johnson",
    "Okonkwo", "Ibrahim", "Oluwaseun", "Afolabi", "Nwosu", "Abdullahi",
    "Oni", "Okeke", "Mohammed", "Adeleke", "Chibueze", "Sani", "Udoh",
    "Hassan", "Ogunleye", "Nnamdi", "Yusuf", "Akinola", "Chukwu", "Usman",
    "Adebisi", "Nnaji", "Aliyu", "Taiwo", "Okoro", "Garba", "Ojo"
]

NIGERIAN_BANKS = [
    "Access Bank", "GTBank", "First Bank", "Zenith Bank", "UBA",
    "Stanbic IBTC", "Fidelity Bank", "Union Bank", "Sterling Bank",
    "Ecobank", "FCMB", "Wema Bank", "Polaris Bank", "Keystone Bank"
]

NIGERIAN_STATES = [
    "Lagos", "Kano", "Kaduna", "Oyo", "Rivers", "Abuja", "Ogun",
    "Anambra", "Enugu", "Delta", "Imo", "Edo", "Akwa Ibom", "Ondo",
    "Osun", "Kwara", "Benue", "Plateau", "Abia", "Sokoto"
]

EMPLOYMENT_SECTORS = [
    "Banking & Finance", "Oil & Gas", "Telecommunications", "Technology",
    "Government", "Healthcare", "Education", "Agriculture", "Real Estate",
    "Manufacturing", "Retail", "Hospitality", "Construction", "Transportation",
    "Professional Services", "Media & Entertainment"
]

LOAN_PURPOSES = [
    "Business expansion", "Working capital", "Equipment purchase",
    "Inventory financing", "Real estate investment", "Vehicle purchase",
    "Education", "Home renovation", "Debt consolidation",
    "Agricultural inputs", "Trade financing", "Medical expenses"
]

EMPLOYMENT_STATUSES = ["EMPLOYED", "SELF_EMPLOYED", "UNEMPLOYED"]
EDUCATION_LEVELS = ["SECONDARY", "OND/NCE", "HND/BACHELOR", "MASTERS", "PHD"]
MARITAL_STATUSES = ["SINGLE", "MARRIED", "DIVORCED", "WIDOWED"]


def generate_nigerian_name() -> tuple:
    """
    Generate a realistic Nigerian name.

    Returns:
        (first_name, last_name) tuple
    """
    first_name = random.choice(NIGERIAN_FIRST_NAMES)
    last_name = random.choice(NIGERIAN_LAST_NAMES)
    return first_name, last_name


def generate_loan_applications(n_samples: int = 10000) -> pd.DataFrame:
    """
    Generate synthetic loan application data.

    Args:
        n_samples: Number of loan applications to generate

    Returns:
        DataFrame with loan applications
    """
    logger.info(f"Generating {n_samples} loan applications...")

    data = []

    for i in range(n_samples):
        # Personal information
        first_name, last_name = generate_nigerian_name()
        full_name = f"{first_name} {last_name}"

        age = np.random.randint(22, 65)
        gender = random.choice(["Male", "Female"])
        marital_status = random.choice(MARITAL_STATUSES)
        education = random.choice(EDUCATION_LEVELS)

        # Employment
        employment_status = np.random.choice(
            EMPLOYMENT_STATUSES,
            p=[0.60, 0.30, 0.10]  # 60% employed, 30% self-employed, 10% unemployed
        )

        # Income based on employment status and education
        if employment_status == "UNEMPLOYED":
            monthly_income = np.random.uniform(0, 50000)
        elif employment_status == "SELF_EMPLOYED":
            monthly_income = np.random.uniform(50000, 800000)
        else:  # EMPLOYED
            # Income varies by education
            if education == "PHD":
                monthly_income = np.random.uniform(400000, 2000000)
            elif education == "MASTERS":
                monthly_income = np.random.uniform(250000, 1200000)
            elif education == "HND/BACHELOR":
                monthly_income = np.random.uniform(150000, 800000)
            elif education == "OND/NCE":
                monthly_income = np.random.uniform(80000, 400000)
            else:  # SECONDARY
                monthly_income = np.random.uniform(50000, 200000)

        years_employed = 0 if employment_status == "UNEMPLOYED" else np.random.randint(1, min(age - 20, 40))
        employer_sector = None if employment_status == "UNEMPLOYED" else random.choice(EMPLOYMENT_SECTORS)

        # Loan details (correlated with income)
        # Typically, loan amount is 2-30x monthly income
        max_loan_multiplier = 30 if employment_status == "EMPLOYED" else 15
        loan_amount = monthly_income * np.random.uniform(2, max_loan_multiplier)
        loan_amount = round(loan_amount / 100000) * 100000  # Round to nearest 100k

        loan_purpose = random.choice(LOAN_PURPOSES)
        requested_tenure_months = random.choice([6, 12, 18, 24, 36, 48, 60])

        # Financial history
        has_existing_loans = random.choice([True, False])
        num_existing_loans = np.random.randint(0, 5) if has_existing_loans else 0
        total_existing_debt = np.random.uniform(0, loan_amount * 2) if has_existing_loans else 0

        has_savings_account = random.choice([True, False], p=[0.7, 0.3])
        savings_balance = np.random.uniform(0, monthly_income * 6) if has_savings_account else 0

        credit_score = np.random.randint(300, 850)

        # Location
        state = random.choice(NIGERIAN_STATES)
        bank = random.choice(NIGERIAN_BANKS)

        # Calculate default probability (this is our target variable)
        # Higher risk factors:
        # - Low income
        # - High debt-to-income ratio
        # - Unemployed
        # - Low credit score
        # - Many existing loans
        # - Young age with high loan amount

        debt_to_income = (total_existing_debt + loan_amount) / (monthly_income * 12) if monthly_income > 0 else 10
        employment_risk = 0.8 if employment_status == "UNEMPLOYED" else (0.3 if employment_status == "SELF_EMPLOYED" else 0.1)
        credit_risk = (850 - credit_score) / 550  # Normalize credit score
        age_risk = 0.5 if age < 25 else (0.3 if age > 60 else 0.1)

        # Combine risk factors
        default_probability = (
            0.30 * min(debt_to_income / 5, 1.0) +  # 30% weight to debt ratio
            0.25 * employment_risk +                # 25% weight to employment
            0.25 * credit_risk +                    # 25% weight to credit score
            0.10 * age_risk +                       # 10% weight to age
            0.10 * (num_existing_loans / 10)        # 10% weight to existing loans
        )

        # Add some randomness
        default_probability = max(0, min(1, default_probability + np.random.normal(0, 0.1)))

        # Determine if loan defaulted (1) or not (0)
        defaulted = 1 if random.random() < default_probability else 0

        # Append to data
        data.append({
            "application_id": f"APP{i+1:06d}",
            "applicant_name": full_name,
            "age": age,
            "gender": gender,
            "marital_status": marital_status,
            "education_level": education,
            "employment_status": employment_status,
            "employer_sector": employer_sector,
            "years_employed": years_employed,
            "monthly_income": round(monthly_income, 2),
            "loan_amount": loan_amount,
            "loan_purpose": loan_purpose,
            "requested_tenure_months": requested_tenure_months,
            "has_existing_loans": has_existing_loans,
            "num_existing_loans": num_existing_loans,
            "total_existing_debt": round(total_existing_debt, 2),
            "has_savings_account": has_savings_account,
            "savings_balance": round(savings_balance, 2),
            "credit_score": credit_score,
            "state": state,
            "bank": bank,
            "default_probability": round(default_probability, 4),
            "defaulted": defaulted
        })

    df = pd.DataFrame(data)
    logger.info(f"Generated {len(df)} loan applications")

    return df


def print_data_statistics(df: pd.DataFrame):
    """Print summary statistics of the generated data."""

    logger.info("\n" + "=" * 60)
    logger.info("DATA GENERATION SUMMARY")
    logger.info("=" * 60)

    logger.info(f"\nTotal Applications: {len(df)}")
    logger.info(f"Default Rate: {df['defaulted'].mean() * 100:.2f}%")

    logger.info("\nEmployment Status Distribution:")
    for status in df['employment_status'].value_counts().items():
        logger.info(f"  {status[0]}: {status[1]} ({status[1]/len(df)*100:.1f}%)")

    logger.info("\nLoan Amount Statistics (Naira):")
    logger.info(f"  Mean: ₦{df['loan_amount'].mean():,.2f}")
    logger.info(f"  Median: ₦{df['loan_amount'].median():,.2f}")
    logger.info(f"  Min: ₦{df['loan_amount'].min():,.2f}")
    logger.info(f"  Max: ₦{df['loan_amount'].max():,.2f}")

    logger.info("\nMonthly Income Statistics (Naira):")
    logger.info(f"  Mean: ₦{df['monthly_income'].mean():,.2f}")
    logger.info(f"  Median: ₦{df['monthly_income'].median():,.2f}")

    logger.info("\nCredit Score Distribution:")
    logger.info(f"  Mean: {df['credit_score'].mean():.0f}")
    logger.info(f"  Median: {df['credit_score'].median():.0f}")

    logger.info("\nTop 5 States:")
    for state in df['state'].value_counts().head().items():
        logger.info(f"  {state[0]}: {state[1]}")

    logger.info("=" * 60 + "\n")


def main():
    """Main function to generate and save data."""

    # Generate data
    df = generate_loan_applications(n_samples=10000)

    # Print statistics
    print_data_statistics(df)

    # Save to CSV
    output_dir = Path("data/raw")
    output_dir.mkdir(parents=True, exist_ok=True)

    output_file = output_dir / "loan_applications.csv"
    df.to_csv(output_file, index=False)

    logger.info(f"Data saved to: {output_file}")
    logger.info(f"File size: {output_file.stat().st_size / 1024 / 1024:.2f} MB")

    # Save a smaller sample for quick testing
    sample_file = output_dir / "loan_applications_sample.csv"
    df.sample(n=1000, random_state=42).to_csv(sample_file, index=False)
    logger.info(f"Sample data (1000 rows) saved to: {sample_file}")


if __name__ == "__main__":
    main()
```

```python
import pandas as pd

def inspect_data(file_path: str):
    """Load and inspect CSV data."""

    logger.info(f"Loading data from: {file_path}")
    df = pd.read_csv(file_path)

    logger.info("\n" + "=" * 60)
    logger.info("DATASET OVERVIEW")
    logger.info("=" * 60)

    logger.info(f"\nShape: {df.shape[0]} rows x {df.shape[1]} columns")

    logger.info("\nColumn Names:")
    for i, col in enumerate(df.columns, 1):
        logger.info(f"  {i:2d}. {col}")

    logger.info("\nData Types:")
    logger.info(df.dtypes)

    logger.info("\nMissing Values:")
    missing = df.isnull().sum()
    if missing.sum() == 0:
        logger.info("  No missing values! ✓")
    else:
        logger.info(missing[missing > 0])

    logger.info("\nFirst 5 Rows:")
    logger.info("\n" + df.head().to_string())

    logger.info("\nNumerical Summary:")
    logger.info("\n" + df.describe().to_string())

    logger.info("\nCategorical Summary:")
    categorical_cols = df.select_dtypes(include=['object']).columns
    for col in categorical_cols[:5]:  # Show first 5 categorical columns
        logger.info(f"\n{col}:")
        logger.info(df[col].value_counts().head())

    logger.info("\n" + "=" * 60)


if __name__ == "__main__":
    data_file = Path("data/raw/loan_applications.csv")

    if data_file.exists():
        inspect_data(data_file)
```

## Step 6: Create Data Validation Endpoint

Let's add an endpoint to check data quality via API.

```python
"""
Data-related API endpoints.
"""

from fastapi import APIRouter, HTTPException
from pathlib import Path
import pandas as pd

from src.utils.logger import logger

router = APIRouter(prefix="/api/v1/data", tags=["Data"])


@router.get("/stats")
async def get_data_statistics():
    """
    Get statistics about the generated loan data.

    Returns summary statistics without authentication.
    """
    data_file = Path("data/raw/loan_applications.csv")

    if not data_file.exists():
        raise HTTPException(
            status_code=404,
            detail="Data file not found. Run data generation first."
        )

    df = pd.read_csv(data_file)

    stats = {
        "total_applications": len(df),
        "default_rate": round(df['defaulted'].mean() * 100, 2),
        "employment_distribution": df['employment_status'].value_counts().to_dict(),
        "loan_amount_stats": {
            "mean": round(df['loan_amount'].mean(), 2),
            "median": round(df['loan_amount'].median(), 2),
            "min": round(df['loan_amount'].min(), 2),
            "max": round(df['loan_amount'].max(), 2)
        },
        "income_stats": {
            "mean": round(df['monthly_income'].mean(), 2),
            "median": round(df['monthly_income'].median(), 2),
            "min": round(df['monthly_income'].min(), 2),
            "max": round(df['monthly_income'].max(), 2)
        },
        "credit_score_stats": {
            "mean": round(df['credit_score'].mean(), 0),
            "median": round(df['credit_score'].median(), 0),
            "min": int(df['credit_score'].min()),
            "max": int(df['credit_score'].max())
        },
        "top_states": df['state'].value_counts().head(5).to_dict(),
        "education_distribution": df['education_level'].value_counts().to_dict()
    }

    return stats


@router.get("/sample/{n}")
async def get_sample_data(n: int = 10):
    """
    Get n random sample rows from the dataset.

    Args:
        n: Number of rows to return (max 100)
    """
    if n > 100:
        raise HTTPException(
            status_code=400,
            detail="Maximum sample size is 100 rows"
        )

    data_file = Path("data/raw/loan_applications.csv")

    if not data_file.exists():
        raise HTTPException(
            status_code=404,
            detail="Data file not found"
        )

    df = pd.read_csv(data_file)
    sample = df.sample(n=min(n, len(df))).to_dict('records')

    return {
        "total_rows": len(df),
        "sample_size": len(sample),
        "data": sample
    }
```
