from sqlalchemy.orm import Session
from sqlalchemy import and_, func
from typing import List, Dict, Any, Optional
from datetime import datetime, timedelta

from app.db.models import AuditLog


class OptimizedQueries:
    """Optimized database queries with proper indexing."""

    @staticmethod
    def get_recent_predictions(
        db: Session,
        user_id: Optional[str] = None,
        days: int = 7,
        limit: int = 100
    ) -> List[Dict[str, Any]]:
        """
        Get recent predictions with optimized query.

        Uses indexes: idx_audit_user_timestamp, idx_audit_recent
        """
        cutoff = datetime.now() - timedelta(days=days)

        query = db.query(
            AuditLog.id,
            AuditLog.timestamp,
            AuditLog.user_id,
            AuditLog.prediction,
            AuditLog.response_time_ms
        ).filter(
            and_(
                AuditLog.event_type == 'prediction',
                AuditLog.timestamp >= cutoff
            )
        )

        if user_id:
            query = query.filter(AuditLog.user_id == user_id)

        query = query.order_by(AuditLog.timestamp.desc()).limit(limit)

        return [
            {
                'id': str(row.id),
                'timestamp': row.timestamp.isoformat(),
                'user_id': row.user_id,
                'prediction': row.prediction,
                'response_time_ms': row.response_time_ms
            }
            for row in query.all()
        ]

    @staticmethod
    def get_prediction_stats(
        db: Session,
        days: int = 7
    ) -> Dict[str, Any]:
        """
        Get aggregated prediction statistics.

        Uses index: idx_audit_model_timestamp
        """
        cutoff = datetime.now() - timedelta(days=days)

        stats = db.query(
            func.count(AuditLog.id).label('total'),
            func.avg(AuditLog.response_time_ms).label('avg_response_time'),
            func.percentile_cont(0.95).within_group(
                AuditLog.response_time_ms
            ).label('p95_response_time'),
            func.percentile_cont(0.99).within_group(
                AuditLog.response_time_ms
            ).label('p99_response_time')
        ).filter(
            and_(
                AuditLog.event_type == 'prediction',
                AuditLog.timestamp >= cutoff,
                AuditLog.error.is_(None)  # Only successful predictions
            )
        ).first()

        return {
            'period_days': days,
            'total_predictions': stats.total or 0,
            'avg_response_time_ms': float(stats.avg_response_time or 0),
            'p95_response_time_ms': float(stats.p95_response_time or 0),
            'p99_response_time_ms': float(stats.p99_response_time or 0)
        }

**`app/cache/redis_client.py`**:

```python
import redis
from typing import Optional, Any
import json
import pickle
from functools import wraps
import hashlib
import logging

from app.core.config import settings

logger = logging.getLogger(__name__)


class RedisCache:
    """Redis caching client with JSON and pickle support."""

    def __init__(self):
        self.client = redis.Redis(
            host=settings.REDIS_HOST,
            port=settings.REDIS_PORT,
            db=settings.REDIS_DB,
            password=settings.REDIS_PASSWORD if hasattr(settings, 'REDIS_PASSWORD') else None,
            decode_responses=False,  # Handle bytes for pickle
            socket_connect_timeout=5,
            socket_timeout=5,
            retry_on_timeout=True,
            health_check_interval=30
        )

    def get(self, key: str, use_pickle: bool = False) -> Optional[Any]:
        """
        Get value from cache.

        Args:
            key: Cache key
            use_pickle: Use pickle for deserialization (for complex objects)

        Returns:
            Cached value or None
        """
        try:
            value = self.client.get(key)
            if value is None:
                return None

            if use_pickle:
                return pickle.loads(value)
            else:
                return json.loads(value.decode('utf-8'))

        except Exception as e:
            logger.error(f"Cache get error: {e}")
            return None

    def set(
        self,
        key: str,
        value: Any,
        ttl: int = 3600,
        use_pickle: bool = False
    ) -> bool:
        """
        Set value in cache.

        Args:
            key: Cache key
            value: Value to cache
            ttl: Time to live in seconds
            use_pickle: Use pickle for serialization

        Returns:
            Success status
        """
        try:
            if use_pickle:
                serialized = pickle.dumps(value)
            else:
                serialized = json.dumps(value)

            return self.client.setex(key, ttl, serialized)

        except Exception as e:
            logger.error(f"Cache set error: {e}")
            return False

    def delete(self, key: str) -> bool:
        """Delete key from cache."""
        try:
            return bool(self.client.delete(key))
        except Exception as e:
            logger.error(f"Cache delete error: {e}")
            return False

    def exists(self, key: str) -> bool:
        """Check if key exists in cache."""
        try:
            return bool(self.client.exists(key))
        except Exception as e:
            logger.error(f"Cache exists error: {e}")
            return False

    def clear_pattern(self, pattern: str) -> int:
        """
        Delete all keys matching pattern.

        Args:
            pattern: Key pattern (e.g., "user:*")

        Returns:
            Number of keys deleted
        """
        try:
            keys = self.client.keys(pattern)
            if keys:
                return self.client.delete(*keys)
            return 0
        except Exception as e:
            logger.error(f"Cache clear pattern error: {e}")
            return 0


# Global cache instance
cache = RedisCache()


def cached(ttl: int = 3600, key_prefix: str = "", use_pickle: bool = False):
    """
    Decorator for caching function results.

    Args:
        ttl: Cache TTL in seconds
        key_prefix: Prefix for cache key
        use_pickle: Use pickle serialization

    Example:
        @cached(ttl=300, key_prefix="predictions")
        def get_prediction(user_id: str):
            ...
    """
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            # Generate cache key from function name and arguments
            key_data = f"{func.__name__}:{args}:{kwargs}"
            key_hash = hashlib.md5(key_data.encode()).hexdigest()
            cache_key = f"{key_prefix}:{key_hash}" if key_prefix else key_hash

            # Try to get from cache
            cached_value = cache.get(cache_key, use_pickle=use_pickle)
            if cached_value is not None:
                logger.debug(f"Cache hit: {cache_key}")
                return cached_value

            # Cache miss - compute value
            logger.debug(f"Cache miss: {cache_key}")
            result = func(*args, **kwargs)

            # Store in cache
            cache.set(cache_key, result, ttl=ttl, use_pickle=use_pickle)

            return result

        return wrapper
    return decorator
```

from typing import Dict, Any
import hashlib
import json

from app.ml.predictor import Predictor
from app.cache.redis_client import cache


class CachedPredictor(Predictor):
    """Predictor with caching for identical inputs."""

    def __init__(self, model_path: str, cache_ttl: int = 300):
        """
        Initialize cached predictor.

        Args:
            model_path: Path to model file
            cache_ttl: Cache TTL in seconds (default 5 minutes)
        """
        super().__init__(model_path)
        self.cache_ttl = cache_ttl

    def predict(self, features: Dict[str, Any]) -> Dict[str, Any]:
        """
        Make prediction with caching.

        Identical inputs return cached results.
        """
        # Generate cache key from features
        features_str = json.dumps(features, sort_keys=True)
        cache_key = f"prediction:{hashlib.md5(features_str.encode()).hexdigest()}"

        # Check cache
        cached_result = cache.get(cache_key)
        if cached_result is not None:
            cached_result['cached'] = True
            return cached_result

        # Cache miss - compute prediction
        result = super().predict(features)

        # Store in cache
        cache.set(cache_key, result, ttl=self.cache_ttl)

        result['cached'] = False
        return result
```

### Step 1: Batch Prediction Support

import numpy as np
import pandas as pd
from typing import List, Dict, Any


class BatchPredictor(Predictor):
    """Optimized predictor with batch inference support."""

    def predict_batch(
        self,
        features_list: List[Dict[str, Any]],
        batch_size: int = 32
    ) -> List[Dict[str, Any]]:
        """
        Predict in batches for better performance.

        Args:
            features_list: List of feature dictionaries
            batch_size: Batch size for inference

        Returns:
            List of predictions
        """
        from app.ml.preprocessing import preprocess_features

        results = []

        # Process in batches
        for i in range(0, len(features_list), batch_size):
            batch = features_list[i:i + batch_size]

            # Preprocess batch
            processed = [preprocess_features(f) for f in batch]
            features_df = pd.DataFrame(processed)

            # Batch predict
            probabilities = self.model.predict_proba(features_df)[:, 1]
            predictions = (probabilities > 0.5).astype(int)

            # Format results
            for j, (prob, pred) in enumerate(zip(probabilities, predictions)):
                results.append({
                    'default_probability': float(prob),
                    'prediction': 'default' if pred == 1 else 'no_default',
                    'risk_category': self._get_risk_category(prob)
                })

        return results

    def _get_risk_category(self, probability: float) -> str:
        """Classify risk level."""
        if probability < 0.2:
            return 'low'
        elif probability < 0.5:
            return 'medium'
        else:
            return 'high'
```

import pickle
import onnx
import onnxruntime as ort
from skl2onnx import convert_sklearn
from skl2onnx.common.data_types import FloatTensorType


def quantize_model(model_path: str, output_path: str):
    """
    Convert sklearn model to ONNX and quantize.

    Args:
        model_path: Path to pickle model
        output_path: Path for ONNX model
    """
    # Load model
    with open(model_path, 'rb') as f:
        model = pickle.load(f)

    # Define input type
    initial_type = [('float_input', FloatTensorType([None, 20]))]  # Adjust feature count

    # Convert to ONNX
    onnx_model = convert_sklearn(model, initial_types=initial_type)

    # Save ONNX model
    with open(output_path, 'wb') as f:
        f.write(onnx_model.SerializeToString())

    print(f"Quantized model saved to {output_path}")
