```python
"""
Complete system integration test.

Tests all 45 days of features working together.
"""
import pytest
import asyncio
import requests
import websockets
import json
from datetime import datetime


class TestCompleteSystem:
    """Test entire system integration."""

    @classmethod
    def setup_class(cls):
        cls.base_url = "http://localhost:8000"
        cls.ws_url = "ws://localhost:8000"

    def test_01_health_check_all_systems(self):
        """Verify all systems are healthy."""
        # REST API
        response = requests.get(f"{self.base_url}/health")
        assert response.status_code == 200
        health = response.json()
        assert health['database'] == 'healthy'
        assert health['redis'] == 'healthy'
        assert health['model'] == 'healthy'

    def test_02_sync_prediction_flow(self):
        """Test synchronous prediction (Days 001-010)."""
        payload = self._get_test_payload()

        response = requests.post(
            f"{self.base_url}/api/v1/predict",
            json=payload
        )

        assert response.status_code == 200
        result = response.json()
        assert 'default_probability' in result
        assert 'prediction_id' in result
        assert result['blockchain_recorded'] == True

        return result['prediction_id']

    def test_03_async_prediction_flow(self):
        """Test async prediction via Kafka (Day 042)."""
        payload = self._get_test_payload()

        # Send async request
        response = requests.post(
            f"{self.base_url}/api/v1/predict/async",
            json=payload
        )

        assert response.status_code == 200
        result = response.json()
        assert result['status'] == 'queued'
        request_id = result['request_id']

        # Poll for result (wait up to 10 seconds)
        import time
        for _ in range(20):
            time.sleep(0.5)
            response = requests.get(
                f"{self.base_url}/api/v1/predict/async/{request_id}"
            )
            if response.json().get('status') == 'completed':
                break

        assert response.json()['status'] == 'completed'

    async def test_04_websocket_streaming(self):
        """Test WebSocket real-time updates (Day 041)."""
        async with websockets.connect(
            f"{self.ws_url}/ws/predictions"
        ) as websocket:

            # Wait for connection message
            message = await websocket.recv()
            data = json.loads(message)
            assert data['type'] == 'connected'

            # Make prediction in background
            import threading
            def make_prediction():
                requests.post(
                    f"{self.base_url}/api/v1/predict",
                    json=self._get_test_payload()
                )

            thread = threading.Thread(target=make_prediction)
            thread.start()

            # Should receive prediction via WebSocket
            message = await asyncio.wait_for(
                websocket.recv(),
                timeout=5.0
            )
            data = json.loads(message)
            assert data['type'] == 'new_prediction'
            assert 'prediction' in data

            thread.join()

    def test_05_blockchain_verification(self):
        """Test blockchain audit trail (Day 043)."""
        # Make prediction
        prediction_id = self.test_02_sync_prediction_flow()

        # Verify on blockchain
        response = requests.get(
            f"{self.base_url}/api/v1/blockchain/verify/{prediction_id}"
        )

        assert response.status_code == 200
        result = response.json()
        assert result['verified'] == True
        assert 'block_hash' in result

    def test_06_shap_explanation(self):
        """Test SHAP explainability (Day 036)."""
        payload = self._get_test_payload()

        response = requests.post(
            f"{self.base_url}/api/v1/explain/explain",
            json=payload
        )

        assert response.status_code == 200
        result = response.json()
        assert 'contributions' in result
        assert 'top_factors' in result
        assert len(result['top_factors']) > 0

    def test_07_what_if_analysis(self):
        """Test what-if analysis (Day 036)."""
        payload = {
            "base_features": self._get_test_payload(),
            "scenarios": [
                {
                    "name": "Higher income",
                    "changes": {"income": 100000}
                }
            ]
        }

        response = requests.post(
            f"{self.base_url}/api/v1/whatif/scenarios",
            json=payload
        )

        assert response.status_code == 200
        result = response.json()
        assert 'scenarios' in result
        assert len(result['scenarios']) == 1

    def test_08_model_card(self):
        """Test model card (Day 036)."""
        response = requests.get(
            f"{self.base_url}/api/v1/model-info/model-card"
        )

        assert response.status_code == 200
        card = response.json()
        assert 'model_details' in card
        assert 'metrics' in card
        assert 'ethical_considerations' in card

    def test_09_cqrs_pattern(self):
        """Test CQRS commands/queries (Day 044)."""
        payload = self._get_test_payload()

        # Send command
        response = requests.post(
            f"{self.base_url}/api/v1/cqrs/commands/predict",
            json=payload
        )

        assert response.status_code == 200
        result = response.json()
        command_id = result['command_id']

        # Query result
        import time
        time.sleep(2)  # Wait for processing

        response = requests.get(
            f"{self.base_url}/api/v1/cqrs/queries/prediction/{command_id}"
        )

        assert response.status_code == 200

    def test_10_saga_pattern(self):
        """Test Saga pattern (Day 044)."""
        payload = self._get_test_payload()

        response = requests.post(
            f"{self.base_url}/api/v1/predict/saga",
            json=payload
        )

        assert response.status_code == 200
        result = response.json()
        assert result['status'] in ['completed', 'failed']

        if result['status'] == 'failed':
            assert result['compensated'] == True

    def test_11_performance_under_load(self):
        """Test system performs well under load."""
        import concurrent.futures
        import time

        start = time.time()
        results = []

        def make_request():
            response = requests.post(
                f"{self.base_url}/api/v1/predict",
                json=self._get_test_payload()
            )
            return response.status_code == 200

        # 100 concurrent requests
        with concurrent.futures.ThreadPoolExecutor(max_workers=20) as executor:
            futures = [executor.submit(make_request) for _ in range(100)]
            results = [f.result() for f in concurrent.futures.as_completed(futures)]

        duration = time.time() - start

        # Verify
        assert all(results), "Some requests failed"
        assert duration < 30, f"Too slow: {duration}s for 100 requests"

        print(f"✅ 100 requests completed in {duration:.2f}s")
        print(f"✅ Average: {duration/100:.3f}s per request")
        print(f"✅ Throughput: {100/duration:.2f} req/s")

    @staticmethod
    def _get_test_payload():
        """Get test prediction payload."""
        return {
            "loan_amount": 50000,
            "age": 35,
            "employment_length": 5,
            "income": 75000,
            "debt_to_income": 0.35,
            "credit_score": 720,
            "num_credit_lines": 8,
            "loan_purpose": "debt_consolidation",
            "state_code": "LA"
        }
```

```python
from locust import HttpUser, task, between, events
import random
import json


class CompleteSystemUser(HttpUser):
    """
    Load test for complete system.

    Tests all endpoints with realistic traffic distribution.
    """

    wait_time = between(1, 3)

    @task(50)  # 50% of traffic
    def sync_prediction(self):
        """Synchronous prediction (most common)."""
        self.client.post("/api/v1/predict", json=self._get_payload())

    @task(20)  # 20% of traffic
    def async_prediction(self):
        """Async prediction via Kafka."""
        response = self.client.post(
            "/api/v1/predict/async",
            json=self._get_payload()
        )
        if response.status_code == 200:
            request_id = response.json()['request_id']
            # Poll for result
            self.client.get(f"/api/v1/predict/async/{request_id}")

    @task(10)  # 10% of traffic
    def get_explanation(self):
        """Get SHAP explanation."""
        self.client.post(
            "/api/v1/explain/explain",
            json=self._get_payload()
        )

    @task(10)  # 10% of traffic
    def what_if_analysis(self):
        """What-if analysis."""
        payload = {
            "base_features": self._get_payload(),
            "scenarios": [
                {"name": "Test", "changes": {"income": 90000}}
            ]
        }
        self.client.post("/api/v1/whatif/scenarios", json=payload)

    @task(5)  # 5% of traffic
    def verify_blockchain(self):
        """Verify prediction on blockchain."""
        # Use a random prediction ID (would be real in production)
        prediction_id = f"test-{random.randint(1000, 9999)}"
        self.client.get(
            f"/api/v1/blockchain/verify/{prediction_id}",
            name="/api/v1/blockchain/verify/[id]"
        )

    @task(3)  # 3% of traffic
    def get_model_card(self):
        """Get model card."""
        self.client.get("/api/v1/model-info/model-card")

    @task(2)  # 2% of traffic
    def health_check(self):
        """Health check."""
        self.client.get("/health")

    @staticmethod
    def _get_payload():
        """Generate random realistic payload."""
        return {
            "loan_amount": random.randint(10000, 100000),
            "age": random.randint(21, 65),
            "employment_length": random.randint(0, 30),
            "income": random.randint(30000, 200000),
            "debt_to_income": round(random.uniform(0.1, 0.6), 2),
            "credit_score": random.randint(600, 850),
            "num_credit_lines": random.randint(1, 15),
            "loan_purpose": random.choice([
                "debt_consolidation",
                "home_improvement",
                "business",
                "education"
            ]),
            "state_code": random.choice(["LA", "KA", "AB", "FC", "RI"])
        }


@events.test_start.add_listener
def on_test_start(environment, **kwargs):
    print("🚀 Starting complete system load test...")
    print("Testing: REST API + WebSockets + Kafka + Blockchain")


@events.test_stop.add_listener
def on_test_stop(environment, **kwargs):
    print("\n✅ Load test completed!")
    print(f"Total requests: {environment.stats.total.num_requests}")
    print(f"Failures: {environment.stats.total.num_failures}")
    print(f"Avg response time: {environment.stats.total.avg_response_time:.2f}ms")
    print(f"RPS: {environment.stats.total.current_rps:.2f}")
```
