## Understanding Advanced Feature Engineering

**Why advanced features?**
Simple features capture basic patterns, but advanced features can:
- Capture non-linear relationships
- Model feature interactions
- Incorporate domain knowledge
- Improve model performance by 5-15%

**Types of advanced features:**

1. **Polynomial Features:**
   - age² (squared age)
   - loan_amount³ (cubed loan amount)
   - Captures non-linear patterns

2. **Interaction Features:**
   - age × income
   - loan_amount × debt_ratio
   - Two features that matter together

3. **Domain-Specific:**
   - Credit utilization patterns
   - Payment-to-income trends
   - Risk concentration indices

"""
Advanced feature engineering with polynomial and interaction features.
"""

import pandas as pd
import numpy as np
from sklearn.preprocessing import PolynomialFeatures
from typing import List, Tuple

from src.utils.logger import logger


def create_polynomial_features(
    df: pd.DataFrame,
    features: List[str],
    degree: int = 2
) -> pd.DataFrame:
    """
    Create polynomial features for specified columns.

    Args:
        df: Input DataFrame
        features: List of column names to transform
        degree: Polynomial degree (2 = squared, 3 = cubed)

    Returns:
        DataFrame with additional polynomial features
    """
    logger.info(f"Creating polynomial features (degree={degree})...")

    df = df.copy()

    for feature in features:
        if feature not in df.columns:
            logger.warning(f"Feature '{feature}' not found, skipping")
            continue

        # Create squared feature
        if degree >= 2:
            df[f'{feature}_squared'] = df[feature] ** 2

        # Create cubed feature
        if degree >= 3:
            df[f'{feature}_cubed'] = df[feature] ** 3

    return df


def create_interaction_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Create interaction features between important variables.

    Interaction features capture combined effects of two features.
    Example: High income + low loan = very safe
    """
    logger.info("Creating interaction features...")

    df = df.copy()

    # 1. Income × Employment interactions
    df['income_employment_score'] = df['monthly_income'] * df['employment_stability']

    # 2. Loan × Risk interactions
    df['loan_credit_interaction'] = df['loan_amount'] * (850 - df['credit_score']) / 850

    # 3. Age × Experience interactions
    df['age_experience_product'] = df['age'] * df['years_employed']

    # 4. Debt × Income interactions
    df['debt_income_product'] = df['total_existing_debt'] * df['monthly_income']

    # 5. Savings × Loan ratio
    df['savings_loan_product'] = df['savings_balance'] * df['loan_amount']

    # 6. Income per year of experience
    df['income_per_year_experience'] = (
        df['monthly_income'] / np.maximum(df['years_employed'], 1)
    )

    # 7. Total financial capacity
    df['total_financial_capacity'] = (
        df['monthly_income'] * 12 +
        df['savings_balance'] -
        df['total_existing_debt']
    )

    # 8. Risk concentration
    df['risk_concentration'] = (
        df['loan_amount'] /
        np.maximum(df['total_financial_capacity'], 1)
    )

    return df


def create_advanced_credit_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Create advanced credit risk features based on industry best practices.
    """
    logger.info("Creating advanced credit risk features...")

    df = df.copy()

    # 1. Credit Utilization Score (0-1)
    # How much of available credit is being used
    estimated_credit_limit = df['monthly_income'] * 24  # 2 years income
    df['credit_utilization'] = (
        (df['total_existing_debt'] + df['loan_amount']) /
        np.maximum(estimated_credit_limit, 1)
    ).clip(0, 2)  # Cap at 200%

    # 2. Debt Service Coverage Ratio (DSCR)
    # Can they cover monthly payments?
    monthly_debt_payment = (
        df['total_existing_debt'] / 60 +  # Assume 5-year loans
        df['estimated_monthly_payment']
    )
    df['debt_service_coverage_ratio'] = (
        df['monthly_income'] / np.maximum(monthly_debt_payment, 1)
    )

    # 3. Loan-to-Value Equivalent
    # For personal loans, use savings as "collateral"
    df['loan_to_savings'] = (
        df['loan_amount'] / np.maximum(df['savings_balance'], 1)
    )

    # 4. Disposable Income After Loan
    df['disposable_income_after_loan'] = (
        df['monthly_income'] - df['estimated_monthly_payment']
    )

    # 5. Years to Pay Off Loan (with current income)
    df['years_to_payoff'] = (
        df['loan_amount'] /
        np.maximum(df['monthly_income'] * 0.3, 1) / 12  # Assume 30% income to loan
    )

    # 6. Income Stability Score
    # Combine employment status and tenure
    df['income_stability_score'] = (
        df['employment_stability'] *
        np.minimum(df['years_employed'] / 10, 1)  # Max out at 10 years
    )

    # 7. Financial Stress Index
    # Higher = more financial stress
    df['financial_stress_index'] = (
        df['debt_to_income_ratio'] * 0.4 +
        df['payment_to_income_ratio'] * 0.3 +
        (1 - df['credit_score'] / 850) * 0.2 +
        (1 - df['employment_stability']) * 0.1
    )

    # 8. Repayment Capacity Score
    # Can they repay? (higher = better)
    df['repayment_capacity_score'] = (
        (df['monthly_income'] / np.maximum(df['loan_amount'], 1)) * 0.3 +
        (df['savings_balance'] / np.maximum(df['loan_amount'], 1)) * 0.2 +
        df['employment_stability'] * 0.2 +
        (df['credit_score'] / 850) * 0.3
    )

    # 9. Default Probability Indicators
    # Binary flags for high-risk situations
    df['high_debt_burden'] = (df['debt_to_income_ratio'] > 0.5).astype(int)
    df['low_savings'] = (df['savings_balance'] < df['loan_amount'] * 0.1).astype(int)
    df['poor_credit'] = (df['credit_score'] < 580).astype(int)
    df['high_loan_to_income'] = (df['loan_to_income_ratio'] > 5).astype(int)

    # 10. Composite Risk Score
    # Weighted combination of all risk factors
    df['composite_risk_score'] = (
        df['financial_stress_index'] * 0.3 +
        (1 - df['repayment_capacity_score']) * 0.25 +
        df['credit_utilization'] * 0.15 +
        (df['high_debt_burden'] +
         df['low_savings'] +
         df['poor_credit'] +
         df['high_loan_to_income']) / 4 * 0.3
    )

    return df


def create_behavioral_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Create behavioral/temporal features.

    These simulate patterns we'd see in real data over time.
    """
    logger.info("Creating behavioral features...")

    df = df.copy()

    # 1. Loan Amount Trend (vs typical for this income level)
    # Group by income brackets to find typical loan amounts
    income_brackets = pd.cut(
        df['monthly_income'],
        bins=5,
        labels=['Very Low', 'Low', 'Medium', 'High', 'Very High']
    )

    typical_loan_by_income = df.groupby(income_brackets)['loan_amount'].median()
    df['income_bracket'] = income_brackets
    df['typical_loan_for_income'] = df['income_bracket'].map(typical_loan_by_income)

    df['loan_vs_typical'] = (
        (df['loan_amount'] - df['typical_loan_for_income']) /
        np.maximum(df['typical_loan_for_income'], 1)
    )

    # 2. Credit Score Trajectory Estimate
    # Estimate if credit score is improving or declining
    # Based on existing loans and savings
    df['estimated_credit_trajectory'] = (
        (df['savings_balance'] > df['monthly_income'] * 3).astype(int) -
        (df['num_existing_loans'] > 3).astype(int)
    )

    # 3. Life Stage Indicator
    # Different risk profiles at different life stages
    df['life_stage'] = 'Unknown'
    df.loc[df['age'] < 30, 'life_stage'] = 'Early Career'
    df.loc[(df['age'] >= 30) & (df['age'] < 45), 'life_stage'] = 'Established'
    df.loc[(df['age'] >= 45) & (df['age'] < 60), 'life_stage'] = 'Peak Earning'
    df.loc[df['age'] >= 60, 'life_stage'] = 'Pre-Retirement'

    # 4. Financial Maturity Score
    # Combination of age, savings, and credit behavior
    df['financial_maturity_score'] = (
        np.minimum(df['age'] / 60, 1) * 0.3 +
        np.minimum(df['savings_balance'] / (df['monthly_income'] * 6), 1) * 0.4 +
        (df['credit_score'] / 850) * 0.3
    )

    return df


def select_top_features(
    X: pd.DataFrame,
    y: pd.Series,
    n_features: int = 50
) -> List[str]:
    """
    Select top N most important features using correlation.

    Args:
        X: Feature DataFrame
        y: Target Series
        n_features: Number of features to select

    Returns:
        List of selected feature names
    """
    logger.info(f"Selecting top {n_features} features...")

    # Calculate correlation with target
    correlations = {}
    for col in X.columns:
        try:
            corr = abs(X[col].corr(y))
            if not np.isnan(corr):
                correlations[col] = corr
        except:
            pass

    # Sort by correlation
    sorted_features = sorted(correlations.items(), key=lambda x: x[1], reverse=True)

    # Select top N
    selected = [feat for feat, corr in sorted_features[:n_features]]

    logger.info(f"Selected {len(selected)} features")
    logger.info(f"Top 10 by correlation:")
    for feat, corr in sorted_features[:10]:
        logger.info(f"  {feat}: {corr:.4f}")

    return selected


def create_all_advanced_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Apply all advanced feature engineering techniques.

    Args:
        df: Input DataFrame with basic features

    Returns:
        DataFrame with all advanced features
    """
    logger.info("=" * 60)
    logger.info("CREATING ALL ADVANCED FEATURES")
    logger.info("=" * 60)

    # Start with input
    result = df.copy()
    initial_features = len(result.columns)

    # Create polynomial features for key variables
    poly_features = [
        'age', 'monthly_income', 'loan_amount',
        'credit_score', 'years_employed'
    ]
    result = create_polynomial_features(result, poly_features, degree=2)

    # Create interactions
    result = create_interaction_features(result)

    # Create advanced credit features
    result = create_advanced_credit_features(result)

    # Create behavioral features
    result = create_behavioral_features(result)

    final_features = len(result.columns)
    new_features = final_features - initial_features

    logger.info(f"\nFeature Engineering Complete:")
    logger.info(f"  Initial features: {initial_features}")
    logger.info(f"  Final features: {final_features}")
    logger.info(f"  New features created: {new_features}")
    logger.info("=" * 60)

    return result


def main():

    # Load processed data

    logger.info(f"Advanced features saved to {output_file}")


if __name__ == "__main__":
    main()

"""
Train models using advanced features.
"""

import pandas as pd
import xgboost as xgb
from sklearn.metrics import accuracy_score, roc_auc_score
import mlflow

from src.data.advanced_features import create_all_advanced_features, select_top_features
from src.utils.logger import logger


def train_with_advanced_features():
    """Train XGBoost with advanced feature set."""

    logger.info("Training with advanced features...")

    # Load base processed data
    X_train = pd.read_csv("data/processed/X_train.csv")
    X_test = pd.read_csv("data/processed/X_test.csv")
    y_train = pd.read_csv("data/processed/y_train.csv")['defaulted']
    y_test = pd.read_csv("data/processed/y_test.csv")['defaulted']

    # Create advanced features
    X_train_adv = create_all_advanced_features(X_train)
    X_test_adv = create_all_advanced_features(X_test)

    # Select top features (avoid curse of dimensionality)
    selected_features = select_top_features(X_train_adv, y_train, n_features=50)

    X_train_selected = X_train_adv[selected_features]
    X_test_selected = X_test_adv[selected_features]

    # Train with MLflow
    mlflow.set_experiment("loan-default-prediction-advanced")

    with mlflow.start_run(run_name="xgboost_advanced_features"):

        params = {
            'objective': 'binary:logistic',
            'max_depth': 6,
            'learning_rate': 0.1,
            'n_estimators': 200,
            'random_state': 42
        }

        mlflow.log_params(params)
        mlflow.log_param("num_features", len(selected_features))

        model = xgb.XGBClassifier(**params)
        model.fit(X_train_selected, y_train)

        # Evaluate
        y_pred = model.predict(X_test_selected)
        y_pred_proba = model.predict_proba(X_test_selected)[:, 1]

        accuracy = accuracy_score(y_test, y_pred)
        roc_auc = roc_auc_score(y_test, y_pred_proba)

        mlflow.log_metric("accuracy", accuracy)
        mlflow.log_metric("roc_auc", roc_auc)

        logger.info(f"Accuracy: {accuracy:.4f}")
        logger.info(f"ROC-AUC: {roc_auc:.4f}")

        # Compare with baseline
        baseline_auc = 0.9157  # From Day 008
        improvement = roc_auc - baseline_auc

        logger.info(f"\nImprovement over baseline: {improvement:+.4f}")

        mlflow.log_metric("improvement_over_baseline", improvement)


if __name__ == "__main__":
    train_with_advanced_features()

## Feature Engineering Best Practices

**DO:**
- ✅ Start with domain knowledge
- ✅ Create features based on business logic
- ✅ Test feature importance
- ✅ Remove redundant features
- ✅ Document feature meanings

**DON'T:**
- ❌ Create features blindly
- ❌ Use too many features (overfitting)
- ❌ Forget to scale/normalize
- ❌ Include target leakage
- ❌ Ignore missing values

## Troubleshooting

**Issue:** Too many features (slow training)
**Solution:** Use feature selection to keep top 50-100

**Issue:** Features have missing values
**Solution:** Fill with median or create "is_missing" flag

**Issue:** Polynomial features too large
**Solution:** Scale first, then create polynomials

**Issue:** No performance improvement
**Solution:** Check feature correlations, remove redundant ones
