## Understanding Data Preprocessing

**What is preprocessing?**
Machine learning models only understand numbers. We need to:
- Convert categorical data (text) to numbers
- Scale features to similar ranges
- Create new features that capture patterns
- Split data into training and testing sets

**Why is it important?**
- Models perform better with properly scaled data
- Engineered features can dramatically improve accuracy
- Train/test split prevents overfitting (memorizing data)

## Step 2: Create Feature Engineering Module

**File:** `src/data/feature_engineering.py`

```python
"""
Feature engineering for loan applications.
"""

import pandas as pd
import numpy as np
from typing import Tuple

from src.utils.logger import logger


def create_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Create engineered features from raw data.

    Feature engineering is the process of creating new features
    from existing ones to help the model learn better patterns.

    Args:
        df: Raw loan application DataFrame

    Returns:
        DataFrame with additional engineered features
    """
    logger.info("Creating engineered features...")

    df = df.copy()

    # 1. Debt-to-Income Ratio (DTI)
    # How much debt compared to annual income
    # Higher DTI = higher risk
    df['debt_to_income_ratio'] = (
        (df['total_existing_debt'] + df['loan_amount']) /
        (df['monthly_income'] * 12)
    )
    # Replace infinity with high value
    df['debt_to_income_ratio'] = df['debt_to_income_ratio'].replace(
        [np.inf, -np.inf], 10
    )

    # 2. Loan-to-Income Ratio
    # Requested loan amount compared to annual income
    df['loan_to_income_ratio'] = df['loan_amount'] / (df['monthly_income'] * 12)
    df['loan_to_income_ratio'] = df['loan_to_income_ratio'].replace(
        [np.inf, -np.inf], 10
    )

    # 3. Savings-to-Loan Ratio
    # How much savings compared to loan amount
    # Higher ratio = lower risk
    df['savings_to_loan_ratio'] = df['savings_balance'] / df['loan_amount']

    # 4. Monthly Payment Estimate
    # Rough estimate of monthly payment
    # Using simple interest calculation
    annual_rate = 0.20  # Assume 20% annual interest
    monthly_rate = annual_rate / 12
    n_months = df['requested_tenure_months']

    df['estimated_monthly_payment'] = (
        df['loan_amount'] *
        (monthly_rate * (1 + monthly_rate) ** n_months) /
        ((1 + monthly_rate) ** n_months - 1)
    )

    # 5. Payment-to-Income Ratio
    # Can they afford the monthly payment?
    df['payment_to_income_ratio'] = (
        df['estimated_monthly_payment'] / df['monthly_income']
    )
    df['payment_to_income_ratio'] = df['payment_to_income_ratio'].replace(
        [np.inf, -np.inf], 10
    )

    # 6. Age Groups (binning)
    # Sometimes categorical age groups work better than continuous age
    df['age_group'] = pd.cut(
        df['age'],
        bins=[0, 25, 35, 45, 55, 100],
        labels=['18-25', '26-35', '36-45', '46-55', '55+']
    )

    # 7. Income Groups
    df['income_group'] = pd.cut(
        df['monthly_income'],
        bins=[0, 100000, 250000, 500000, 1000000, np.inf],
        labels=['Low', 'Lower-Mid', 'Mid', 'Upper-Mid', 'High']
    )

    # 8. Loan Size Category
    df['loan_size'] = pd.cut(
        df['loan_amount'],
        bins=[0, 1000000, 5000000, 10000000, np.inf],
        labels=['Small', 'Medium', 'Large', 'Very Large']
    )

    # 9. Credit Score Category
    df['credit_score_category'] = pd.cut(
        df['credit_score'],
        bins=[0, 580, 670, 740, 800, 850],
        labels=['Poor', 'Fair', 'Good', 'Very Good', 'Excellent']
    )

    # 10. Employment Stability Score
    # Combination of employment status and years employed
    df['employment_stability'] = 0.0

    # Employed gets base score
    df.loc[df['employment_status'] == 'EMPLOYED', 'employment_stability'] = 0.5
    df.loc[df['employment_status'] == 'SELF_EMPLOYED', 'employment_stability'] = 0.3
    df.loc[df['employment_status'] == 'UNEMPLOYED', 'employment_stability'] = 0.0

    # Add bonus for years employed (max +0.5)
    df['employment_stability'] += np.minimum(df['years_employed'] / 20, 0.5)

    # 11. Financial Health Score
    # Composite score combining multiple factors
    # Normalized to 0-1 scale
    df['financial_health_score'] = (
        (df['credit_score'] / 850) * 0.4 +  # 40% weight
        (1 - np.minimum(df['debt_to_income_ratio'] / 5, 1)) * 0.3 +  # 30% weight
        (df['savings_balance'] / df['loan_amount']).clip(0, 1) * 0.2 +  # 20% weight
        df['employment_stability'] * 0.1  # 10% weight
    )

    # 12. Is High Risk Sector
    # Some sectors are more volatile
    high_risk_sectors = ['Oil & Gas', 'Agriculture', 'Hospitality', 'Media & Entertainment']
    df['is_high_risk_sector'] = df['employer_sector'].isin(high_risk_sectors).astype(int)

    # 13. Has Good Credit
    df['has_good_credit'] = (df['credit_score'] >= 670).astype(int)

    # 14. Is Young Borrower
    df['is_young_borrower'] = (df['age'] < 30).astype(int)

    # 15. Has High Loan Burden
    df['has_high_loan_burden'] = (df['num_existing_loans'] >= 3).astype(int)

    logger.info(f"Created {len(df.columns) - len(df.columns)} new features")
    logger.info(f"Total features now: {len(df.columns)}")

    return df


def get_feature_names() -> Tuple[list, list]:
    """
    Get lists of numerical and categorical features.

    Returns:
        (numerical_features, categorical_features) tuple
    """
    numerical_features = [
        'age',
        'monthly_income',
        'loan_amount',
        'years_employed',
        'num_existing_loans',
        'total_existing_debt',
        'savings_balance',
        'credit_score',
        'requested_tenure_months',
        # Engineered features
        'debt_to_income_ratio',
        'loan_to_income_ratio',
        'savings_to_loan_ratio',
        'estimated_monthly_payment',
        'payment_to_income_ratio',
        'employment_stability',
        'financial_health_score',
        'is_high_risk_sector',
        'has_good_credit',
        'is_young_borrower',
        'has_high_loan_burden',
    ]

    categorical_features = [
        'gender',
        'marital_status',
        'education_level',
        'employment_status',
        'employer_sector',
        'loan_purpose',
        'state',
        'has_existing_loans',
        'has_savings_account',
        # Engineered categorical features
        'age_group',
        'income_group',
        'loan_size',
        'credit_score_category',
    ]

    return numerical_features, categorical_features
```

**Feature Engineering Explained:**

1. **Ratio Features:** Combine two features (e.g., debt/income)
2. **Binning:** Convert continuous values to categories
3. **Composite Scores:** Weighted combinations of features
4. **Boolean Flags:** Binary yes/no features
5. **Domain Knowledge:** Features based on financial expertise

## Step 3: Create Preprocessing Pipeline

**File:** `src/data/preprocessor.py`

```python
"""
Data preprocessing pipeline for ML models.
"""

import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler, LabelEncoder
from pathlib import Path
import pickle
from typing import Tuple

from src.data.feature_engineering import create_features, get_feature_names
from src.utils.logger import logger


class LoanDataPreprocessor:
    """
    Preprocessing pipeline for loan application data.

    Handles:
    - Feature engineering
    - Categorical encoding
    - Feature scaling
    - Train/test split
    """

    def __init__(self):
        self.scaler = StandardScaler()
        self.label_encoders = {}
        self.feature_names = None
        self.is_fitted = False

    def preprocess(
        self,
        df: pd.DataFrame,
        test_size: float = 0.2,
        random_state: int = 42
    ) -> Tuple[pd.DataFrame, pd.DataFrame, pd.Series, pd.Series]:
        """
        Complete preprocessing pipeline.

        Args:
            df: Raw loan application DataFrame
            test_size: Proportion of data for testing (0.2 = 20%)
            random_state: Random seed for reproducibility

        Returns:
            X_train, X_test, y_train, y_test
        """
        logger.info("Starting preprocessing pipeline...")

        # 1. Feature Engineering
        logger.info("Step 1/5: Feature engineering")
        df = create_features(df)

        # 2. Select features
        logger.info("Step 2/5: Selecting features")
        numerical_features, categorical_features = get_feature_names()

        # Check for missing features
        all_features = numerical_features + categorical_features
        missing_features = [f for f in all_features if f not in df.columns]
        if missing_features:
            logger.warning(f"Missing features: {missing_features}")
            # Remove missing features from lists
            numerical_features = [f for f in numerical_features if f in df.columns]
            categorical_features = [f for f in categorical_features if f in df.columns]

        # 3. Encode categorical features
        logger.info("Step 3/5: Encoding categorical features")
        df_encoded = df.copy()

        for col in categorical_features:
            if col not in self.label_encoders:
                self.label_encoders[col] = LabelEncoder()
                df_encoded[col] = self.label_encoders[col].fit_transform(
                    df[col].astype(str)
                )
            else:
                df_encoded[col] = self.label_encoders[col].transform(
                    df[col].astype(str)
                )

        # 4. Prepare X (features) and y (target)
        logger.info("Step 4/5: Preparing feature matrix and target")
        feature_columns = numerical_features + categorical_features
        X = df_encoded[feature_columns]
        y = df_encoded['defaulted']

        self.feature_names = feature_columns
        logger.info(f"Total features: {len(feature_columns)}")

        # 5. Train/Test Split
        logger.info(f"Step 5/5: Splitting data (test_size={test_size})")
        X_train, X_test, y_train, y_test = train_test_split(
            X, y,
            test_size=test_size,
            random_state=random_state,
            stratify=y  # Maintain same default rate in train/test
        )

        logger.info(f"Training set: {len(X_train)} samples")
        logger.info(f"Test set: {len(X_test)} samples")
        logger.info(f"Training default rate: {y_train.mean():.2%}")
        logger.info(f"Test default rate: {y_test.mean():.2%}")

        # 6. Scale numerical features
        logger.info("Scaling numerical features...")
        X_train_scaled = X_train.copy()
        X_test_scaled = X_test.copy()

        # Only scale numerical features
        X_train_scaled[numerical_features] = self.scaler.fit_transform(
            X_train[numerical_features]
        )
        X_test_scaled[numerical_features] = self.scaler.transform(
            X_test[numerical_features]
        )

        self.is_fitted = True
        logger.info("Preprocessing complete!")

        return X_train_scaled, X_test_scaled, y_train, y_test

    def save(self, output_dir: str = "data/processed"):
        """Save preprocessor state (scalers, encoders)."""
        output_path = Path(output_dir)
        output_path.mkdir(parents=True, exist_ok=True)

        preprocessor_file = output_path / "preprocessor.pkl"

        with open(preprocessor_file, 'wb') as f:
            pickle.dump({
                'scaler': self.scaler,
                'label_encoders': self.label_encoders,
                'feature_names': self.feature_names,
                'is_fitted': self.is_fitted
            }, f)

        logger.info(f"Preprocessor saved to {preprocessor_file}")

    def load(self, input_dir: str = "data/processed"):
        """Load preprocessor state."""
        preprocessor_file = Path(input_dir) / "preprocessor.pkl"

        with open(preprocessor_file, 'rb') as f:
            state = pickle.load(f)

        self.scaler = state['scaler']
        self.label_encoders = state['label_encoders']
        self.feature_names = state['feature_names']
        self.is_fitted = state['is_fitted']

        logger.info(f"Preprocessor loaded from {preprocessor_file}")


def main():
    """Main function to preprocess and save data."""

    # Load raw data
    logger.info("Loading raw data...")
    data_file = Path("data/raw/loan_applications.csv")

    if not data_file.exists():
        logger.error(f"Data file not found: {data_file}")
        logger.info("Run 'python src/data/generate_data.py' first")
        return

    df = pd.read_csv(data_file)
    logger.info(f"Loaded {len(df)} loan applications")

    # Preprocess
    preprocessor = LoanDataPreprocessor()
    X_train, X_test, y_train, y_test = preprocessor.preprocess(df)

    # Save processed data
    logger.info("Saving processed data...")
    output_dir = Path("data/processed")
    output_dir.mkdir(parents=True, exist_ok=True)

    X_train.to_csv(output_dir / "X_train.csv", index=False)
    X_test.to_csv(output_dir / "X_test.csv", index=False)
    y_train.to_csv(output_dir / "y_train.csv", index=False)
    y_test.to_csv(output_dir / "y_test.csv", index=False)

    logger.info("Processed data saved:")
    logger.info(f"  - {output_dir / 'X_train.csv'}")
    logger.info(f"  - {output_dir / 'X_test.csv'}")
    logger.info(f"  - {output_dir / 'y_train.csv'}")
    logger.info(f"  - {output_dir / 'y_test.csv'}")

    # Save preprocessor
    preprocessor.save()

    # Print summary
    logger.info("\n" + "=" * 60)
    logger.info("PREPROCESSING SUMMARY")
    logger.info("=" * 60)
    logger.info(f"Total features: {len(preprocessor.feature_names)}")
    logger.info(f"Training samples: {len(X_train)}")
    logger.info(f"Test samples: {len(X_test)}")
    logger.info(f"Train default rate: {y_train.mean():.2%}")
    logger.info(f"Test default rate: {y_test.mean():.2%}")
    logger.info("\nFeature list:")
    for i, feat in enumerate(preprocessor.feature_names, 1):
        logger.info(f"  {i:2d}. {feat}")
    logger.info("=" * 60)


if __name__ == "__main__":
    main()
```

**Preprocessing Concepts Explained:**

1. **StandardScaler:** Scales features to mean=0, std=1
   - Prevents features with large values from dominating
   - Example: loan_amount (millions) vs age (tens)

2. **LabelEncoder:** Converts categories to numbers
   - "EMPLOYED" → 0, "SELF_EMPLOYED" → 1, "UNEMPLOYED" → 2

3. **Train/Test Split:** Separate data for training and evaluation
   - Train: 80% - model learns from this
   - Test: 20% - evaluate model on unseen data
   - Prevents overfitting

4. **Stratify:** Maintains same class distribution in train/test
   - If 30% defaults overall, both train and test have ~30%

```python
# ... existing imports ...
from pathlib import Path
import pandas as pd

# ... existing code ...


@router.get("/preprocessing-info")
async def get_preprocessing_info():
    """
    Get information about preprocessed data.

    Returns details about train/test split and features.
    """
    processed_dir = Path("data/processed")

    # Check if preprocessed data exists
    required_files = ['X_train.csv', 'X_test.csv', 'y_train.csv', 'y_test.csv']
    missing_files = [f for f in required_files if not (processed_dir / f).exists()]

    if missing_files:
        raise HTTPException(
            status_code=404,
            detail=f"Preprocessed data not found. Missing files: {missing_files}. Run preprocessing first."
        )

    # Load data
    X_train = pd.read_csv(processed_dir / "X_train.csv")
    X_test = pd.read_csv(processed_dir / "X_test.csv")
    y_train = pd.read_csv(processed_dir / "y_train.csv")
    y_test = pd.read_csv(processed_dir / "y_test.csv")

    info = {
        "training_samples": len(X_train),
        "test_samples": len(X_test),
        "total_samples": len(X_train) + len(X_test),
        "num_features": len(X_train.columns),
        "train_default_rate": round(y_train['defaulted'].mean() * 100, 2),
        "test_default_rate": round(y_test['defaulted'].mean() * 100, 2),
        "features": list(X_train.columns),
        "feature_statistics": {
            "numerical_features": X_train.select_dtypes(include=[np.number]).columns.tolist(),
            "num_numerical": len(X_train.select_dtypes(include=[np.number]).columns)
        }
    }

    return info
```

## Understanding Train/Test Split

**Why split?**
- **Training set:** Model learns patterns
- **Test set:** Evaluate on unseen data
- Prevents overfitting (memorizing instead of learning)

**Example of overfitting:**
```
Model sees: "Adebayo, age 35, ₦5M loan → defaulted"
Overfitting: Remembers "Adebayo = default"
Good learning: Learns "35 years old + ₦5M loan pattern"
```

**Test set proves the model learned patterns, not memorized data.**
